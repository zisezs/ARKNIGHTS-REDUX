# ============================================================================
# 深池（DUB）专属兵种「死火行尸」
# ----------------------------------------------------------------------------
# 与 arf_special_support_units.txt 那批支援连的区别：这一支是【营级单位】。
#   用户口径「可以无限招募」在支援连里做不到——同一 same_support_type 每师只能挂一连，
#   所以做成 regiments 里可叠任意数量的营，才是字面意义的无限招募。
#
# 三条改动过的账（其余全部与普通步兵一致）：
#   · 装备 need = 1 件（普通步兵 100 件）
#   · 人力 manpower = 10,000（普通步兵 1,000 的十倍）
#   · 训练 training_time = 30 天（普通步兵 90 天）
#
# 面板为什么【不写】任何攻击修正（2026-09-26 实测钉死的口径）：
#   营级面板 = 装备属性 × (1 + sub_unit 的 soft_attack/defense/breakthrough)，
#   与 need 数量无关——need 只决定这一营吃几件装备。
#   证据：本模组基础步兵装备 软攻3/防御20/突破2；need=1 时曾按"÷100"反推写 99，
#   游戏里显示 300/2000/200（正好一百倍），说明 /100 那一步不存在。
#   所以想要"面板与普通步兵一致"，正确做法是【不写修正】：3/20/2 自动相等，
#   且在更高装备档位下也始终相等（两边乘的是同一件装备的属性）。
#
# 解锁：common/technologies/DUB_dead_fire_tech.txt（allow = { always = no }），
#   由人力法案【死火复生】的 on_add 用 set_technology 授予；科技一旦研究不会被回收，
#   所以换掉法案只是失去加成，已经编成军的行尸不会从天上消失。
# ============================================================================

sub_units = {

	dub_dead_fire_walker = {
		abbreviation = "DFW"
		sprite = dub_dead_fire_walker			# 专属图标，注册在 interface/subuniticons.gfx
												# （生成器 docs/gen_dub_walker_icon.js → 152x42 左帧图形＋右帧通用底板）
		map_icon_category = infantry

		priority = 600				# 编制界面里排在步兵旁边（正常位次）
		ai_priority = 900				# 远高于原版步兵的 200：AI 自动设计时先填行尸。
												# 配合 common/ai_strategy/DUB.txt（template_prio +250）与
												# common/ai_templates/DUB_templates.txt（12 营纯行尸的 target_template），
												# 把人机深池钉成"死火军当主力"——这是用户要的形态，不是要纠的偏。
		active = no						# 与其它专属兵种同口径：只由科技 enable_subunits 点亮

		type = {
			infantry
		}

		group = infantry

		categories = {
			category_front_line
			category_light_infantry
			category_all_infantry
			category_army
			category_infantry_and_bicycle
		}

		combat_width = 2

		# ── 尺寸定义：与普通步兵逐项相同，只有人力是十倍 ──
		max_strength = 25
		max_organisation = 60
		default_morale = 0.3
		manpower = 10000				# 步兵 1000 的十倍：一具尸体不够，就十具叠成一营

		# ── 杂项：训练 30 天（步兵 90）；补给按用户要求减半 ──
		training_time = 30
		suppression = 1.5
		weight = 0.5
		supply_consumption = 0.03		# 步兵 0.06 的一半：尸体不吃饭，也不喝水

		# ── 面板：突破 +10，攻击与防御不写修正（= 与普通步兵相同） ──
		# 营级面板 = 装备属性 × (1 + 修正)，与 need 数量无关。
		# 本模组基础步兵装备突破 = 2 ⇒ 修正 5 让面板变成 12，即"比步兵高 10"。
		# 注意这条换算是按 1076 型标定的：装备档位越高，+10 会变成按比例放大
		# （例如某档突破 = 3 时面板会是 18，等于 +15）——引擎只有百分比修正，没有加法字段。
		breakthrough = 5

		need = {
			infantry_equipment = 1		# 一支铳传给下一个站起来的人
		}
	}

}
