#所有执政党意识形态为共产主义或社会主义的国家

	input = game:all_countries
	name = collection_communist_socialist_ruling_party_countries
	operators = {
		limit = {
			OR = {
				has_government = communism
				has_government = socialism
			}
		}
	}
}
