# ==============================================================================
#  大炎·珍玩集册 —— scripted_gui
#  · YAN_Collection_Entry : 决议分类页里的「开卷览珍」入口面板
#  · YAN_Collection_GUI   : 全屏图鉴面板
#
#  面板状态：
#    YAN_col_index        1..294  当前浏览的收藏品
#    YAN_col_cur_tier     0..4    当前件品阶（0 = 未收藏）
#    YAN_col_cur_cat      1..9    当前件分类（左侧页签高亮）
#    YAN_col_result_open  0/1     开箱结果卡是否展示
#    YAN_col_result_kind  0=新收藏 / 1=升阶 / 2=空手
#
#  ⚠️ 控件名必须与 YAN_Collection.gui 完全一致：拼错不会报错，只会得到死按钮。
#     tools（E:/tmp/yan_col/check.js）会同时读两边做对账。
# ==============================================================================

scripted_gui = {

	# ---------------- 决议页入口 ----------------
	YAN_Collection_Entry = {
		context_type = decision_category
		window_name = "YAN_Collection_Entry_GUI"

		visible = {
			always = yes
		}

		effects = {
			YAN_Col_Entry_Btn_click = {
				YAN_Collection_Open = yes
			}
			# 决议页直接看总加成：先开面板旗标，再切到总览视图
			YAN_Col_Entry_Sum_Btn_click = {
				YAN_Collection_Show_Summary = yes
			}
		}

		triggers = {
			YAN_Col_Entry_Btn_click_enabled = {
				NOT = { has_country_flag = YAN_col_open }
			}
			YAN_Col_Entry_Sum_Btn_click_enabled = {
				NOT = { has_country_flag = YAN_col_open }
			}
			# 十段刻度：每段约 29.4 件，与图鉴主面板顶部的进度条同一口径
			YAN_Col_Entry_Pip_1_visible = { check_variable = { YAN_col_total > 29 } }
			YAN_Col_Entry_Pip_2_visible = { check_variable = { YAN_col_total > 58 } }
			YAN_Col_Entry_Pip_3_visible = { check_variable = { YAN_col_total > 88 } }
			YAN_Col_Entry_Pip_4_visible = { check_variable = { YAN_col_total > 117 } }
			YAN_Col_Entry_Pip_5_visible = { check_variable = { YAN_col_total > 146 } }
			YAN_Col_Entry_Pip_6_visible = { check_variable = { YAN_col_total > 176 } }
			YAN_Col_Entry_Pip_7_visible = { check_variable = { YAN_col_total > 205 } }
			YAN_Col_Entry_Pip_8_visible = { check_variable = { YAN_col_total > 235 } }
			YAN_Col_Entry_Pip_9_visible = { check_variable = { YAN_col_total > 264 } }
			YAN_Col_Entry_Pip_10_visible = { check_variable = { YAN_col_total > 293 } }
		}
	}


	# ---------------- 图鉴面板 ----------------
	YAN_Collection_GUI = {
		context_type = player_context
		window_name = "YAN_Collection_Window"

		visible = {
			is_ai = no
			has_country_flag = YAN_col_open
			NOT = { check_variable = { YAN_col_view = 1 compare = equals } }
		}

		effects = {

			YAN_Col_Close_click = {
				YAN_Collection_Close = yes
			}

			YAN_Col_Prev_click = {
				YAN_Collection_Prev = yes
			}
			YAN_Col_Next_click = {
				YAN_Collection_Next = yes
			}

			# ---- 左侧分类页签：跳转到该类的第一件，并把寻珍目标切到该类 ----
			YAN_Col_Tab_1_click = { YAN_Collection_Goto_1 = yes }
			YAN_Col_Tab_2_click = { YAN_Collection_Goto_2 = yes }
			YAN_Col_Tab_3_click = { YAN_Collection_Goto_3 = yes }
			YAN_Col_Tab_4_click = { YAN_Collection_Goto_4 = yes }
			YAN_Col_Tab_5_click = { YAN_Collection_Goto_5 = yes }
			YAN_Col_Tab_6_click = { YAN_Collection_Goto_6 = yes }
			YAN_Col_Tab_7_click = { YAN_Collection_Goto_7 = yes }
			YAN_Col_Tab_8_click = { YAN_Collection_Goto_8 = yes }
			YAN_Col_Tab_9_click = { YAN_Collection_Goto_9 = yes }

			# ---- 左下列筛选：全部 / 只看未收藏 / 只看可升阶 ----
			YAN_Col_F_All_click = { YAN_Collection_Filter_All = yes }
			YAN_Col_F_Own_click = { YAN_Collection_Filter_Unowned = yes }
			YAN_Col_F_Up_click = { YAN_Collection_Filter_Upgradable = yes }

			# ---- 面板内寻珍：花 25 PP，从当前分类里抽一件 ----
			YAN_Col_Draw_click = {
				if = {
					limit = { has_political_power > 24 }
					add_political_power = -25
					YAN_Collection_Draw = yes
				}
			}

			# ---- 切到「集册总加成」窗口 ----
			YAN_Col_Summary_click = {
				YAN_Collection_Show_Summary = yes
			}

			YAN_Col_Result_Ok_click = {
				YAN_Collection_Close_Result = yes
			}

			# 点结果卡外的压暗层也能收起
			YAN_Col_Result_click = {
				YAN_Collection_Close_Result = yes
			}
		}

		triggers = {

			# ---- 页签高亮：当前件所属分类 ----
			YAN_Col_Tab_1_Sel_visible = { check_variable = { YAN_col_cur_cat = 1 compare = equals } }
			YAN_Col_Tab_2_Sel_visible = { check_variable = { YAN_col_cur_cat = 2 compare = equals } }
			YAN_Col_Tab_3_Sel_visible = { check_variable = { YAN_col_cur_cat = 3 compare = equals } }
			YAN_Col_Tab_4_Sel_visible = { check_variable = { YAN_col_cur_cat = 4 compare = equals } }
			YAN_Col_Tab_5_Sel_visible = { check_variable = { YAN_col_cur_cat = 5 compare = equals } }
			YAN_Col_Tab_6_Sel_visible = { check_variable = { YAN_col_cur_cat = 6 compare = equals } }
			YAN_Col_Tab_7_Sel_visible = { check_variable = { YAN_col_cur_cat = 7 compare = equals } }
			YAN_Col_Tab_8_Sel_visible = { check_variable = { YAN_col_cur_cat = 8 compare = equals } }
			YAN_Col_Tab_9_Sel_visible = { check_variable = { YAN_col_cur_cat = 9 compare = equals } }

			# ---- 未收藏：压一层磨砂锁罩 ----
			YAN_Col_Lock_visible = { check_variable = { YAN_col_cur_owned = 0 compare = equals } }

			# ---- 进度刻度：10 段，每段在收藏数越过 1/10 处出现 ----
			YAN_Col_Pip_1_visible = { check_variable = { YAN_col_total > 29 } }
			YAN_Col_Pip_2_visible = { check_variable = { YAN_col_total > 58 } }
			YAN_Col_Pip_3_visible = { check_variable = { YAN_col_total > 88 } }
			YAN_Col_Pip_4_visible = { check_variable = { YAN_col_total > 117 } }
			YAN_Col_Pip_5_visible = { check_variable = { YAN_col_total > 146 } }
			YAN_Col_Pip_6_visible = { check_variable = { YAN_col_total > 176 } }
			YAN_Col_Pip_7_visible = { check_variable = { YAN_col_total > 205 } }
			YAN_Col_Pip_8_visible = { check_variable = { YAN_col_total > 235 } }
			YAN_Col_Pip_9_visible = { check_variable = { YAN_col_total > 264 } }
			YAN_Col_Pip_10_visible = { check_variable = { YAN_col_total > 293 } }

			# ---- 筛选高亮 / 可用性 ----
			YAN_Col_F_All_Sel_visible = { NOT = { check_variable = { YAN_col_filter_mode > 0 } } }
			YAN_Col_F_Own_Sel_visible = { check_variable = { YAN_col_filter_mode = 1 compare = equals } }
			YAN_Col_F_Up_Sel_visible = { check_variable = { YAN_col_filter_mode = 2 compare = equals } }
			YAN_Col_F_Own_click_enabled = { YAN_col_any_unowned = yes }
			YAN_Col_F_Up_click_enabled = { YAN_col_any_upgradable = yes }

			# 当前筛选下没有可跳的目标 → 灰掉上/下一件
			YAN_Col_Prev_click_enabled = { YAN_col_filter_has_matches = yes }
			YAN_Col_Next_click_enabled = { YAN_col_filter_has_matches = yes }

			# ---- 寻珍按钮 ----
			YAN_Col_Draw_click_enabled = {
				has_political_power > 24
				YAN_col_draw_cat_has_candidates = yes
			}

			# ---- 结果卡 ----
			YAN_Col_Result_visible = { check_variable = { YAN_col_result_open = 1 compare = equals } }
			YAN_Col_Result_Card_visible = { check_variable = { YAN_col_result_open = 1 compare = equals } }
			YAN_Col_Result_Title_visible = { check_variable = { YAN_col_result_open = 1 compare = equals } }
			YAN_Col_Result_Ok_visible = { check_variable = { YAN_col_result_open = 1 compare = equals } }
			YAN_Col_Result_Ok_Text_visible = { check_variable = { YAN_col_result_open = 1 compare = equals } }

			# 抽到东西才显示图标/名字；空手只显示提示
			YAN_Col_Result_Icon_visible = {
				check_variable = { YAN_col_result_open = 1 compare = equals }
				NOT = { check_variable = { YAN_col_result_kind = 2 compare = equals } }
			}
			YAN_Col_Result_Frame_visible = {
				check_variable = { YAN_col_result_open = 1 compare = equals }
				NOT = { check_variable = { YAN_col_result_kind = 2 compare = equals } }
			}
			YAN_Col_Result_Kind_visible = {
				check_variable = { YAN_col_result_open = 1 compare = equals }
				NOT = { check_variable = { YAN_col_result_kind = 2 compare = equals } }
			}
			YAN_Col_Result_Name_visible = {
				check_variable = { YAN_col_result_open = 1 compare = equals }
				NOT = { check_variable = { YAN_col_result_kind = 2 compare = equals } }
			}
			YAN_Col_Result_Empty_visible = {
				check_variable = { YAN_col_result_open = 1 compare = equals }
				check_variable = { YAN_col_result_kind = 2 compare = equals }
			}
		}

		properties = {
			# 当前件的原画 / 稀有度外框（结果卡复用同一套链，因为抽中后 index 已挪过去）
			YAN_Col_Icon = { image = "[YAN_GetColIcon]" }
			YAN_Col_Frame = { image = "[YAN_GetColFrame]" }
			YAN_Col_Result_Icon = { image = "[YAN_GetColIcon]" }
			YAN_Col_Result_Frame = { image = "[YAN_GetColFrame]" }
		}
	}


	# ---------------- 集册总加成窗口 ----------------
	# 与图鉴窗口靠 YAN_col_view 互斥，所以这里的控件不需要逐个加显隐触发器。
	# 数值不走链：本地化里用 [?变量] 直接打印 Recalc 拆出的整数位/十分位。
	YAN_Collection_Summary_GUI = {
		context_type = player_context
		window_name = "YAN_Collection_Summary_Window"

		visible = {
			is_ai = no
			has_country_flag = YAN_col_open
			check_variable = { YAN_col_view = 1 compare = equals }
		}

		effects = {
			# 每次打开都重算一遍：不维护累计量，就不会和旗标/精神失步
			YAN_Col_Sum_Back_click = {
				YAN_Collection_Show_Item = yes
			}
			YAN_Col_Sum_Close_click = {
				YAN_Collection_Show_Item = yes
			}
		}
	}
}
