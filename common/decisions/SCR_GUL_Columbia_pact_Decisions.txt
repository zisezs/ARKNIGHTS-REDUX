		icon = GFX_decision_generic_political_discourse
		cost = 100
		fire_only_once = yes

		allowed = {
			OR = {
				original_tag = SCR
				original_tag = GUL
			}
		}

		visible = {
			OR = {
				original_tag = SCR
				original_tag = GUL

			}
		}

				}
			}
		}

		complete_effect = {
			custom_effect_tooltip = SCR_GUL_join_columbia_pact_faction_tt
			hidden_effect = {
				CLB = {
					faction_leader = {
						add_to_faction = ROOT
					}
				}
			}
		}

		ai_will_do = {
			base = 0
		}
	}
