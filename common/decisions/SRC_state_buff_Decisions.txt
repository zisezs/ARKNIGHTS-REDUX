SRC_family_purge_category = {

	# 宪兵、法警与税务局同时进门，一次只拔一根钉子
	SRC_dec_crack_down_families = {
		icon = GFX_decision_generic_police_action

		allowed = {
			original_tag = SRC
		}

		available = {
			any_owned_state = {
				limit = {
					has_dynamic_modifier = {
						modifier = SRC_dyn_family_entrenchment
					}
				}
			}
		}

		cost = 150
		fire_only_once = no
		days_remove = 30

		complete_effect = {
			custom_effect_tooltip = SRC_tt_crackdown_going
			random_owned_state = {
				limit = {
					has_dynamic_modifier = {
						modifier = SRC_dyn_family_entrenchment
					}
				}
				remove_dynamic_modifier = {
					modifier = SRC_dyn_family_entrenchment
				}
			}
		}

		ai_will_do = {
			base = 10
		}
	}

}
