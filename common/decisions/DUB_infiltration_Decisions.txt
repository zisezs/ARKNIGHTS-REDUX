DUB_infiltration_decisions = {
	DUB_infiltrated_state = {
		icon = GFX_decision_generic_political_discourse

		visible = {
			FROM = {
				NOT = {
					is_controlled_by = ROOT
				}
				NOT = {
					has_dynamic_modifier = {
						modifier = DUB_infiltrated_state_3_modifier
						scope = DUB
					}
				}
			}
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = DUB_infiltrated_state_num_available_tt
				check_variable = {
					var = DUB_infiltrated_state_num
					value = DUB_infiltrated_state_max_num
					compare = less_than
				}
			}
			FROM = {
				controller = {
					NOT = {
						has_war_with = ROOT
					}
				}
			}
		}

		state_target = yes
		target_trigger = {
			FROM = {
				controller = {
					DUB_victorian_country = yes
				}
				any_neighbor_state = {
					OR = {
						is_controlled_by_ROOT_or_subject = yes
						has_dynamic_modifier = {
							modifier = DUB_infiltrated_state_1_modifier
							scope = DUB
						}
						has_dynamic_modifier = {
							modifier = DUB_infiltrated_state_2_modifier
							scope = DUB
						}
						has_dynamic_modifier = {
							modifier = DUB_infiltrated_state_3_modifier
							scope = DUB
						}
					}
				}
			}
		}
		on_map_mode = map_and_decisions_view
		fire_only_once = no

		cost = DUB_infiltration_cost
		days_remove = 30

		complete_effect = {
			DUB = {
				add_to_variable = { DUB_infiltrated_state_num = 1 }
			}
		}

		remove_effect = {
			FROM = {
				controller = {
					if = {
						limit = {
							NOT = {
								has_country_flag = NOT_DUB_counter_infiltration
							}
						}
						set_country_flag = NOT_DUB_counter_infiltration
					}
				}
				if = {
					limit = {
						has_dynamic_modifier = {
							modifier = DUB_infiltrated_state_2_modifier
							scope = DUB
						}
					}
					remove_dynamic_modifier = {
						modifier = DUB_infiltrated_state_2_modifier
					}
					add_dynamic_modifier = {
						modifier = DUB_infiltrated_state_3_modifier
						scope = DUB
					}
				}
				else_if = {
					limit = {
						has_dynamic_modifier = {
							modifier = DUB_infiltrated_state_1_modifier
							scope = DUB
						}
					}
					remove_dynamic_modifier = {
						modifier = DUB_infiltrated_state_1_modifier
					}
					add_dynamic_modifier = {
						modifier = DUB_infiltrated_state_2_modifier
						scope = DUB
					}
				}
				else = {
					add_dynamic_modifier = {
						modifier = DUB_infiltrated_state_1_modifier
						scope = DUB
					}
				}
				DUB = {
					add_manpower = 5000
				}
			}

			DUB = {
				add_to_variable = { DUB_infiltrated_state_num = -1 }
			}
		}

		cancel_trigger = {
			FROM = {
				is_controlled_by = ROOT
			}
		}

		cancel_effect = {
			DUB = {
				add_to_variable = { DUB_infiltrated_state_num = -1 }
			}
		}

		ai_will_do = {
			base = 3
			modifier = {
				factor = 2
				FROM = {
					has_dynamic_modifier = {
						modifier = DUB_infiltrated_state_2_modifier
						scope = DUB
					}
				}
			}
		}
	}
}
