OMN_Civil_War_Decisions = {

	OMN_battle_round = {
		icon = GFX_decision_generic_political_discourse

		available = {
			always = no
		}

		activation = {
			OR = {
				has_country_flag = OMN_civil_war_chantashaw
				has_country_flag = OMN_civil_war_aryusz
			}
		}

		is_good = no

		days_mission_timeout = 30

		cancel_trigger = {
			OR = {
				custom_trigger_tooltip = {
					tooltip = OMN_chantashaw_battle_round_check_tt
					check_variable = { OMN_chantashaw_advantage > 8	}
				}
				custom_trigger_tooltip = {
					tooltip = OMN_aryusz_battle_round_check_tt
					check_variable = { OMN_aryusz_advantage > 8	}
				}
			}
		}
		
		cancel_effect = {
			add_to_variable = { OMN_round = 1 }
			if = {
				limit = {
					check_variable = { OMN_chantashaw_advantage > 8	}
				}
				add_to_variable = { OMN_chantashaw_round_victory = 1 }
				set_temp_variable = { chantashaw_manpower = 500 }
				OMN_Chantashaw_Manpower_Change = yes
				set_temp_variable = { aryusz_manpower = 800 }
				OMN_Aryusz_Manpower_Change = yes
			}
			if = {
				limit = {
					check_variable = { OMN_aryusz_advantage > 8	}
				}
				add_to_variable = { OMN_aryusz_round_victory = 1 }
				set_temp_variable = { chantashaw_manpower = 800 }
				OMN_Chantashaw_Manpower_Change = yes
				set_temp_variable = { aryusz_manpower = 500 }
				OMN_Aryusz_Manpower_Change = yes
			}
			set_variable = { OMN_chantashaw_advantage = 5 }
			set_variable = { OMN_aryusz_advantage = 4 }
			if = {
				limit = {
					check_variable = { OMN_chantashaw_round_victory > 2	}
				}
				set_country_flag = OMN_chantashaw_victory
				mark_focus_tree_layout_dirty = yes
				hidden_effect = {
					country_event = {
						id = decision_omn.1
					}
				}
				clr_country_flag = OMN_civil_war_chantashaw
				clr_country_flag = OMN_civil_war_aryusz
			}
			if = {
				limit = {
					check_variable = { OMN_aryusz_round_victory > 2	}
				}
				set_country_flag = OMN_aryusz_victory
				mark_focus_tree_layout_dirty = yes
				hidden_effect = {
					country_event = {
						id = decision_omn.2
					}
				}
				clr_country_flag = OMN_civil_war_chantashaw
				clr_country_flag = OMN_civil_war_aryusz
			}
			hidden_effect = {
				if = {
					limit = {
						OR = {
							has_country_flag = OMN_civil_war_chantashaw
							has_country_flag = OMN_civil_war_aryusz
						}
					}
					activate_mission = OMN_battle_round
				}
			}
		}

		timeout_effect = {
			add_to_variable = { OMN_round = 1 }
			if = {
				limit = {
					check_variable = { OMN_chantashaw_advantage > OMN_aryusz_advantage	}
				}
				add_to_variable = { OMN_chantashaw_round_victory = 1 }
				set_temp_variable = { chantashaw_manpower = 500 }
				OMN_Chantashaw_Manpower_Change = yes
				set_temp_variable = { aryusz_manpower = 800 }
				OMN_Aryusz_Manpower_Change = yes
			}
			if = {
				limit = {
					check_variable = { OMN_aryusz_advantage > OMN_chantashaw_advantage	}
				}
				add_to_variable = { OMN_aryusz_round_victory = 1 }
				set_temp_variable = { chantashaw_manpower = 800 }
				OMN_Chantashaw_Manpower_Change = yes
				set_temp_variable = { aryusz_manpower = 500 }
				OMN_Aryusz_Manpower_Change = yes

			}
			set_variable = { OMN_chantashaw_advantage = 5 }
			set_variable = { OMN_aryusz_advantage = 4 }
			if = {
				limit = {
					check_variable = { OMN_chantashaw_round_victory > 2	}
				}
				set_country_flag = OMN_chantashaw_victory
				mark_focus_tree_layout_dirty = yes
				hidden_effect = {
					country_event = {
						id = decision_omn.1
					}
				}
				clr_country_flag = OMN_civil_war_chantashaw
				clr_country_flag = OMN_civil_war_aryusz
			}
			if = {
				limit = {
					check_variable = { OMN_aryusz_round_victory > 2	}
				}
				set_country_flag = OMN_aryusz_victory
				mark_focus_tree_layout_dirty = yes
				hidden_effect = {
					country_event = {
						id = decision_omn.2
					}
				}
				clr_country_flag = OMN_civil_war_chantashaw
				clr_country_flag = OMN_civil_war_aryusz
			}
			hidden_effect = {
				if = {
					limit = {
						OR = {
							has_country_flag = OMN_civil_war_chantashaw
							has_country_flag = OMN_civil_war_aryusz
						}
					}
					activate_mission = OMN_battle_round
