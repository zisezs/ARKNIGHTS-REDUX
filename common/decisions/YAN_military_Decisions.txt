######################################################
# 大炎 · 决议组【大营演武】
#
# 由国策【懋戒三军】解锁；后续国策逐条追加决议。
#   1. 安抚将官与士卒  <- 懋戒三军（上限 10 次，20 陆军经验/次）
#   2. 干戈的边界      <- 止戈为武（上限 5 次，30 政治点/次）
#   3. 矛盾的损耗      <- 烘炉御者（上限 5 次，30 政治点/次）
#   4. 谋定而后动      <- 天下共此劫（上限 5 次，30 陆军经验/次）
#
# 次数上限用变量计数器（YAN_mil_rf_dec_*_used，由 YAN_mil_reform_init 归零），
# 冷却统一 30 天（days_re_enable）。
# 所有效果都写进民族精神【大炎帝国军事改革】的同名动态变量，直接 add_to_variable，
# 不额外发 idea，避免玩家面板出现一堆零碎小精神。
######################################################

YAN_mil_great_drill = {

	# ---------------- 1. 安抚将官与士卒 ----------------
	YAN_mil_dec_comfort = {
		icon = GFX_decision_generic_army_support

		# 次数用尽后直接隐藏，避免把 check_variable 的原始条件显示在 tooltip 里
		visible = {
			has_country_flag = YAN_mil_reform_started
			has_country_flag = YAN_mil_exercise_unlocked
			check_variable = { YAN_mil_rf_dec_comfort_used < 10 }
		}

		available = {
			has_army_experience > 19
		}

		custom_cost_trigger = {
			has_army_experience > 19
		}
		custom_cost_text = YAN_mil_cost_20_army_exp

		priority = 100
		days_re_enable = 30

		complete_effect = {
			add_to_variable = { YAN_mil_rf_dec_comfort_used = 1 }
			custom_effect_tooltip = YAN_mil_dec_comfort_tt
			add_to_variable = { YAN_mil_rf_army_xp = 0.02 tooltip = experience_gain_army_factor_tt }
			add_to_variable = { YAN_mil_rf_air_xp = 0.02 tooltip = experience_gain_air_factor_tt }
			add_to_variable = { YAN_mil_rf_army_org = 0.02 tooltip = army_org_factor_tt }
			add_to_variable = { YAN_mil_rf_army_speed = 0.02 tooltip = army_speed_factor_tt }
			army_experience = -20
		}

		ai_will_do = {
			factor = 10
		}
	}

	# ---------------- 2. 干戈的边界 ----------------
	YAN_mil_dec_border = {
		icon = GFX_decision_generic_military

		visible = {
			has_country_flag = YAN_mil_reform_started
			has_country_flag = YAN_mil_dec_border_unlocked
			check_variable = { YAN_mil_rf_dec_border_used < 5 }
		}

		cost = 30
		priority = 90
		days_re_enable = 30

		complete_effect = {
			add_to_variable = { YAN_mil_rf_dec_border_used = 1 }
			custom_effect_tooltip = YAN_mil_dec_border_tt
			add_to_variable = { YAN_mil_rf_breakthrough = 0.02 tooltip = breakthrough_factor_tt }
			add_to_variable = { YAN_mil_rf_inf_defence = 0.01 tooltip = army_infantry_defence_factor_tt }
		}

		ai_will_do = {
			factor = 10
		}
	}

	# ---------------- 3. 矛盾的损耗 ----------------
	YAN_mil_dec_attrition = {
		icon = GFX_decision_generic_decision

		visible = {
			has_country_flag = YAN_mil_reform_started
			has_country_flag = YAN_mil_dec_attrition_unlocked
			check_variable = { YAN_mil_rf_dec_attrition_used < 5 }
		}

		cost = 30
		priority = 80
		days_re_enable = 30

		complete_effect = {
			add_to_variable = { YAN_mil_rf_dec_attrition_used = 1 }
			custom_effect_tooltip = YAN_mil_dec_attrition_tt
			add_to_variable = { YAN_mil_rf_supply = -0.02 tooltip = supply_consumption_factor_tt }
			add_to_variable = { YAN_mil_rf_civ_factory_speed = 0.02 tooltip = production_speed_industrial_complex_factor_tt }
		}

		ai_will_do = {
			factor = 10
		}
	}

	# ---------------- 4. 谋定而后动 ----------------
	YAN_mil_dec_planning = {
		icon = GFX_decision_generic_political_discourse

		visible = {
			has_country_flag = YAN_mil_reform_started
			has_country_flag = YAN_mil_dec_planning_unlocked
			check_variable = { YAN_mil_rf_dec_planning_used < 5 }
		}

		available = {
			has_army_experience > 29
		}

		custom_cost_trigger = {
			has_army_experience > 29
		}
		custom_cost_text = YAN_mil_cost_30_army_exp

		priority = 70
		days_re_enable = 30

		complete_effect = {
			add_to_variable = { YAN_mil_rf_dec_planning_used = 1 }
			custom_effect_tooltip = YAN_mil_dec_planning_tt
			add_to_variable = { YAN_mil_rf_supply = 0.01 tooltip = supply_consumption_factor_tt }
			add_to_variable = { YAN_mil_rf_army_speed = 0.02 tooltip = army_speed_factor_tt }
			army_experience = -30
		}

		ai_will_do = {
			factor = 10
		}
	}

}
