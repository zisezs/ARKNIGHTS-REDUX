REU_East_Daria_Industrialization = {

	REU_industrialize_core_state = {
		icon = GFX_decision_generic_factory

		state_target = yes
		target_array = ROOT.core_states

		on_map_mode = map_and_decisions_view

		target_root_trigger = {
			has_country_flag = East_Ursus_Unification_After
		}

		target_trigger = {
			FROM = {
				is_core_of = ROOT
				is_owned_by = ROOT
				NOT = {
					has_state_flag = REU_state_industrialized
				}
			}
		}

		cost = 50

		days_remove = 30

		modifier = {
			civilian_factory_use = 2
		}

		remove_effect = {
			add_to_variable = { REU_agri_industrialization_count = 1 }
			FROM = {
				set_state_flag = REU_state_industrialized
				add_extra_state_shared_building_slots = 1
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
		}

		ai_will_do = {
			base = 0

			modifier = {
				add = 80
				check_variable = { REU_agri_industrialization_count < 20 }
				has_political_power > 150
				num_of_civilian_factories > 42
			}
		}
	}
}
