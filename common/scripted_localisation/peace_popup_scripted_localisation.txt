defined_text = {
	name = GetWarConclusionLoc
	text = {
		trigger = {
			event_target:losing_country = {
				ursus_unifier = yes
			}
			event_target:winning_country = {
				ursus_unifier = yes
			}
		}
		localization_key = PEACE_POPUP_WAR_CONCLUSION_URS
	}
	text = {
		trigger = {
			event_target:losing_country = {
				original_tag = CFC
			}
			event_target:winning_country = {
				original_tag = CLB
			}
		}
		localization_key = PEACE_POPUP_WAR_CONCLUSION_CFC
	}
	text = {
		trigger = {
			event_target:losing_country = {
				victoria_unifier = yes
			}
			event_target:winning_country = {
				original_tag = TAR
				has_country_flag = TAR_independence_flag
			}
		}
		localization_key = PEACE_POPUP_WAR_CONCLUSION_TAR_independence
	}
	text = {
		localization_key = PEACE_POPUP_WAR_CONCLUSION_DEFAULT
	}
