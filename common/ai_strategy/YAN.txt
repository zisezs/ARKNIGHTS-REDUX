# ==============================================================================
# 大炎 (YAN) 专属 AI 战略（2026-10-31）
#
# 目标：把大炎从"低配步兵海"掰成"重装拳头 + 满编步兵"。
#   大炎开局 50 州 / 36 座军工 / 144 座民工厂 / 2.64 亿人口，
#   还有重坦、机械化、火炮与防空的全套开局科技；
#   但此前没有任何专属 ai_strategy，只能吃全局默认：
#     infantry 40 / militia 5 / armor 20 / motorized 10，
#     equipment_production_factor infantry 25 / artillery 20 / support 30 / armor 10。
#   结果就是 AI 把产能继续喂给步枪与民兵，重装线停在开局那 2 个实验旅。
#
# 与全局层的关系（很重要，数值会算错）：
#   common/ai_strategy 下各块是**叠加**的，同一个 type+id 的 value 相加，
#   所以这里只写增量。基线来自：
#     · default.txt        infantry 40 / militia 5 / armor 20 / motorized 10 /
#                          mountaineers 4 / air_assault 3 / landship 2 /
#                          equipment infantry 40、artillery 25
#     · REC_global_ai      template_prio infantry 25、ai_wanted_divisions_factor 25、
#                          equipment infantry 25 / artillery 20 / support 30 /
#                          anti_tank 15 / anti_air 15 / armor 10 / motorized 10、
#                          land_xp division_template 120、front_control 邻国 rush priority 60
#
# 编制模板见 common/ai_templates/YAN_templates.txt（步兵四档 / 重装三档 / 摩托化两档）。
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. 平时建军：步兵海为主力，装甲与机械化吃满，民兵归零
# ------------------------------------------------------------------------------
YAN_army_quality_first = {

	allowed = {
		original_tag = YAN
	}
	enable = {
		original_tag = YAN
	}
	abort_when_not_enabled = yes

	# —— 编制配额 ——
	# 2026-11-01：改为步兵海为主力，配额从 -10 抬到 +20（基础 120%）。
	ai_strategy = {
		type = role_ratio
		id = infantry
		value = 20
	}
	# 36 座军工 + 重坦科技，装甲从 20% 抬到 35%。
	ai_strategy = {
		type = role_ratio
		id = armor
		value = 15
	}
	# 机械化伴随群：15%。
	ai_strategy = {
		type = role_ratio
		id = motorized
		value = 5
	}
	# 大炎不缺人，缺的是装备；民兵会稀释枪械与 XP，配额压到 0。
	ai_strategy = {
		type = role_ratio
		id = militia
		value = -5
	}
	# 没有空突/直升机与 landship 科技，去掉悬空配额，避免工业需求算虚。
	ai_strategy = {
		type = role_ratio
		id = air_assault
		value = -3
	}
	ai_strategy = {
		type = role_ratio
		id = landship
		value = -2
	}
	ai_strategy = {
		type = role_ratio
		id = mountaineers
		value = -4
	}

	# —— 换代优先级：XP 花在装甲与步兵上，别雕民兵 ——
	ai_strategy = {
		type = template_prio
		id = heavy_armor
		value = 40
	}
	ai_strategy = {
		type = template_prio
		id = infantry
		value = 30
	}
	ai_strategy = {
		type = template_prio
		id = motorized
		value = 10
	}
	ai_strategy = {
		type = template_prio
		id = militia
		value = -60
	}

	# —— 陆军 XP：先改编制、再换装备改型、最后学说 ——
	ai_strategy = {
		type = land_xp_spend_priority
		id = division_template
		value = 80
	}
	ai_strategy = {
		type = land_xp_spend_priority
		id = equipment_variant
		value = 40
	}
	ai_strategy = {
		type = land_xp_spend_priority
		id = upgrade_xp_cutoff
		value = -10
	}

	# —— 数量：2.64 亿人口，常规扩军目标保持全局水平；3M 冲刺见第 7 节 ——
	ai_strategy = {
		type = ai_wanted_divisions_factor
		value = 15
	}
	ai_strategy = {
		type = build_army
		id = heavy_armor
		value = 30
	}
	ai_strategy = {
		type = build_army
		id = infantry
		value = 50
	}
	# 预备队不要蹲在后方：全局 +20，这里压回接近 0，让新师立刻上前线。
	ai_strategy = {
		type = spare_unit_factor
		value = -15
	}
}

# 7. 疯狂扩军：陆军人力未到 3M 前，全力爆大编制步兵师
#    has_army_manpower 统计已部署陆军人力；达到 3,000,000 后本策略自动停用，
#    扩军意愿回落到第 1/6 节的常规值。
# ------------------------------------------------------------------------------
YAN_army_expansion_to_3m = {

	allowed = {
		original_tag = YAN
	}

		original_tag = YAN

	abort_when_not_enabled = yes

	# 大幅抬高 AI 想要的师数，直到人力达到 3M。

		type = force_build_armies
		value = 100
	}
