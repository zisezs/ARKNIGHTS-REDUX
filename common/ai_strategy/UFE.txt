#保存实力
UFE_focus_defense_when_attack_REU = {
	allowed = {
		original_tag = UFE
	}
	enable = {
		has_war_with = REU
	}
	abort = {
		OR = {
			has_country_flag = {
				flag = UFE_vs_REU_war_flag
				days > 140
			}
			REU = { exists = no }
		}
	}
	ai_strategy = {
		type = front_control

		tag = REU
		ratio = 0.1

		priority = 200
		ordertype = front
		execution_type = careful
		execute_order = no
	}
}
