# ==============================================================================
# 全局 AI 强化战略 —— 对**所有国家**生效
#
# 为什么要这份文件：descriptor.mod 里 replace_path 掉了 common/ai_strategy，
# 原版那份 default.txt 不存在，本模组的 AI 基础行为只由两份文件构成：
#   · common/ai_strategy/default.txt   —— 原版同名词条的模组改写版
#   · 本文件                            —— 补上原版给各国的那些通用战略条目
#
# 语法要点（写错不会报错，会被静默忽略，务必照抄）：
#   · 块名自由，enable/abort 是国家作用域触发符；abort_when_not_enabled 表示
#     enable 一不成立就立刻撤掉本块。
#   · 同一个 type+id 在多个块里出现是**叠加**，因此下面凡与 default.txt 撞项的
#     都只给增量，不重复给绝对值。
#   · num_of_* 一律用 > / < 比较，写 >= 会让整块静默失效。
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. 陆军 XP 与换代编：让 AI 真的去改师、改装备、点学说

# ------------------------------------------------------------------------------
REC_global_land_xp_push = {
	enable = {
		always = yes
	}

	}

		type = land_xp_spend_priority

		value = 80

		value = 60
	}
}

# ------------------------------------------------------------------------------
# 7. 进攻欲望·战时：与我国接壤的战线一律全力压上，其余战线保持推进
#    front_control 的 country_trigger 作用域 = 敌国，FROM = 我们自己
#    priority 越高越优先；IBR 等手搓国家的专属战略用 200，不会互相打架
# ------------------------------------------------------------------------------
REC_global_aggression_neighbours = {
	enable = {
		has_war = yes
		num_divisions > 15
	}
	abort = {
		OR = {
			has_war = no
			num_divisions < 10
		}
	}

	# 邻国战线：ready 就开打（配合 defines 里 LOCATION_BALANCE_TO_ADVANCE 的负值）
	ai_strategy = {
		type = front_control
		country_trigger = {
			is_neighbor_of = FROM
		}
		ratio = 0.05
		priority = 60
		ordertype = front
		execution_type = rush
		execute_order = yes
		manual_attack = yes
	}
	# 跨海/远程战线：也要执行命令，但用 balanced，别把师扔进绞肉机
	ai_strategy = {
		type = front_control
		country_trigger = {
			NOT = {
				is_neighbor_of = FROM
			}
		}
		ratio = 0.05
		priority = 30
		ordertype = front
		execution_type = balanced
		execute_order = yes
		manual_attack = yes
	}
	# 兵力往主攻方向集中，而不是沿边境铺一 thin line
	ai_strategy = {
		type = force_concentration_factor
		value = 30
	}
	ai_strategy = {
		type = force_concentration_target_weight
		country_trigger = {
			is_neighbor_of = FROM
		}
		value = 40
	}
	# 邻国战线多要兵：让 AI 真的把编制填满到进攻方向
	ai_strategy = {
		type = front_unit_request
		country_trigger = {
			is_neighbor_of = FROM
		}
		value = 30
	}
}
