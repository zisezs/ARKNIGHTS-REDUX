## 限量部队「近卫军之锋」##

# 解锁（第三幕根国策「军乐」）：
# 12 个摩托化步兵团的固定编制、1 个支援连（工兵），上限 4 个师，并立刻拉出 2 个满编师。
# set_technology 必须在 division_template 之前，否则摩托化营尚未存在、建出的编制会缺装。
SCR_vanguard_guard_unlock_effect = {
	if = {
		limit = {
			NOT = {
				has_country_flag = SCR_vanguard_guard_unlocked_flag
			}
		}
		set_country_flag = SCR_vanguard_guard_unlocked_flag
		set_technology = {
			motorized_infantry_0 = 1
		}
		division_template = {
			name = "近卫军之锋"
			is_locked = yes
			regiments = {
				motorized = { x = 0 y = 0 }
				motorized = { x = 0 y = 1 }
				motorized = { x = 0 y = 2 }
				motorized = { x = 1 y = 0 }
				motorized = { x = 1 y = 1 }
				motorized = { x = 1 y = 2 }
				motorized = { x = 2 y = 0 }
				motorized = { x = 2 y = 1 }
				motorized = { x = 2 y = 2 }
				motorized = { x = 3 y = 0 }
				motorized = { x = 3 y = 1 }
				motorized = { x = 3 y = 2 }
			}
			support = {
				engineer = { x = 0 y = 0 }
			}
			priority = 3
		}
		set_variable = {
			var = SCR_vanguard_guard_cap
			value = 4
		}
		set_division_template_cap = {
			division_template = "近卫军之锋"
			division_cap = SCR_vanguard_guard_cap
		}
		capital_scope = {
			create_unit = {
				division = "name = \"近卫军之锋 第一师\" division_template = \"近卫军之锋\" start_experience_factor = 1 start_equipment_factor = 1 start_manpower_factor = 1"
				owner = ROOT
			}
			create_unit = {
				division = "name = \"近卫军之锋 第二师\" division_template = \"近卫军之锋\" start_experience_factor = 1 start_equipment_factor = 1 start_manpower_factor = 1"
				owner = ROOT
			}
		}
	}
}

# 陆军线其余 11 个国策各调用一次：师的数量上限 +2（走完即 4 + 11×2 = 26）。
# 菱形支线同层两翼的完成顺序不固定，故一律按计数器累加，不写死绝对值。
SCR_vanguard_guard_cap_effect = {
	if = {
		limit = {
			has_country_flag = SCR_vanguard_guard_unlocked_flag
		}
		add_to_variable = {
			var = SCR_vanguard_guard_cap
			value = 2
		}
		set_division_template_cap = {
			division_template = "近卫军之锋"
			division_cap = SCR_vanguard_guard_cap
		}
	}
}

# 剧团招聘中！·近卫军之锋：在首都生成 1 个满编近卫军之锋师（模板由「军乐」创建，此处仅发放）。
SCR_vanguard_guard_recruit_effect = {
	if = {
		limit = {
			has_country_flag = SCR_vanguard_guard_unlocked_flag
		}
		capital_scope = {
			create_unit = {
				division = "division_template = \"近卫军之锋\" start_experience_factor = 1 start_equipment_factor = 1 start_manpower_factor = 1"
				owner = ROOT
			}
		}
	}
}

# 【高卢，高卢！】30 天限时精神：先摘再挂，重复触发不会报错。
SCR_gaul_gaul_timed_effect = {
	if = { limit = { has_idea = SCR_dyn_gaul_gaul } remove_ideas = SCR_dyn_gaul_gaul }
	if = { limit = { has_idea = SCR_gaul_vic_bonus } remove_ideas = SCR_gaul_vic_bonus }
	add_timed_idea = { idea = SCR_dyn_gaul_gaul days = 30 }
	add_timed_idea = { idea = SCR_gaul_vic_bonus days = 30 }
}
