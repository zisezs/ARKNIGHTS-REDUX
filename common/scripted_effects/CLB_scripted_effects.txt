	set_variable = { communism_all_votes = 0 }
	set_variable = { socialism_all_votes = 0 }
	set_variable = { progressivism_all_votes = 0 }
	set_variable = { liberalism_all_votes = 0 }
	set_variable = { conservatism_all_votes = 0 }
	set_variable = { authoritarianism_all_votes = 0 }
	set_variable = { fascism_all_votes = 0 }

	clr_country_flag = CLB_communism_win_election_flag
	clr_country_flag = CLB_socialism_win_election_flag
	clr_country_flag = CLB_liberalism_win_election_flag

}

CLB_election_results = {
	set_variable = { A_all_votes = 0 }
	set_variable = { B_all_votes = 0 }
	add_to_variable = { election_num = 1 }

	if = {
		limit = {
			has_country_flag = CLB_communism_in_A_flag
		}
		add_to_variable = { A_all_votes = communism_all_votes }
	}
	if = {
		limit = {
			has_country_flag = CLB_socialism_in_A_flag
		}
		add_to_variable = { A_all_votes = socialism_all_votes }
	}
	if = {
		limit = {
			has_country_flag = CLB_progressivism_in_A_flag
		}
		add_to_variable = { A_all_votes = progressivism_all_votes }
	}
	if = {
		limit = {
			has_country_flag = CLB_liberalism_in_A_flag
		}
		add_to_variable = { A_all_votes = liberalism_all_votes }
	}

	if = {
		limit = {
			has_country_flag = CLB_liberalism_in_B_flag
		}
		add_to_variable = { B_all_votes = liberalism_all_votes }
	}
	if = {
		limit = {
			has_country_flag = CLB_conservatism_in_B_flag
		}
		add_to_variable = { B_all_votes = conservatism_all_votes }
	}
	if = {
		limit = {
			has_country_flag = CLB_authoritarianism_in_B_flag
		}
		add_to_variable = { B_all_votes = authoritarianism_all_votes }
	}
	if = {
		limit = {
			has_country_flag = CLB_fascism_in_B_flag
		}
		add_to_variable = { B_all_votes = fascism_all_votes }
	}

	#找出得票最多的意识形态（平票全显示）
	set_temp_variable = { max_votes = A_all_votes }

	if = {
		limit = { check_variable = { B_all_votes > max_votes } }
		set_temp_variable = { max_votes = B_all_votes }
	}
	if = {
		limit = { check_variable = { communism_all_votes > max_votes } }
		set_temp_variable = { max_votes = communism_all_votes }
	}
	if = {
		limit = { check_variable = { socialism_all_votes > max_votes } }
		set_temp_variable = { max_votes = socialism_all_votes }
	}
	if = {
		limit = { check_variable = { liberalism_all_votes > max_votes } }
		set_temp_variable = { max_votes = liberalism_all_votes }
	}
	if = {
		limit = { check_variable = { fascism_all_votes > max_votes } }
		set_temp_variable = { max_votes = fascism_all_votes }
	}

	#输出所有等于最高票数的意识形态
	set_variable = { win_election_parties = 0 }
	if = {
		limit = { check_variable = { communism_all_votes = max_votes } }
		set_country_flag = CLB_communism_win_election_flag
		add_to_variable = { win_election_parties = 1 }
	}
	if = {
		limit = { check_variable = { socialism_all_votes = max_votes } }
		set_country_flag = CLB_socialism_win_election_flag
		add_to_variable = { win_election_parties = 1 }
	}
	if = {
		limit = { check_variable = { liberalism_all_votes = max_votes } }
		set_country_flag = CLB_liberalism_win_election_flag
		add_to_variable = { win_election_parties = 1 }
	}
	if = {
		limit = { check_variable = { fascism_all_votes = max_votes } }
		set_country_flag = CLB_fascism_win_election_flag
		add_to_variable = { win_election_parties = 1 }
	}
	if = {
		limit = { check_variable = { A_all_votes = max_votes } }
		set_country_flag = CLB_A_win_election_flag
		add_to_variable = { win_election_parties = 1 }
	}
	if = {
		limit = { check_variable = { B_all_votes = max_votes } }
		set_country_flag = CLB_B_win_election_flag
		add_to_variable = { win_election_parties = 1 }
	}
}

# ----------------------------------------------------------------------------
# 谢拉格战线脱身
#
# 问题：决议 CLB_KJG（common/decisions/CLB_decisions.txt）把谢拉格 add_to_faction 进
#       哥伦比亚阵营，fire_only_once 且无退出条件 —— 哥伦比亚被永久拖进谢拉格战场，
#       与维多利亚/莱塔尼亚在雪境无限绞肉、兵力堆积、卡顿。
#
# 修法：谢拉格「投降」或「丢地过半」时，哥伦比亚收手：
#   ① 把谢拉格以及阵营内其它已投降成员踢出阵营
#      （remove_from_faction 由阵营领袖调用，参数=要踢的国家）
#   ② 哥伦比亚自己对维多利亚、莱塔尼亚白和撤军
# 被踢出去的势力各自的战争照旧，从此自己面对维多利亚/莱塔尼亚。
#
# 触发挂在 common/on_actions/CLB_on_actions.txt 的 on_monthly_CLB，一次性旗标去重。
# 作用域：国家（哥伦比亚）。
# ----------------------------------------------------------------------------
CLB_kjerag_disengage = {
	if = {
		limit = {
			NOT = { has_country_flag = CLB_kjerag_disengaged }
			KJG = {
				exists = yes
				is_in_faction_with = ROOT
				OR = {
					has_capitulated = yes
					surrender_progress > 0.5
				}
			}
		}
		set_country_flag = CLB_kjerag_disengaged

		hidden_effect = {
			# ① 踢人：谢拉格 + 阵营内其它已投降成员
			#    用 every_country（遍历国表，稳定）而非 every_faction_member
			#    —— 后者一边遍历阵营成员一边删成员，迭代器会漂。
			if = {
				limit = { is_faction_leader = yes }
				every_country = {
					limit = {
						NOT = { tag = ROOT }
						is_in_faction_with = ROOT
						OR = {
							has_capitulated = yes
							surrender_progress > 0.5
						}
					}
					ROOT = {
						remove_from_faction = PREV
					}
				}
			}

			# ② 哥伦比亚对维多利亚、莱塔尼亚白和，撤军保全自身
			if = {
				limit = { has_war_with = VIC }
				white_peace = VIC
			}
			if = {
