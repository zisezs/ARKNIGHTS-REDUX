UFE_VS_REU_GAME_EFFECT = {
	#回合数
	add_to_variable = { UFE.UFE_vs_REU_turn = 1 }
	#整合运动特质
	if = {
		limit = {
			check_variable = { UFE.game_points = REU.game_points } 
		}
		add_to_variable = { REU.game_points = 1 }
	}
	###正式统计###
	if = { #某方大于21导致的胜负
		limit = { 
			check_variable = { UFE.game_points = 21 } 
		}
		add_to_variable = { UFE.win_points = 1 }
	}
	else_if = { #远东特质
		limit = { 
			check_variable = { UFE.game_points > 21 } 
			check_variable = { REU.game_points > 21 } 
		}
		add_to_variable = { UFE.win_points = 1 }
	}
	else_if = { #远东胜利
		limit = { 
			check_variable = { UFE.game_points < 22 } 
			check_variable = { REU.game_points > 21 } 
		}
		add_to_variable = { UFE.win_points = 1 }
	}
	else_if = { #整合胜利
		limit = { 
			check_variable = { UFE.game_points > 21 } 
			check_variable = { REU.game_points < 22 } 
		}
		add_to_variable = { REU.win_points = 1 }
	}
	else_if = { #大小较量引起的胜负，远东胜利
		limit = {
			check_variable = { UFE.game_points > REU.game_points }
		}
		add_to_variable = { UFE.win_points = 1 }
	}
	else_if = { #大小较量引起的胜负，整合胜利
		limit = {
			check_variable = { UFE.game_points < REU.game_points }
		}
		add_to_variable = { REU.win_points = 1 }
	}
	#计算工具清零
	set_variable = { UFE.game_points = 0 }
	set_variable = { REU.game_points = 0 }
}
