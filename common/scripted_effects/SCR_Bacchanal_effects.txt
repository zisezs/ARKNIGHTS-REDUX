# ==============================================================================
#  猩红剧团 ·「酒神的大地巡演」—— scripted_effects
#
#  状态变量（全部国家 scope）：
#    SCR_tour_state    当前选中的州 id（0 = 未选）
#    SCR_tour_play     当前剧目 1..5
#    SCR_tour_count    已巡演州数
#    SCR_tour_factory  已获得的地图外民用工厂数
#
#  州旗标：
#    scr_bac_selected  当前选中的舞台（只允许一州同时持有）
#    scr_bac_visited   已巡演过（永久，避免同一州反复刷工厂）
# ==============================================================================

# ---------------- 初始化（老存档也安全，幂等） ----------------
SCR_Bacchanal_Init = {
	if = {
		limit = { NOT = { has_variable = SCR_tour_state } }
		set_variable = { SCR_tour_state = 0 }
	}
	if = {
		limit = { NOT = { has_variable = SCR_tour_play } }
		set_variable = { SCR_tour_play = 1 }
	}
	if = {
		limit = { NOT = { has_variable = SCR_tour_count } }
		set_variable = { SCR_tour_count = 0 }
	}
	if = {
		limit = { NOT = { has_variable = SCR_tour_factory } }
		set_variable = { SCR_tour_factory = 0 }
	}
	if = {
		limit = { NOT = { has_variable = SCR_tour_click } }
		set_variable = { SCR_tour_click = 0 }
	}
	if = {
		limit = { NOT = { has_variable = SCR_bac_tick } }
		set_variable = { SCR_bac_tick = 0 }
	}
	if = {
		limit = { NOT = { has_variable = SCR_tour_cd } }
		set_variable = { SCR_tour_cd = 0 }
	}
	if = {
		limit = { NOT = { has_variable = SCR_tour_all } }
		set_variable = { SCR_tour_all = 0 }
	}
	if = {
		limit = { NOT = { has_country_flag = scr_bac_fx_init } }
		set_country_flag = scr_bac_fx_init
		SCR_Bacchanal_InitFx = yes
	}
}

# ---------------- 开 / 关 ----------------
SCR_Bacchanal_Open = {
	SCR_Bacchanal_Init = yes
	set_country_flag = scr_bac_open
	add_to_variable = { SCR_bac_tick = 1 }
	if = {
		limit = { NOT = { check_variable = { SCR_tour_play > 0 } } }
		set_variable = { SCR_tour_play = 1 }
	}
}

SCR_Bacchanal_Close = {
	clr_country_flag = scr_bac_open
	every_state = {
		clr_state_flag = scr_bac_selected
	}
	set_variable = { SCR_tour_state = 0 }
	force_update_map_mode = {
		limit = { always = yes }
		mapmode = scr_bacchanal_map_mode
	}
	add_to_variable = { SCR_bac_tick = 1 }
}

SCR_Bacchanal_Toggle = {
	if = {
		limit = { has_country_flag = scr_bac_open }
		SCR_Bacchanal_Close = yes
	}
	else = {
		SCR_Bacchanal_Open = yes
	}
}

# ---------------- 地图图标点击：选定舞台 ----------------
# ⚠ state_mapicon 的 effects 作用域是「被点击的州」，但 ROOT 同样落在州上，
#   不会自动变成玩家国（_documentation.md 只保证 scope = state）。因此这里
#   绝不直接写 ROOT.* 国家变量：先把州旗标点好，再交给国家作用域同步。
#   自己控制的州 → controller 就是玩家；他国领土 → 广播给人类玩家（单机即玩家）。
SCR_Bacchanal_SelectState = {
	every_state = {
		clr_state_flag = scr_bac_selected
	}
	if = {
		limit = { NOT = { has_state_flag = scr_bac_visited } }
		set_state_flag = scr_bac_selected
		if = {
			limit = { controller = { is_ai = no } }
			controller = { SCR_Bacchanal_AdoptSelected = yes }
		}
		else = {
			every_country = {
				limit = { is_ai = no }
				SCR_Bacchanal_AdoptSelected = yes
			}
		}
	}
}

# ---------------- 国家作用域：把州旗标同步成国家变量并刷新面板 ----------------
# 所有选州路径最后都走这里：面板按钮在玩家国作用域调用；地图图标经
# controller / every_country 转到国家作用域后再调用。
SCR_Bacchanal_AdoptSelected = {
	SCR_Bacchanal_Init = yes
	set_variable = { SCR_tour_state = 0 }
	random_state = {
		limit = { has_state_flag = scr_bac_selected }
		# ⚠ 这里不能用 ROOT：从 state_mapicon 进来时 ROOT 仍是「被点的州」，
		#   即使已经切到 controller/every_country，ROOT 也不会跟着变成国家。
		#   而 random_state 的上一级 PREV 一定就是当前国家作用域。
		set_variable = { PREV.SCR_tour_state = THIS.id }
	}
	SCR_Bacchanal_Open = yes
	force_update_map_mode = {
		limit = { always = yes }
		mapmode = scr_bacchanal_map_mode
	}
	# 面板按钮/地图图标都不会自然触发 dirty 变量，必须手动刷新 GUI。
	add_to_variable = { SCR_bac_tick = 1 }
}

# ---------------- 面板备用选州：随机挑一个未巡演州 ----------------
# 地图标记是主交互；这一条保证在标记因地图模式/分辨率被遮住时仍能选定舞台。
SCR_Bacchanal_RandomPick = {
	SCR_Bacchanal_Init = yes
	every_state = {
		clr_state_flag = scr_bac_selected
	}
	if = {
		limit = { check_variable = { SCR_tour_all = 1 } }
		random_state = {
			limit = { NOT = { has_state_flag = scr_bac_visited } }
			set_state_flag = scr_bac_selected
		}
	}
	else = {
		random_controlled_state = {
			limit = { NOT = { has_state_flag = scr_bac_visited } }
			set_state_flag = scr_bac_selected
		}
	}
	SCR_Bacchanal_AdoptSelected = yes
}

# ---------------- 面板内静态地图点击：按 SCR_tour_click 选州 ----------------
SCR_Bacchanal_PickByVariable = {
	SCR_Bacchanal_Init = yes
	every_state = {
		clr_state_flag = scr_bac_selected
	}
	random_state = {
		limit = {
			# 这个 effect 由 player_context 面板按钮调用，ROOT 是玩家国；
			# random_state 里已切换到州作用域，所以国家变量必须带 ROOT 前缀。
			state = var:ROOT.SCR_tour_click
			NOT = { has_state_flag = scr_bac_visited }
		}
		set_state_flag = scr_bac_selected
	}
	SCR_Bacchanal_AdoptSelected = yes
}

# ---------------- 启程：扣政治点、挂州 BUFF、给地图外工厂 ----------------
SCR_Bacchanal_Depart = {
	if = {
		limit = {
			# 以州旗标为准：SCR_tour_state 只负责面板显示，避免变量作用域问题把按钮锁死。
			# HOI4 不支持 >=；75 点门槛写成 > 74。
			has_political_power > 74
			# 冷却期与按钮 _click_enabled 保持一致，防止脚本/连点绕过。
			NOT = { check_variable = { SCR_tour_cd > 0 } }
			any_state = {
				has_state_flag = scr_bac_selected
				NOT = { has_state_flag = scr_bac_visited }
			}
		}

		add_political_power = -75

		random_state = {
			limit = { has_state_flag = scr_bac_selected }
			set_state_flag = scr_bac_visited
			clr_state_flag = scr_bac_selected
			SCR_Bacchanal_ApplyPlay = yes
		}

		add_to_variable = { SCR_tour_count = 1 }
		add_to_variable = { SCR_tour_factory = 1 }
		set_variable = { SCR_tour_cd = 30 }
		add_offsite_building = { type = industrial_complex level = 1 }

		# 立刻刷新「酒神巡演」地图模式，避免新挂的地块修正等到次日才显色。
		force_update_map_mode = {
			limit = { always = yes }
			mapmode = scr_bacchanal_map_mode
		}
		add_to_variable = { SCR_bac_tick = 1 }

		# 保留 SCR_tour_state：面板上继续显示刚刚落幕的舞台；
		# 但 scr_bac_selected 已清空，所以「启程」会立刻灰掉，防止重复结算。
	}
}

# ---------------- 按剧目挂 365 天州级动态修正 ----------------
SCR_Bacchanal_ApplyPlay = {
	if = {
		limit = { check_variable = { ROOT.SCR_tour_play = 1 compare = equals } }
		add_dynamic_modifier = {
			modifier = SCR_tour_dm_der_ring
			days = 365
		}
	}
	if = {
		limit = { check_variable = { ROOT.SCR_tour_play = 2 compare = equals } }
		add_dynamic_modifier = {
			modifier = SCR_tour_dm_one_hundred_one_days
			days = 365
		}
	}
	if = {
		limit = { check_variable = { ROOT.SCR_tour_play = 3 compare = equals } }
		add_dynamic_modifier = {
			modifier = SCR_tour_dm_wonderland
			days = 365
		}
	}
	if = {
		limit = { check_variable = { ROOT.SCR_tour_play = 4 compare = equals } }
		add_dynamic_modifier = {
			modifier = SCR_tour_dm_frost_deer
			days = 365
		}
	}
	if = {
		limit = { check_variable = { ROOT.SCR_tour_play = 5 compare = equals } }
		add_dynamic_modifier = {
			modifier = SCR_tour_dm_mother_wolf
			days = 365
		}
	}
	SCR_Bacchanal_SetFx = yes
}
