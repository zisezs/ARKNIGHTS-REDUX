East_Ursus_Upheaval_Chaos_Modifier_Setup = {
	if = {
		limit = { check_variable = { Upheaval_Chaos > 0.9 } }
		set_variable = { Upheaval_Chaos_temp = Upheaval_Chaos }
		add_to_variable = { Upheaval_Chaos_temp = -0.3 }
		set_variable = { Upheaval_Chaos_Modifier_conscription_factor = Upheaval_Chaos_temp }
		multiply_variable = { Upheaval_Chaos_Modifier_conscription_factor = -0.5 }

		set_variable = { Upheaval_Chaos_Modifier_org_loss_when_moving = Upheaval_Chaos_temp }
		multiply_variable = { Upheaval_Chaos_Modifier_org_loss_when_moving = 0.5 }

		set_variable = { Upheaval_Chaos_Modifier_out_of_supply_factor = Upheaval_Chaos_temp }
		multiply_variable = { Upheaval_Chaos_Modifier_out_of_supply_factor = 0.5 }

		set_variable = { Upheaval_Chaos_Modifier_surrender_limit = Upheaval_Chaos }
		multiply_variable = { Upheaval_Chaos_Modifier_surrender_limit = -0.7 }

		set_variable = { Upheaval_Chaos_Modifier_stability_factor = Upheaval_Chaos }
		multiply_variable = { Upheaval_Chaos_Modifier_stability_factor = -0.7 }

		set_variable = { Upheaval_Chaos_Modifier_war_support_factor = Upheaval_Chaos }
		multiply_variable = { Upheaval_Chaos_Modifier_war_support_factor = -0.7 }
	}
	else_if = {
		limit = { check_variable = { Upheaval_Chaos > 0.3 } }
		set_variable = { Upheaval_Chaos_temp = Upheaval_Chaos }
		add_to_variable = { Upheaval_Chaos_temp = -0.3 }
		set_variable = { Upheaval_Chaos_Modifier_conscription_factor = Upheaval_Chaos_temp }
		multiply_variable = { Upheaval_Chaos_Modifier_conscription_factor = -0.5 }

		set_variable = { Upheaval_Chaos_Modifier_org_loss_when_moving = Upheaval_Chaos_temp }
		multiply_variable = { Upheaval_Chaos_Modifier_org_loss_when_moving = 0.5 }

		set_variable = { Upheaval_Chaos_Modifier_out_of_supply_factor = Upheaval_Chaos_temp }
		multiply_variable = { Upheaval_Chaos_Modifier_out_of_supply_factor = 0.5 }

		set_variable = { Upheaval_Chaos_Modifier_surrender_limit = Upheaval_Chaos_temp }
		multiply_variable = { Upheaval_Chaos_Modifier_surrender_limit = -0.5 }

		set_variable = { Upheaval_Chaos_Modifier_stability_factor = Upheaval_Chaos }
		multiply_variable = { Upheaval_Chaos_Modifier_stability_factor = -0.5 }

		set_variable = { Upheaval_Chaos_Modifier_war_support_factor = Upheaval_Chaos }
		multiply_variable = { Upheaval_Chaos_Modifier_war_support_factor = -0.5 }
	}
	else = {
		set_variable = { Upheaval_Chaos_temp = 0.3 }
		subtract_from_variable = { Upheaval_Chaos_temp = Upheaval_Chaos }
		set_variable = { Upheaval_Chaos_Modifier_conscription_factor = Upheaval_Chaos_temp }

		set_variable = { Upheaval_Chaos_Modifier_stability_factor = Upheaval_Chaos_temp }

		set_variable = { Upheaval_Chaos_Modifier_war_support_factor = Upheaval_Chaos_temp }
	}
	force_update_dynamic_modifier = yes
}

Ursus_area_uprising = {
	every_state = {
		limit = {
			UCM_uprising_area = yes
			NOT = {
				has_state_flag = URS_suppressed
			}
		}
		random_country = {
			limit = {
				ursus_unifier = yes
				has_government = communism
			}
			PREV = {
				add_core_of = PREV
				force_enable_resistance = URS
				start_resistance = URS
				set_resistance = 65
				set_compliance = 0
			}
		}
	}
	every_state = {
		limit = {
			ULB_uprising_area = yes
			NOT = {
				has_state_flag = URS_suppressed
			}
		}
		random_country = {
			limit = {
				ursus_unifier = yes
				OR = {
					has_government = liberalism
					original_tag = UFE
				}
			}
			PREV = {
				add_core_of = PREV
				force_enable_resistance = URS
				start_resistance = URS
				set_resistance = 65
				set_compliance = 0
			}
		}
	}
	every_state = {
		limit = {
			UFS_uprising_area = yes
			NOT = {
				has_state_flag = URS_suppressed
			}
		}
		random_country = {
			limit = {
				ursus_unifier = yes
				OR = {
					has_government = fascism
					original_tag = KMG
				}
			}
			PREV = {
				add_core_of = PREV
				force_enable_resistance = URS
				start_resistance = URS
				set_resistance = 65
				set_compliance = 0
			}
		}
	}
	every_state = {
		limit = {
			KZC_uprising_area = yes
			NOT = {
				has_state_flag = URS_suppressed
			}
		}
		transfer_state_to = KZC
	}
	every_state = {
		limit = {
			BUS_uprising_area = yes
			NOT = {
				has_state_flag = URS_suppressed
			}
		}
		transfer_state_to = BUS
	}
	every_state = {
		limit = {
			BLT_uprising_area = yes
			NOT = {
				has_state_flag = URS_suppressed
			}
		}
		transfer_state_to = BLT
	}
	KZC = {
		declare_war_on = { target = URS type = annex_everything }

		inherit_technology = URS
		load_oob = KZC_uprising
	}
	BUS = {
		declare_war_on = { target = URS type = annex_everything }

		inherit_technology = URS
		load_oob = BUS_uprising
	}
	BLT = {
		declare_war_on = { target = URS type = annex_everything }

		inherit_technology = URS
		load_oob = BLT_uprising
	}
}
