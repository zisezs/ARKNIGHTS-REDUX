# ==============================================================
# 乌萨斯圣愚国 · 政治转型（脚本效果）
# --------------------------------------------------------------
# 由国策「圣愚帝国」(URS_fool_empire) 调用。邪魔化运动走完全程后，
# 乌萨斯不再是帝国，而是「圣愚锚定会」治下的信仰国家：
#
#   1) 执政党改宗神秘主义（mysticism），党名 → 圣愚锚定会
#      政党名走 <TAG>_<ideology>_party 键（原版 GER_fascism_party 先例），
#      本 mod 侧见 ARF_URS_Rework_l_simp_chinese.yml 的 URS_mysticism_party。
#   2) 国名 → 乌萨斯圣愚国（set_cosmetic_tag，配色见 common/countries/cosmetic.txt）
#   3) 国家领袖 → 圣愚（角色 URS_The_Fool 自带 mysticism_subtype 领袖角色，
#      所以既不必 recruit_character 也不必 add_country_leader_role）
#   4) 废除旧政权的全部遗留民族精神（清单由 docs/_urs_fool_state_build.py
#      从 common/ideas/URS_ideas (Ursus).txt 自动取全集）
#
# 为什么可以直接删：这些精神都没有 removal_cost = -1；且三个恢复脚本效果
# （URS_economic_recovery / URS_social_stabilization / URS_army_reformation）
# 在旧国策树下线后已无任何调用点，不会把精神重新加回来。
#
# 由 docs/_urs_fool_state_build.py 生成，勿手改。
# ==============================================================

URS_become_fool_state = {
	log = "URS: 圣愚帝国成立——政党/党名/国名/领袖/旧精神一次性转换"

	# ---- 1) 执政党 → 神秘主义，党名 → 圣愚锚定会 ----
	set_politics = {
		ruling_party = mysticism
		elections_allowed = no
	}
	set_popularities = {
		communism = 0
		socialism = 0
		progressivism = 0
		liberalism = 0
		conservatism = 0
		authoritarianism = 0
		despotism = 3
		fascism = 2
		mysticism = 95
		special = 0
	}
	set_party_name = {
		ideology = mysticism
		long_name = URS_mysticism_party
		name = URS_mysticism_party
	}

	# ---- 2) 国名 → 乌萨斯圣愚国 ----
	set_cosmetic_tag = URS_fool_state

	# ---- 3) 领袖 → 圣愚 ----
	hidden_effect = {
		if = {
			limit = {
				NOT = {
					has_country_leader = {
						character = URS_The_Fool
						ruling_only = yes
					}
				}
			}
			retire_country_leader = yes
		}
		promote_character = {
			character = URS_The_Fool
			ideology = mysticism_subtype
		}
		log = "URS: 国家领袖已换成圣愚"
	}

	# ---- 4) 与所有其他国家的关系 → -200（邪魔本质的外交后果） ----
	#  为什么不在特质里做：「与某国关系」是**国对国**的一个带衰减的值，
	#  HOI4 没有任何 country 作用域修饰符能表达「对所有人的固定关系 -200」；
	#  正确做法是新建 opinion_modifier token，再对全世界施加（先例：
	#  common/opinion_modifiers/RHL_opinion_modifiers.txt + reverse_add_opinion_modifier）。
	#  特质里另带 opinion_gain_monthly = -5（关系永远无法回暖）作为持续面。
	if = {
		limit = { NOT = { has_country_flag = URS_demonic_dread_flag } }
		set_country_flag = URS_demonic_dread_flag
		every_other_country = {
			limit = {
				exists = yes
				NOT = { has_war_with = ROOT }
			}
			reverse_add_opinion_modifier = {
				target = ROOT
				modifier = URS_demonic_dread
			}
		}
		log = "URS: 邪魔本质已对全世界施加 -200 关系"
	}

	# ---- 5) 废除旧政权的遗留民族精神 ----

			limit = { has_idea = URS_afterwave_of_great_rebellion }
