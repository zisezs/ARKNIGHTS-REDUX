# 配套文件（六处接线，漏一处即静默失效）：
#   决议正文  common/decisions/URS_Kazimierz_War_Decisions.txt
#   决议分类  common/decisions/categories/URS_Kazimierz_War_categories.txt
#   事件      events/URS_Kazimierz_War.txt            namespace = ursus_kzmwar
#   钩子      common/on_actions/URS_Kazimierz_War_on_actions.txt  （on_monthly_URS / on_weekly_URS）
#   战时精神  common/dynamic_modifiers/URS_Kazimierz_War_dynamic_modifiers.txt
#   文案      localisation/simp_chinese/ARF_URS_Kazimierz_War_l_simp_chinese.yml
#
# 三段设计（2026-09-26 重制版，一切都握在玩家手里）：
#   ① 境内六路叛军：各一条 0–100 进度变量 URS_revolt_<x>，每月按 URS_revolt_growth 增长。
#      满格有两条出路：玩家用「引蛇出洞」替它选日子（次日／一月／两月／三月之后），或放任不管 ——
#      月度判定会当场公示并排在一个月后自炸。两条路都只走一次，反过的那路此后进度钉在 0。
#      起义落地时割走该路起义区、建政、承继科技、向帝国宣战；被标过 URS_suppressed 的州夺不走，
#      若一整路都被标过，则不立国、不宣战（只有一声空响）。
#      真拿到地满两个月后这路又没了地盘，即判为平定：备战度 +1 加一批缴获 ——「以叛制战」。
#   ② 备战度 URS_kzmwar_prep_level：四级平行备战决议各 +1，每平定一路叛军也 +1（以叛制战）。
#      它既是对卡西米尔宣战的唯一闸门，也直接折算成战时精神 URS_kzmwar_war_spirit 的强度。
#   ③ 自动获胜：KZM 投降度 > 65% 时（周度轮询）割乌卡争议三地 + 白和 + 傀儡，
#      并补发 worldnews_kzm.10「骑士蒙羞」。
#
# 2026-09-26 修掉的三处静默失效（都在 error.log 里躺着）：
#   · any_owned_state / any_state 这类 any_* 触发符里不能再写 limit = { }（那是 every_* 效果用的），
#     写了报 "Unknown trigger-type: limit"，判定直接废掉 —— 原第 130/144/158/172/186/200 行。
#   · URS_kzmwar_weekly_check 原来发的是 ursus_kzmwar.30（那是发给卡西米尔的「帝国军集结」预警），
#     该发 .40；后果是 65% 自动获胜的结算整个是死代码。
#   · 月度兜底 URS_kzmwar_backstop（备好空转三个月 / 1102.6.1 无战事 / 1102.12.1 硬期限，
#     外加 URS_kzmwar_finish_prep 与 URS_kzmwar_idle_months、通知事件 .41）已整段删除：
#     本链改为纯玩家主动引爆，AI 走的是同一条链、靠 ai_will_do 自己动身，不再有人替它下决心的代劳。
#
# 引擎要点：
#   · 决议的 complete_effect 是「点下去当天」执行，days_remove 到点执行 remove_effect。
#   · news_event 必须用 news_event = { } 触发，写成 country_event 会静默不出。
#   · 起义效果全部带 exists = no 守卫，与旧的 Ursus_area_uprising（East_Ursus_Upheaval_scripted_effects.txt:551
#     起，那份没有 exists 守卫）并发也不会重复建国。
#   · check_variable 用 > / < 比较（>= 在本引擎若干处会静默失效），阈值一律写成 >99 / >64。
#   · has_country_flag = { flag = X days > 60 } 是「旗标已挂满 60 天」的判法（原版 AUS.txt:2513 先例），
#     平叛判定靠它挡住「起义当天就被判定为已平定」这种白送红利。

# ── 变量初始化：六路进度 + 增长率 + 备战度 + 战时精神的折算值 ──
URS_kzmwar_init = {
	# 乌卡争议三地（哥萨茨 / 别列斯尼亚 / 波罗的海）开局就更不安分
	set_variable = { URS_revolt_ucm = 15 }
	set_variable = { URS_revolt_ulb = 10 }
	set_variable = { URS_revolt_ufs = 10 }
	set_variable = { URS_revolt_kzc = 30 }
	set_variable = { URS_revolt_bus = 25 }
	set_variable = { URS_revolt_blt = 25 }
	set_variable = { URS_revolt_growth = 2 }

	set_country_flag = URS_kzmwar_prep_inited
	URS_kzmwar_init_prep_vars = yes
}

# 重制版新增的三个变量单独立旗初始化：老档（v1 只立了 URS_kzmwar_inited）进月度趟子时会自己补上，
# 而不会被整份 init 重置掉已经跑了好几年的叛乱进度。
URS_kzmwar_init_prep_vars = {
	set_variable = { URS_kzmwar_prep_level = 0 }
	set_variable = { URS_kzmwar_pacified_num = 0 }
	set_variable = { URS_kzmwar_erupt_delay = 0 }

	URS_kzmwar_update_war_spirit = yes
}

# ── 月度：进度推进 + 酝酿起义 + 平叛报捷 ──
URS_kzmwar_monthly_tick = {
	if = {
		limit = {
			NOT = { has_country_flag = URS_kzmwar_inited }
		}
		set_country_flag = URS_kzmwar_inited
		URS_kzmwar_init = yes
	}
	if = {
		limit = {
			NOT = { has_country_flag = URS_kzmwar_prep_inited }
		}
		set_country_flag = URS_kzmwar_prep_inited
		URS_kzmwar_init_prep_vars = yes
	}

	# 增长率：平时 2；本国卷入战争 叛军越乱 +2；卡西米尔正在对乌开战（境外输血）+2；清剿期间 -2
	set_variable = { URS_revolt_growth = 2 }
	if = {
		limit = { has_war = yes }
		add_to_variable = { URS_revolt_growth = 2 }
	}
	if = {
		limit = {
			has_country_flag = URS_kzmwar_crackdown_on
		}
		add_to_variable = { URS_revolt_growth = -2 }
	}
	if = {
		limit = { KZM = { has_war_with = URS } }
		add_to_variable = { URS_revolt_growth = 2 }
	}

	# 已建国的势力不再计「境内进度」（那是战场，不是隐患）；反过一次的也不再接着涨（钉在 0）
	if = {
		limit = {
			UCM = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_pacified_ucm }
		}
		add_to_variable = { URS_revolt_ucm = URS_revolt_growth }
	}
	else = {
		set_variable = { URS_revolt_ucm = 0 }
	}
	if = {
		limit = {
			ULB = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_pacified_ulb }
		}
		add_to_variable = { URS_revolt_ulb = URS_revolt_growth }
	}
	else = {
		set_variable = { URS_revolt_ulb = 0 }
	}
	if = {
		limit = {
			UFS = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_pacified_ufs }
		}
		add_to_variable = { URS_revolt_ufs = URS_revolt_growth }
	}
	else = {
		set_variable = { URS_revolt_ufs = 0 }
	}
	if = {
		limit = {
			KZC = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_pacified_kzc }
		}
		add_to_variable = { URS_revolt_kzc = URS_revolt_growth }
	}
	else = {
		set_variable = { URS_revolt_kzc = 0 }
	}
	if = {
		limit = {
			BUS = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_pacified_bus }
		}
		add_to_variable = { URS_revolt_bus = URS_revolt_growth }
	}
	else = {
		set_variable = { URS_revolt_bus = 0 }
	}
	if = {
		limit = {
			BLT = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_pacified_blt }
		}
		add_to_variable = { URS_revolt_blt = URS_revolt_growth }
	}
	else = {
		set_variable = { URS_revolt_blt = 0 }
	}

	clamp_variable = { var = URS_revolt_ucm min = 0 max = 100 }
	clamp_variable = { var = URS_revolt_ulb min = 0 max = 100 }
	clamp_variable = { var = URS_revolt_ufs min = 0 max = 100 }
	clamp_variable = { var = URS_revolt_kzc min = 0 max = 100 }
	clamp_variable = { var = URS_revolt_bus min = 0 max = 100 }
	clamp_variable = { var = URS_revolt_blt min = 0 max = 100 }

	# 满格即酝酿；手里没这块地就没什么可反的，进度回落到 90 等下次
	# （any_owned_state 里直接写状态触发符，不能再套 limit —— 套了整个判定静默失效）
	if = {
		limit = {
			check_variable = { URS_revolt_ucm > 99 }
			UCM = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_brewing_ucm }
		}
		if = {
			limit = { any_owned_state = { UCM_uprising_area = yes } }
			URS_kzmwar_brew_ucm = yes
			URS_kzmwar_brew_goes_public = yes
		}

		}
	}

			ULB = { exists = no }

			URS_kzmwar_brew_ulb = yes
			URS_kzmwar_brew_goes_public = yes
		}
		else = {
			set_variable = { URS_revolt_ulb = 90 }
		}
	}
	if = {
		limit = {
			check_variable = { URS_revolt_ufs > 99 }
			UFS = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_brewing_ufs }
		}
		if = {
			limit = { any_owned_state = { UFS_uprising_area = yes } }
			URS_kzmwar_brew_ufs = yes
			URS_kzmwar_brew_goes_public = yes
		}
		else = {

			KZC = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_brewing_kzc }
		}
		if = {
			limit = { any_owned_state = { KZC_uprising_area = yes } }
			URS_kzmwar_brew_kzc = yes
			URS_kzmwar_brew_goes_public = yes
		}
		else = {
			set_variable = { URS_revolt_kzc = 90 }
		}
	}
	if = {

			check_variable = { URS_revolt_bus > 99 }

			NOT = { has_country_flag = URS_kzmwar_brewing_bus }

		}
		else = {

		limit = {
			check_variable = { URS_revolt_blt > 99 }
			BLT = { exists = no }
			NOT = { has_country_flag = URS_kzmwar_brewing_blt }
		}
		if = {
			limit = { any_owned_state = { BLT_uprising_area = yes } }
			URS_kzmwar_brew_blt = yes
			URS_kzmwar_brew_goes_public = yes
		}
		else = {
			set_variable = { URS_revolt_blt = 90 }
		}
	}

	URS_kzmwar_pacify_sweep = yes
}

# ── 定下起事之日（同一时刻只允许一纸起事令在路上；日子按变量分派成字面天数）──
URS_kzmwar_schedule_eruption = {
	if = {
		limit = { NOT = { has_country_flag = URS_kzmwar_erupt_scheduled } }
		set_country_flag = URS_kzmwar_erupt_scheduled
		if = {
			limit = { check_variable = { URS_kzmwar_erupt_delay < 1 } }
			country_event = { id = ursus_kzmwar.20 days = 1 }
		}
		else_if = {
			limit = { check_variable = { URS_kzmwar_erupt_delay < 45 } }
			country_event = { id = ursus_kzmwar.20 days = 30 }
		}
		else_if = {
			limit = { check_variable = { URS_kzmwar_erupt_delay < 75 } }
			country_event = { id = ursus_kzmwar.20 days = 60 }
		}
		else = {
			country_event = { id = ursus_kzmwar.20 days = 90 }
		}
	}
}

# ── 六路起义落地：写法对齐 Ursus_area_uprising（割该路起义区 → 建国 → 承继科技 → 宣乌萨斯）──
# 只有真拿到地才算「反成了」：起义区被整肃净尽的那一路就地 fizzled，不立国、不宣战，
# 自然也不会在两个月后被误判成「已平定」而白送一份备战红利。
URS_kzmwar_erupt_ucm = {
	if = {
		limit = { UCM = { exists = no } }
		every_owned_state = {
			limit = {
				UCM_uprising_area = yes
				NOT = { has_state_flag = URS_suppressed }
			}
			add_core_of = UCM
			transfer_state_to = UCM
		}
		if = {
			limit = { UCM = { any_owned_state = { always = yes } } }
			set_country_flag = URS_kzmwar_erupted_ucm
			UCM = {
				set_politics = {
					ruling_party = communism
					elections_allowed = no
				}
				set_popularities = { communism = 100 }
				inherit_technology = URS
				load_oob = UCM_uprising
				declare_war_on = { target = URS type = annex_everything }
			}
		}
	}
	URS = { set_variable = { URS_revolt_ucm = 0 } }
}

		declare_war_on = { target = KZM type = annex_everything }
		if = {
			limit = { KZC = { exists = yes NOT = { is_puppet_of = URS } } }
			URS = { declare_war_on = { target = KZC type = annex_everything } }
		}
		if = {
			limit = { BUS = { exists = yes NOT = { is_puppet_of = URS } } }
			URS = { declare_war_on = { target = BUS type = annex_everything } }
		}
		if = {
			limit = { BLT = { exists = yes NOT = { is_puppet_of = URS } } }
			URS = { declare_war_on = { target = BLT type = annex_everything } }
		}
		# 开战即把备战度结算成战时精神，两年内有效；之后再平叛涨的档数靠变量重算自动跟上
		if = {
			limit = { NOT = { has_country_flag = URS_kzmwar_spirit_on } }
			set_country_flag = URS_kzmwar_spirit_on
			URS_kzmwar_update_war_spirit = yes
			add_dynamic_modifier = {
				modifier = URS_kzmwar_war_spirit
				days = 730
			}
		}
	}
}

# ── 自动获胜：割乌卡争议三地 → 白和 → 傀儡 ──
# 顺序照抄 Peace_on_actions.txt:152-167（先割地再白和，白和不会回退脚本割走的州）
URS_kzmwar_autovictory_settle = {
	if = {
		limit = { NOT = { has_country_flag = URS_kzmwar_autowon } }
		set_country_flag = URS_kzmwar_autowon
		KZM = {
			every_owned_state = {
				limit = {
					OR = {
						is_Ursus_Kozacy_area = yes
						is_Ursus_Belaursnia_area = yes
						is_Ursus_Balti_area = yes
					}
				}
				transfer_state_to = URS
			}
		}
		URS = {
			if = {
				limit = { has_war_with = KZM }
				white_peace = KZM
			}
			if = {
				limit = { has_war_with = KZC }
				white_peace = KZC
			}
			if = {
				limit = { has_war_with = BUS }
				white_peace = BUS
			}
			if = {
				limit = { has_war_with = BLT }
				white_peace = BLT
			}
			puppet = KZM
		}
		news_event = { id = worldnews_kzm.10 }
	}
}
