# ---------------- ① 本国出货并累加进全球池 ----------------
ARF_Trade_Supply = {
	# 自愈：开局之后才出现的国家（附庸释放、内战分家、读档新建）没跑过 on_startup，
	# 权重会空缺成 0 —— 那样它一滴商品都投不出去。这里补一次初始化。
	if = {
		limit = { NOT = { has_variable = arf_tm_w_yan } }
		ARF_Trade_Init = yes
	}
	set_variable = { arf_tm_tg = num_of_civilian_factories }
	# 竞争力 = 1 + 0.2 × 科研速度修正（修正 0 = 100% 科研 → 1.0；+200% → 1.4）
	# ⚠️ 这里的科研修正**包含本模块自己发的科研加成**（上月结算值），
	#    即「出口多 → 科研快 → 竞争力更高 → 出口更多」的闭环。
	#    科研加成自身封顶 +30% ⇒ 竞争力最多只多 +0.06，是有界正反馈，不会发散。
	set_variable = { arf_tm_comp = { value = modifier@research_speed_factor multiply = 0.20 add = 1 } }
	set_variable = { arf_tm_comp_pct = { value = arf_tm_comp multiply = 100 round = yes } }
	# 权重和（未定义按 0）；下限 1 防除零
	set_variable = { arf_tm_wsum = { value = arf_tm_w_yan add = arf_tm_w_urs add = arf_tm_w_lei add = arf_tm_w_ibe add = arf_tm_w_col add = arf_tm_w_sar add = arf_tm_w_vic } }
	if = {
		limit = { check_variable = { arf_tm_wsum < 1 } }
		set_variable = { arf_tm_wsum = 1 }
	}
	set_variable = { arf_tm_sum_eff = 0 }
	# yan：有效出货 = T × w / Σw × comp

	set_variable = { arf_tm_stab_pct = { value = arf_tm_stab multiply = 100 round = yes } }

	# 基础加成 = 出口额 × 系数（封顶）；三项共用「出口额」= arf_tm_sum_fill

	set_variable = { arf_tm_pp_base = { value = arf_tm_sum_fill multiply = 0.0100 clamp = { min = 0 max = 0.60 } } }
	set_variable = { arf_tm_res_base = { value = arf_tm_sum_fill multiply = 0.0025 clamp = { min = 0 max = 0.30 } } }

	# 实际加成 = 基础加成 × 稳定度（稳定度 100% 时两者相等）
	# ⚠️ 动态修正吃的是下面这个**折算后**的值，不是 _base
	set_variable = { arf_tm_build_bonus = { value = arf_tm_build_base multiply = arf_tm_stab clamp = { min = 0 max = 0.40 } } }
	set_variable = { arf_tm_pp_bonus = { value = arf_tm_pp_base multiply = arf_tm_stab clamp = { min = 0 max = 0.60 } } }

	# 显示量（整数百分比）：基础 / 折算后各一份
	set_variable = { arf_tm_build_base_pct = { value = arf_tm_build_base multiply = 100 round = yes } }
	set_variable = { arf_tm_build_pct = { value = arf_tm_build_bonus multiply = 100 round = yes } }
	set_variable = { arf_tm_pp_base_pct = { value = arf_tm_pp_base multiply = 100 round = yes } }
	set_variable = { arf_tm_pp_pct = { value = arf_tm_pp_bonus multiply = 100 round = yes } }
	set_variable = { arf_tm_res_base_pct = { value = arf_tm_res_base multiply = 100 round = yes } }
	set_variable = { arf_tm_res_pct = { value = arf_tm_res_bonus multiply = 100 round = yes } }
	set_variable = { arf_tm_sum_fill = { value = arf_tm_sum_fill round = yes } }
	# 动态修正默认按天刷新；这里强制立刻刷新，按钮点击后数字当帧就变
	force_update_dynamic_modifier = yes
}

# ---------------- 主流程（全库两趟；只从**一个**国家调用）----------------
ARF_Trade_Run = {
	# 记下本次结算日（月闸门与冷启动共用）
	set_temp_variable = { arf_tm_today = global.num_days }
	set_variable = { global.arf_tm_run_day = arf_tm_today }
	ARF_Trade_Reset_Markets = yes
	every_country = {
		limit = { exists = yes }
		ARF_Trade_Supply = yes
	}
	every_country = {
		limit = { exists = yes }
		ARF_Trade_Settle = yes
	}
}

# ---------------- 冷启动（on_startup 里只需一个国家安全地跑一趟）----------------
# on_startup 的默认作用域是 none，直接调脚本效果不稳妥；
# 这里从国家作用域里用一次性旗标把它收成「只跑一趟」。
ARF_Trade_Cold_Start = {
	if = {
		limit = { NOT = { has_global_flag = arf_tm_boot } }
		set_global_flag = arf_tm_boot
		ARF_Trade_Run = yes
	}
}

# ---------------- 月闸门 ----------------
# on_monthly 是「每个国家各触发一次」，全局结算必须只跑一趟。
# 用 global.num_days 与上次结算日比对：同月第二次起直接跳过。
# 这个判据与引擎的遍历顺序无关，不需要「谁先谁后」的假设。
ARF_Trade_Gate = {
	set_temp_variable = { arf_tm_today = global.num_days }
	set_temp_variable = { arf_tm_prev = global.arf_tm_run_day }
	if = {
		limit = { check_variable = { var = arf_tm_today value = arf_tm_prev compare = not_equals } }
		ARF_Trade_Run = yes
	}
}

	if = {
		limit = { check_variable = { arf_tm_w_vic < 9 } }
		add_to_variable = { arf_tm_w_vic = 1 }
		ARF_Trade_Run = yes
	}

	if = {
		limit = { check_variable = { arf_tm_w_vic > 0 } }
		subtract_from_variable = { arf_tm_w_vic = 1 }
		ARF_Trade_Run = yes
	}
}
