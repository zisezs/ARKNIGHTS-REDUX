# ==============================================================================
#  西北都护府 · 异变 —— 链的启动 / 推进 / 终幕
#  · YNW_upheaval_start_check        在 YAN 作用域调用（挂在 YAN_monthly_tick 上）
#  · YNW_upheaval_awakening_step     在 YNW 作用域调用，每步推高岁兽苏醒度
#  · YNW_upheaval_become_sui_state   终幕：易帜山海众 / 改名岁国 / 蒙恩-睚上台 /
#                                    建军 / 岁怒 / 对大炎宣战
#  · 链完结 = global flag YNW_upheaval_complete，「岁怒」超事件以它为触发条件。
#  · 三条硬约束（都踩过）：
#     1) 换领袖只走 retire_country_leader + promote_character 两步。**不要**再调
#        add_country_leader_role（角色文件已自带该角色，重复加会抛错）——抛错点
#        之后的效果会整段不执行，宣战/建军就是这么丢的。
#     2) **不要**在本文件里用 recruit_character，引擎只允许写在 game/history 文件
#        （error.log: "recruit_character should only happen in game/history files"）。
#     3) 完成旗标 YNW_upheaval_done_flag 放在整段最末尾：万一中途抛错，
#        on_monthly_YNW 的月度自愈才能识别「没做完」并重试。
#  · 建军那块另有 YNW_upheaval_army_flag 去重，保证自愈重试不会重复刷 100 个师。
# ==============================================================================

# 启动判定：ROOT = YAN
YNW_upheaval_start_check = {
	if = {
		limit = {
			NOT = { has_global_flag = YNW_upheaval_started }
			YNW = {
				exists = yes
				NOT = { has_country_flag = YNW_upheaval_active_flag }
				NOT = { has_country_flag = YNW_upheaval_done_flag }
			}
		}
		# 硬口径：1098.1.1 之后才起链（用户实测过苏醒度门槛会被 gameplay 提前顶穿，
		# 1095 就点火）。苏醒度不再当起链开关，只被链条本身往上推，作为大炎侧的感应。
		if = {
			limit = {
				date > 1098.1.1
			}
			set_global_flag = YNW_upheaval_started
			YNW = {
				set_country_flag = YNW_upheaval_active_flag
				set_variable = { YNW_upheaval_stall = 0 }
				log = "YNW_upheaval: 起链，yan_upheaval.1 已排入"
				country_event = { id = yan_upheaval.1 days = 7 }
			}
		}
	}
}

# 献祭在推进祂的苏醒：ROOT = YNW，改的是大炎那个国家级变量
YNW_upheaval_awakening_step = {
	if = {
		limit = {
			YAN = { exists = yes }
		}
		YAN = {
			add_to_variable = { yan_sui_awakening = 6 }
			clamp_variable = { var = yan_sui_awakening min = 0 max = 100 }
		}
	}
}

# 终幕：ROOT = YNW
YNW_upheaval_become_sui_state = {
	log = "YNW_upheaval: 进入终幕 YNW_upheaval_become_sui_state"

	if = {
		limit = { has_idea = YNW_bewitched_legions }
		remove_ideas = YNW_bewitched_legions
	}

	# 都护府不再奉炎正朔。不假设宗主是谁：从「当前宗主」手里放出来。
	# 写死 YAN = { end_puppet = YNW } 会空转（宗主未必是大炎），
	# 空转后 YNW 仍带 autonomy_yan_protectorate 的 can_not_declare_war，
	# 后面的 declare_war_on 会被规则静默挡掉。
	if = {
		limit = { is_subject = yes }
		overlord = {
			end_puppet = PREV
		}
	}
	# 第二道保险：上面那招没生效的话，用原版也有先例的 release_puppet 再兜一次
	if = {
		limit = {
			is_subject = yes
			YAN = { exists = yes }
		}
		YAN = {
			release_puppet = YNW
		}
	}
	if = {
		limit = { is_subject = yes }
		log = "YNW_upheaval: 警告 —— 两道解除附属都没生效，仍是附属国"
	}
	if = {
		limit = { is_subject = no }
		log = "YNW_upheaval: 附属已解除"
	}

	set_politics = {
		ruling_party = mysticism
		elections_allowed = no
	}
	set_popularities = {
		communism = 0
		socialism = 0
		progressivism = 0
		liberalism = 2
		conservatism = 0
		authoritarianism = 0
		despotism = 3
		fascism = 5
		mysticism = 90
		special = 0
	}
	set_cosmetic_tag = YNW_sui_state
	add_ideas = YNW_sui_grace

	hidden_effect = {
		set_global_flag = YNW_upheaval_complete

		# 换人：退在任者 + 推蒙恩-睚。角色文件已自带 mysticism_subtype 领袖角色
		# 与 YNW_the_blessed 特质，所以既不必 add_country_leader_role 也不必 recruit_character
		if = {
			limit = {
				NOT = {
					has_country_leader = {
						character = YNW_Meng_En_Ya
						ruling_only = yes
					}
				}
			}
			retire_country_leader = yes
		}
		promote_character = {
			character = YNW_Meng_En_Ya
			ideology = mysticism_subtype
		}
		log = "YNW_upheaval: 领袖已换成蒙恩-睚"

		# 祂落地这一下，大炎那边是有感应的
		if = {
			limit = {
				YAN = { exists = yes }
			}
			YAN = {
				add_to_variable = { yan_sui_awakening = 15 }
				clamp_variable = { var = yan_sui_awakening min = 0 max = 100 }
			}
		}

		# 「岁怒」超事件：大炎与岁国各看一次，旗标去重
		every_country = {
			limit = {
				exists = yes
				OR = {
					original_tag = YAN
					original_tag = YNW
				}
				NOT = { has_country_flag = yan_sui_wrath_news_flag }
			}
			set_country_flag = yan_sui_wrath_news_flag
			news_event = { id = yanse.1 }
		}

		# 祂的袍泽从雪里站起来：先模板、再库存、后生成，顺序不能反。
		# 整块用旗标去重 —— 月度自愈重试时不能重复刷 100 个师或重复塞装备
		if = {
			limit = { NOT = { has_country_flag = YNW_upheaval_army_flag } }
			set_country_flag = YNW_upheaval_army_flag

			division_template = {
				name = "山海狂士"
				is_locked = yes
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
					infantry = { x = 0 y = 3 }
					infantry = { x = 1 y = 0 }
					infantry = { x = 1 y = 1 }
					infantry = { x = 1 y = 2 }
					infantry = { x = 1 y = 3 }
					infantry = { x = 2 y = 0 }
					infantry = { x = 2 y = 1 }
					infantry = { x = 2 y = 2 }
					infantry = { x = 2 y = 3 }
					infantry = { x = 3 y = 0 }
					infantry = { x = 3 y = 1 }
					infantry = { x = 3 y = 2 }
					infantry = { x = 3 y = 3 }
				}
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = 170000
				producer = YNW
			}
			capital_scope = {
				create_unit = {
					division = "name = \"山海狂士\" division_template = \"山海狂士\" start_experience_factor = 0.3 start_equipment_factor = 1"
					owner = YNW
					count = 100
				}
			}
			log = "YNW_upheaval: 山海狂士 100 个师已生成"
		}

		add_ideas = YNW_sui_wrath
	}

	# 脱离阵营：都护府原本既是保护国、又在大炎的阵营里，
	# 而 HOI4 不允许对同阵营国家宣战 —— 不退阵营的话下面的 declare_war_on 会被静默挡掉
	if = {
		limit = { is_in_faction = yes }
		leave_faction = yes
		log = "YNW_upheaval: 已脱离原阵营"
	}

	# 对大炎宣战：脱离附属 + 脱离阵营之后才做得了
	if = {
		limit = {
			YAN = { exists = yes }
			is_subject = no
			NOT = { has_war_with = YAN }
		}
		declare_war_on = {
			target = YAN
			type = annex_everything
		}
		log = "YNW_upheaval: 已对大炎宣战"
	}

	# 全部走完才置完成旗标 —— 中途抛错时这里不会执行，
	# on_monthly_YNW 才能识别「没做完」并重试整段
	set_country_flag = YNW_upheaval_done_flag
	log = "YNW_upheaval: 终幕完成"
}
