# ---- 莱昂图索·贝洛内出任国家领袖（执政党不变：威权主义）----
# add_country_leader_role 带 promote_leader = yes 一步到位。
# 不要再补 retire_country_leader + promote_character：那两段会在国策 tooltip 被反复求值时
# 刷出「Character does not have a country leader role」（2026-09-19 error.log 418 次）。
SRC_leontuzzo_takeover_effect = {
	if = {
		limit = {
			NOT = {
				has_country_flag = SRC_flag_leontuzzo_authoritarian_role
			}
		}
		add_country_leader_role = {
			character = SRC_Leontuzzo_Bellone
			promote_leader = yes
			country_leader = {
				ideology = technocracy_subtype
				traits = {
					SRC_grey_hall_spokesman
					SRC_dignity_of_leadership
				}
				expire = "1965.1.1.1"
			}
		}
		set_country_flag = SRC_flag_leontuzzo_authoritarian_role
	}
}

# ---- 宣布叙拉古联邦共和国（终局国策 / 制宪会议决议共用，幂等）----
# 顺序要紧：先改执政党，再带着 promote_leader 补角色，否则升的是旧党的领袖。
SRC_federal_republic_proclaim_effect = {
	if = {
		limit = {
			NOT = {
				has_country_flag = SRC_flag_federal_republic
			}
		}
		set_country_flag = SRC_flag_federal_republic
	}
	set_cosmetic_tag = Siracusa_federal_republic
	set_politics = {
		ruling_party = corporate_technocracy
		last_election = "1103.1.1"
		election_frequency = 48
		elections_allowed = yes
	}
	set_popularities = {
		corporate_technocracy = 55
		liberalism = 20
		conservatism = 15
		socialism = 5
		fascism = 5
	}
	if = {
		limit = {
			NOT = {
				has_country_flag = SRC_flag_leontuzzo_technocrat_role
			}
		}
		add_country_leader_role = {
			character = SRC_Leontuzzo_Bellone
			promote_leader = yes
			country_leader = {
				ideology = corporate_technocracy_subtype
				traits = {
					SRC_grey_hall_spokesman
					SRC_dignity_of_leadership
				}
				expire = "1965.1.1.1"
			}
		}
		set_country_flag = SRC_flag_leontuzzo_technocrat_role
	}
}

# ---- 老一辈怒火中烧：月度结算，约每 90 天流失一座非首都核心州 ----
SRC_old_guard_fury_tick_effect = {
	add_to_variable = { SRC_fury_months = 1 }
	if = {
		limit = {
			check_variable = { SRC_fury_months > 2 }
		}
		set_variable = { SRC_fury_months = 0 }
		random_owned_state = {
			limit = {
				is_core_of = ROOT
				NOT = {
					is_capital = yes
				}
			}
			remove_core_of = ROOT
			hidden_effect = {
				ROOT = {
					country_event = { id = src_federal.6 }
				}
			}
		}
	}
}
