######################
## 塔拉复国 · 外交簇逻辑（scripted_effects）
## 调用方：common/national_focus/DUB_redeem_diplo_shared.txt 的 completion_reward
## 作用域：全部以 ROOT ＝ DUB（深池）为发起方调用
##
## 三条从本模组实机验证里抄来的硬规矩（见记忆 east-ursus-upheaval-autowar）：
##   1) 宣战写成「发起方作用域 ＋ target ＝ 字面 tag 或 PREV」，不要用 save_event_target_as 的目标当靶子；
##      宣战后立刻用 has_war_with 复查，没开成战要在日志里留痕。
##   2) 附庸既不能自主宣战、也不能被第三国宣战 —— 双方都要 is_subject = no。
##   3) annex_country 不给核心：先转地／add_core_of，再吞。
##
## 作用域提醒：在 every_country = { ... } 里面，当前作用域＝被遍历国、PREV＝进循环前的 ROOT；
##   所以"由我们对该国做的事"一律写成 ROOT = { <effect> = { target = PREV } }。
##
## 维多利亚宣称国名册（VAN 安科茨男爵领没有 history 文件，永不 exists，故不入册）
##
## 2026-09-26 外交簇重制为战争带「赤铁之路」后的现状：
##   TAR_dip_break_march_effect → 由战争带 TAR_rd_war_thread（把引线递回谈判桌）调用；
##   TAR_dip_nap_vca_tar_effect   → 暂时无人调用（互不侵犯与"一路吞到底"的口径冲突），保留备用，不是漏接。
######################

# ── 内部工具：把"我们脚下、别人名下"的省转成我们的核心 ──
TAR_dip_core_held_effect = {
	every_owned_state = {
		limit = {
			NOT = { is_core_of = ROOT }
			OR = {
				is_core_of = VIC is_core_of = VCA is_core_of = VWI is_core_of = VGO
				is_core_of = VAB is_core_of = VFI is_core_of = VAS is_core_of = VCR
				is_core_of = VVI is_core_of = VEA is_core_of = VWA is_core_of = VFA
				is_core_of = VSA is_core_of = VRN is_core_of = VRA is_core_of = VRB
				is_core_of = VRC is_core_of = VRD is_core_of = VRE is_core_of = TAR
			}
		}
		add_core_of = ROOT
	}
}

# ── A 段：让联合进军再失败一次（把维多利亚的分裂按下去） ──
TAR_dip_break_march_effect = {
	if = {
		limit = {
			VIC = { exists = yes }
			NOT = { has_global_flag = VIC_civil_war_start }
			VCA = { exists = no }
		}
		VIC = { country_event = { id = event_vic.3 hours = 6 } }
		set_country_flag = TAR_rd_dip_broke_march
	}
}

# ── A 段：替开斯特与威灵顿居中缔约 ──
# hidden_vic.2（谈判破裂）里的判据是 VCA 作用域下的 NOT has_non_aggression_pact_with = TAR，
# 满足才会同时对威灵顿与深池宣战。所以这条纸必须签在 VCA↔TAR 之间，签在我们身上不算数。
TAR_dip_nap_vca_tar_effect = {
	if = {
		limit = { VCA = { exists = yes } TAR = { exists = yes } }
		VCA = { diplomatic_relation = { country = TAR relation = non_aggression_pact active = yes } }
	}
	else_if = {
		limit = { VCA = { exists = yes } }
		VCA = { diplomatic_relation = { country = ROOT relation = non_aggression_pact active = yes } }
	}
}

# ── B 段：开斯特 ──
TAR_dip_war_vca_effect = {
	if = {
		limit = {
			VCA = { exists = yes is_subject = no }
			is_subject = no
			NOT = { has_war_with = VCA }
		}
		declare_war_on = { target = VCA type = annex_everything }
		if = {
			limit = { has_war_with = VCA }
			add_war_support = 0.05
		}
		else = {
			log = "TAR dip: 对开斯特宣战被引擎拒绝"
		}
	}
	reverse_add_opinion_modifier = { target = VCA modifier = TAR_dip_bad_blood }
}

TAR_dip_deal_vca_effect = {
	if = {
		limit = { VCA = { exists = yes is_subject_of = ROOT } }
		annex_country = { target = VCA transfer_troops = yes }
	}
	TAR_dip_core_held_effect = yes
	add_political_power = 60
	add_intel = { target = VCA amount = 40 }
}

# 剑：已经是家臣的直接吞，还独立的一律开战
TAR_dip_unify_sword_effect = {
	every_country = {
		limit = {
			OR = {
				tag = VCA tag = VWI tag = VGO tag = VAB tag = VFI tag = VAS tag = VCR
				tag = VVI tag = VEA tag = VWA tag = VFA tag = VSA tag = VRN
				tag = VRA tag = VRB tag = VRC tag = VRD tag = VRE tag = VIC
			}
			exists = yes
		}
		if = {
			limit = { is_subject_of = ROOT }
			ROOT = { annex_country = { target = PREV transfer_troops = yes } }
		}
		else_if = {
			limit = {
				is_subject = no
				NOT = { has_war_with = ROOT }
				ROOT = { is_subject = no }
			}
			ROOT = { declare_war_on = { target = PREV type = annex_everything } }
		}
	}
	TAR_dip_core_held_effect = yes
	set_cosmetic_tag = TAR_unified_realm
}

# 盟：不吞、不降，拉成一个共同防务的圈子
TAR_dip_unify_commonwealth_effect = {
	if = {
		limit = { NOT = { is_in_faction = yes } }
		set_rule = { can_create_factions = yes }
		create_faction_from_template = faction_template_tarlanian_commonwealth
		add_faction_initiative = 1
	}
	every_country = {
		limit = {
			OR = {
				tag = VCA tag = VWI tag = VGO tag = VAB tag = VFI tag = VAS tag = VCR
				tag = VVI tag = VEA tag = VWA tag = VFA tag = VSA tag = VRN
				tag = VRA tag = VRB tag = VRC tag = VRD tag = VRE tag = TAR tag = VIC
			}
			exists = yes
			is_subject = no
			NOT = { has_war_with = ROOT }
			NOT = { is_in_faction_with = ROOT }
		}
		PREV = { add_to_faction = ROOT }
	}
	TAR_dip_core_held_effect = yes
	set_cosmetic_tag = TAR_unified_realm
}
