# 开局初始化（on_startup 调用，只在未赋值时写初值）
YAN_init_mechanics = {
	if = {
		limit = { NOT = { has_variable = yan_sui_awakening } }
		set_variable = { yan_sui_awakening = 40 }
	}
	if = {
		limit = { NOT = { has_variable = yan_shudao_integrity } }
		set_variable = { yan_shudao_integrity = 100 }
	}
	if = {
		limit = { NOT = { has_variable = yan_bufan_prep } }
		set_variable = { yan_bufan_prep = 0 }
	}
	if = {
		limit = { NOT = { has_variable = yan_agent_trust } }
		set_variable = { yan_agent_trust = 30 }
	}
	# 岁压初值：岁醒树已删除，不再在开局挂「岁临天下」DEBUFF。
	# 底层变量/效果保留在 YAN_sui_pressure_scripted_effects.txt，供后续重制调用。
}

# 月度结算（on_monthly 调用）

	# 钳制到合法区间

	clamp_variable = { var = yan_bufan_prep min = 0 max = 100 }
	clamp_variable = { var = yan_agent_trust min = 0 max = 100 }
