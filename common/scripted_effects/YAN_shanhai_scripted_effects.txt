# ==============================================================================
#  大炎 ·【岁醒】叛军循环 v2 —— 「某某地区山海众」（十九司各一股 + 通用）
#  · YAN_shanhai_check     月度入口（YAN_monthly_tick 调用）：先平叛趟子，再起新波
#  · YAN_shanhai_liberate  把「控制权已回到大炎手里」的州，所有权也交还，并锁定该区
#  · YAN_shanhai_wave      一波：未达上限就占一州，达到就往既有山寨增兵
#  · YAN_shanhai_seize     占州（THIS = 州）：立国/宣战/给地/爆兵/扣损失（不弹事件）
#  · YAN_shz_found_here    建政（THIS = 该股叛军）：幂等，旗标去重
#  · YAN_shz_spawn_tiers   爆兵入口：转 YAN_shz_spawn_dispatch（各司编制/数量见 identity 文件）
#  · YAN_shz_costs         全局损失（YAN 作用域）：稳定度/战争支持度/工厂/岁压/首次新闻
#  · YAN_shanhai_disband   「岁陵之约」收束：交还所有州 + 遣散本司乱兵 + 停战
#
#  口径：每 120 天一波（yan_shz.23 静默计时，到点即起下一波；起事全程不弹事件）；同时在占 6 州封顶；每波 -0.1 稳定度 / -0.1 战争支持度 /
#        破坏 2 民 1 军工厂 / 岁压 +2。十九司各有民族精神与编制：首波 7–10 师，
#        封顶波每座山寨补 4–8 师（详见 YAN_shanhai_identity_scripted_effects.txt）。
#  两条引擎实证（v1 就是栽在这里，game.log 2026-09-21）：
#    ① 零领土 tag 的 exists = no —— 叛军必须真的拿到州的所有权才存在，所以用 transfer_state_to；
#    ② on_monthly 按现存国家各跑一次（本 mod 场上 101 国）—— 月度钩子已改 on_monthly_YAN。
#  一地一经平定即挂 YAN_shz_region_done 永久锁死：既写"山海众在那一带散了"，
#  也避免去给一个已经被吞掉的 tag 再递州（那种 scope 会静默什么都不发生）。
#  生成器：占州/收束等本体 E:/tmp/sui12/gen_shanhai.js（旧）；
#          found_here / spawn_tiers / reinforce / disband 由 docs/gen_yan_shanhai_identity.cjs 生成。

YAN_shanhai_check = {
	# ROOT = YAN
	YAN_shanhai_liberate = yes
	YAN_shz_recount = yes
	if = {
		limit = {
			NOT = { has_country_flag = yan_sui_agreement_done }
			# 【岁陵之乱！】120 天任务结算后，山海众偃旗息鼓，不再起新波
			NOT = { has_country_flag = yan_sui_banner_lowered }
			# 起事冷却：起一波挂旗，120 天后由静默事件 yan_shz.23 摘旗并起下一波
			NOT = { has_country_flag = YAN_shz_wave_cd }
		}
		YAN_shanhai_wave = yes
		set_country_flag = YAN_shz_wave_cd
		country_event = { id = yan_shz.23 days = 120 }
	}
}

YAN_shanhai_wave = {
	# ROOT = YAN
	if = {
		limit = { NOT = { has_variable = YAN_shz_wave } }
		set_variable = { YAN_shz_wave = 0 }
	}
	add_to_variable = { YAN_shz_wave = 1 }

	if = {
		limit = { check_variable = { YAN_shz_held < 6 } }
		random_owned_controlled_state = {
			limit = {
				is_core_of = YAN
				NOT = { is_capital = yes }
				NOT = { has_state_flag = yan_zhenlong_direct_state }
				NOT = { has_state_flag = YAN_shz_region_done }
			}
			YAN_shanhai_seize = yes
		}
	}
	else = {
		log = "YAN_shanhai: 在占州数已达上限，本波只增兵不扩地"
		YAN_shanhai_reinforce = yes
	}
}
