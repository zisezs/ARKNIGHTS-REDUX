# 哥伦比亚 / 莱茵生命 —— 十科国策树（RHL 路线）初始化与月度结算
# RHL_td_init：由 RHL 历史文件调用（CLB 玩家路线由根国策 RHL_td_council 完成时等效初始化）
# RHL_td_monthly_tick：由 common/on_actions/RHL_td_on_actions.txt 每月调用，仅做量程钳制

RHL_td_init = {
	if = {
		limit = {
			NOT = {
				has_country_flag = CLB_flag_td_init_done
			}
		}
		set_country_flag = CLB_flag_td_init_done

		# 四大量表（与 CLB_rhine_init 同初值，不进精神，仅作读数）
		if = {
			limit = { NOT = { has_variable = CLB_beast_acquiescence } }
			set_variable = { CLB_beast_acquiescence = 40 }
		}
		if = {
			limit = { NOT = { has_variable = CLB_maylander_influence } }
			set_variable = { CLB_maylander_influence = 50 }
		}
		if = {
			limit = { NOT = { has_variable = CLB_sky_fever } }
			set_variable = { CLB_sky_fever = 10 }
		}
		if = {
			limit = { NOT = { has_variable = CLB_neurotransmitter } }
			set_variable = { CLB_neurotransmitter = 0 }
		}

		# 十科科室变量初值（防止 add_to_variable 作用于未定义变量）
		if = {
			limit = { NOT = { has_variable = RHL_td_s1_res } }
			set_variable = { RHL_td_s1_res = 0 }

			set_variable = { RHL_td_s2_stab = 0 }
			set_variable = { RHL_td_s2_energy = 0 }
			set_variable = { RHL_td_s2_fuel = 0 }
			set_variable = { RHL_td_s2_air = 0 }
			set_variable = { RHL_td_s2_extract = 0 }
			set_variable = { RHL_td_s2_opinion = 0 }
			set_variable = { RHL_td_s3_res = 0 }
			set_variable = { RHL_td_s3_stab = 0 }
			set_variable = { RHL_td_s3_det = 0 }
			set_variable = { RHL_td_s3_factory = 0 }
			set_variable = { RHL_td_s3_atk = 0 }
			set_variable = { RHL_td_s4_res = 0 }
			set_variable = { RHL_td_s4_build = 0 }
			set_variable = { RHL_td_s4_air = 0 }
			set_variable = { RHL_td_s4_det = 0 }
			set_variable = { RHL_td_s5_res = 0 }
			set_variable = { RHL_td_s5_stab = 0 }
			set_variable = { RHL_td_s5_det = 0 }
			set_variable = { RHL_td_s5_air = 0 }
			set_variable = { RHL_td_s5_opinion = 0 }
			set_variable = { RHL_td_s6_res = 0 }
			set_variable = { RHL_td_s6_stab = 0 }
			set_variable = { RHL_td_s6_det = 0 }
			set_variable = { RHL_td_s6_agri = 0 }
			set_variable = { RHL_td_s7_atk = 0 }
			set_variable = { RHL_td_s7_def = 0 }
			set_variable = { RHL_td_s7_res = 0 }
			set_variable = { RHL_td_s7_xp = 0 }
			set_variable = { RHL_td_s7_stab = 0 }
			set_variable = { RHL_td_s7_ws = 0 }
			set_variable = { RHL_td_s8_pp = 0 }
			set_variable = { RHL_td_s8_stab = 0 }

			set_variable = { RHL_td_s8_ws = 0 }

			set_variable = { RHL_td_s9_factory = 0 }

			set_variable = { RHL_crb_factory = 0 }
			set_variable = { RHL_crb_pp = 0 }

			set_variable = { RHL_field_tier = 0 }
		}

}

RHL_td_monthly_tick = {
	clamp_variable = { var = CLB_beast_acquiescence min = 0 max = 100 }
	clamp_variable = { var = CLB_maylander_influence min = 0 max = 100 }
	clamp_variable = { var = CLB_sky_fever min = 0 max = 100 }
	clamp_variable = { var = CLB_neurotransmitter min = 0 max = 100 }
	# 安保部门外勤部队：月度点验（现役分队数 = 全国师数，战损自动释放名额）
	if = {
		limit = {
			has_country_flag = RHL_flag_field_active
		}
		set_variable = { RHL_field_count = 0 }
		every_country_division = {
			PREV = {
				add_to_variable = { RHL_field_count = 1 }
			}
		}
	}
}
