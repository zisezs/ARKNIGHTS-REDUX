RHL_td_monthly_tick = {
	clamp_variable = { var = CLB_beast_acquiescence min = 0 max = 100 }
	clamp_variable = { var = CLB_maylander_influence min = 0 max = 100 }
	clamp_variable = { var = CLB_sky_fever min = 0 max = 100 }
	clamp_variable = { var = CLB_neurotransmitter min = 0 max = 100 }
	# 安保部门外勤部队：月度点验（现役分队数 = 全国师数，战损自动释放名额）
	if = {
		limit = {
			has_country_flag = RHL_flag_field_active
		}
		set_variable = { RHL_field_count = 0 }
		every_country_division = {
			PREV = {
				add_to_variable = { RHL_field_count = 1 }
			}
		}
	}
}

#  十科自动运转（核心集团式自推，2026-10-27）

#  根国策【十科合议厅】完成后调用 RHL_td_autoprog_start：十科并行，
#  每 365 天自动完成一格，顺序 l1→l2→l3→l4→r1→r2→r3→r4→fN。
#  RHL_td_grand（总终局）与四结局不进入自动链，仍由玩家手动点。
#
#  自动链只在当前国策树为 RHL_td 时推进；切走即停并清掉该科旗标，
#  切回后由 RHL_td_on_actions 的月度兜底重新启动（RHL_td_autoprog_start 幂等）。
#  s10（总辖科）的 l2..l4 / r1..r2 / r4 需要 RHL_rocket_first_light：
#  火箭尚未首飞时停在原地，每 365 天重试一次，不会越级完成。
#
#  配套事件：events/RHL_td_auto_events.txt（RHL_td_auto.1 ~ .10，hidden）。
#  改十科格子时，同步改这里的顺序表与事件编号。
