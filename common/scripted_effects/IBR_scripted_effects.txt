# 黄金时代终将归来：1 → 4 逐级推进，无精神时获得一级
# 树里有 7 个国策挂这条阶梯而满级只有 4 级：已满级时再调用什么都不做，不会叠加一级
IBR_upgrade_golden_age_effect = {
	if = {
		limit = {
			has_idea = IBR_golden_age_3
		}
		swap_ideas = {
			remove_idea = IBR_golden_age_3
			add_idea = IBR_golden_age_4
		}
	}
	else_if = {
		limit = {
			has_idea = IBR_golden_age_2
		}
		swap_ideas = {
			remove_idea = IBR_golden_age_2
			add_idea = IBR_golden_age_3
		}
	}
	else_if = {
		limit = {
			has_idea = IBR_golden_age_1
		}
		swap_ideas = {
			remove_idea = IBR_golden_age_1
			add_idea = IBR_golden_age_2
		}
	}
	else_if = {
		limit = {
			NOT = {
				has_idea = IBR_golden_age_4
			}
		}
		add_ideas = {
			IBR_golden_age_1
		}
	}
}

# 大静谧余波消退一级：余波 → 余波渐息 → 余波已散
# 只推进一级，剩下的交给“修复灯塔”决议，避免国策把决议链整条吃掉
IBR_calm_the_aftermath_effect = {
	if = {
		limit = {
			has_idea = IBR_aftermath_of_Profound_Silence
		}
		swap_ideas = {
			remove_idea = IBR_aftermath_of_Profound_Silence
			add_idea = IBR_aftermath_mended
		}
		set_variable = {
			var = IBR_lighthouse_restored
			value = 1
		}
	}
	else_if = {
		limit = {
			has_idea = IBR_aftermath_mended
		}
		swap_ideas = {
			remove_idea = IBR_aftermath_mended
			add_idea = IBR_aftermath_of_Profound_Silence_faded
		}
		set_variable = {
			var = IBR_lighthouse_restored
			value = 2
		}
	}
}
