development_initialization = {
	set_variable = { agricultural_development_var = 0 }
	set_variable = { industrial_development_var = 0 }
	set_variable = { educational_development_var = 0 }
	set_variable = { administrative_development_var = 0 }
	set_variable = { poverty_development_var = 0 }
	set_variable = { agricultural_development_monthly_effect = 0 }
	set_variable = { industrial_development_monthly_effect = 0 }
	set_variable = { educational_development_monthly_effect = 0 }
	set_variable = { administrative_development_monthly_effect = 0 }
	set_variable = { poverty_development_monthly_effect = 0 }
}

update_development = {
	add_to_variable = {
		var = agricultural_development_var
		value = modifier@agricultural_development_monthly_modifier
	}
	add_to_variable = {
		var = agricultural_development_var
		value = agricultural_development_monthly_effect
	}
	check_agricultural_change = yes
	add_to_variable = {
		var = industrial_development_var
		value = modifier@industrial_development_monthly_modifier
	}
	add_to_variable = {
		var = industrial_development_var
		value = industrial_development_monthly_effect
	}
	check_industrial_change = yes
	add_to_variable = {
		var = educational_development_var
		value = modifier@educational_development_monthly_modifier
	}
	add_to_variable = {
		var = educational_development_var
		value = educational_development_monthly_effect
	}
	check_educational_change = yes
	add_to_variable = {
		var = administrative_development_var
		value = modifier@administrative_development_monthly_modifier
	}
	add_to_variable = {
		var = administrative_development_var
		value = administrative_development_monthly_effect
	}
	check_administration_change = yes
	add_to_variable = {
		var = poverty_development_var
		value = modifier@poverty_development_monthly_modifier
	}

		var = poverty_development_var
		value = poverty_development_monthly_effect
	}
	check_poverty_change = yes
}

agricultural_improve = {
	hidden_effect = {
		if = {
			limit = {
				has_idea = mechanized_agriculture
			}
			swap_ideas = {
				remove_idea = mechanized_agriculture
				add_idea = modern_agriculture
			}
			country_event = {
				id = development.1
			}
		}
		if = {
			limit = {
				has_idea = intensive_agriculture
			}
			swap_ideas = {
				remove_idea = intensive_agriculture
				add_idea = mechanized_agriculture
			}
			country_event = {
				id = development.2
			}
		}
		if = {

check_agricultural_change = {
	if = {
		limit = {
			check_variable = {
				var = agricultural_development_var
				value = 1
				compare = greater_than_or_equals
			}
		}
		agricultural_improve = yes
		add_to_variable = { agricultural_development_var = -1 }
	}
	if = {
		limit = {
			check_variable = {
				var = agricultural_development_var
				value = -1
				compare = less_than_or_equals
			}
		}
		agricultural_decrease = yes
		add_to_variable = { agricultural_development_var = 1 }
	}
}
