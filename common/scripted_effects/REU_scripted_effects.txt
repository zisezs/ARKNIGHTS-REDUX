REU_transfer = {
	hidden_effect = {
		annex_country = {
			target = BWS
		}
		set_capital = {
			state = 251
			remember_old_capital = no
		}

		if = {
			limit = {
				NOT = {
					OR = {
						has_completed_focus = REU_Light_Pack
						has_completed_focus = REU_Take_Everything
					}
				}
			}
			unlock_national_focus = REU_Light_Pack
		}
		if = {
			limit = {
				NOT = {
					OR = {
						has_completed_focus = REU_whole_to_parts
						has_completed_focus = REU_Avoid_involvement
					}
				}
			}
			unlock_national_focus = REU_whole_to_parts
		}
		complete_national_focus = REU_enter_Blagoveshchensk
		news_event = {
			id = worldnews_reu.2
		}
	}
}
