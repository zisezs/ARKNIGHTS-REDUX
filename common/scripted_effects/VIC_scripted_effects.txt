VIC_proposal_calculation = {
	set_variable = { VIC_proposal_yes = 0 }

	if = {
		limit = {
			has_country_flag = VIC_Gododdin_yes_flag
		}
		add_to_variable = { VIC_proposal_yes = VIC_Gododdin_influence }
		add_to_variable = { VIC_proposal_yes = VIC_Ashworth_influence }
		add_to_variable = { VIC_proposal_yes = VIC_Abercorn_influence }
	}
	if = {
		limit = {
			has_country_flag = VIC_Wellington_yes_flag
		}
		add_to_variable = { VIC_proposal_yes = VIC_Wellington_influence }
	}
	if = {
		limit = {
			has_country_flag = VIC_Windermere_yes_flag
		}
		add_to_variable = { VIC_proposal_yes = VIC_Windermere_influence }

		add_to_variable = { VIC_proposal_yes = VIC_Caster_influence }

	}
}

VIC_civil_war_recruit_divisions = {
	set_state_flag = VIC_civil_war_recruited_state

	set_temp_variable = { spawn_units_num = state_population_k }
	divide_temp_variable = { spawn_units_num = 1000 }
	clamp_temp_variable = {
		var = spawn_units_num
		min = 1
		max = 50
	}
	if = {
		limit = { is_core_of = ROOT }
		multiply_temp_variable = { spawn_units_num = 2 }
	}
	set_temp_variable = { war_preparations_bonus = 100 }
	add_to_temp_variable = { war_preparations_bonus = VIC_war_preparations }
	divide_temp_variable = { war_preparations_bonus = 100 }
	multiply_temp_variable = { spawn_units_num = war_preparations_bonus }
	round_temp_variable = spawn_units_num

	if = {
		limit = {
			check_variable = {
				var = spawn_units_num
				value = 1
				compare = greater_than_or_equals
			}
		}
		ROOT = {
			if = {
				limit = {
					NOT = { has_template = "民兵师" }
				}
				division_template = {
					name = "民兵师"
					is_locked = yes
					division_names_group = VIC_INF_02
					regiments = {
						militia = { x = 0 y = 0 }
						militia = { x = 0 y = 1 }
						militia = { x = 1 y = 0 }
						militia = { x = 1 y = 1 }
					}
				}
			}
		}
		for_loop_effect = {
			start = 1
			end = spawn_units_num
			compare = less_than_or_equals

			create_unit = {
				division = "name = \"民兵师\" division_template = \"民兵师\" start_experience_factor = 0 start_equipment_factor = 0.7"
				owner = ROOT
			}
		}
	}
}

TAR_independence_war_win = {
	white_peace = TAR
	set_country_flag = VIC_peace_with_tara
	every_owned_state = {
		limit = {
			has_state_flag = Tara_state_flag
		}
		transfer_state_to = TAR
		add_core_of = TAR
		remove_core_of = PREV
	}
	every_subject_country = {
		white_peace = TAR
		set_country_flag = VIC_peace_with_tara
		every_owned_state = {
			limit = {
				has_state_flag = Tara_state_flag
			}
			transfer_state_to = TAR
			add_core_of = TAR
			remove_core_of = PREV
		}
	}

	hidden_effect = {
		set_global_flag = show_peace_popup_alert

		TAR = {
			save_global_event_target_as = winning_country
		}
		ROOT = {
			save_global_event_target_as = losing_country
		}

		if = {
			limit = {
				event_target:losing_country = {
					any_allied_country = {
						has_war_together_with = event_target:losing_country
					}
				}
			}
			set_global_flag = losing_country_multiple
		}

		set_global_flag = war_won
	}
}
