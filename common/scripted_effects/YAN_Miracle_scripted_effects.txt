#    YAN_mir_s<n>_<key>            州 id    该工程第 n 处台账（已落成）
#    YAN_mir_done_<key> / YAN_mir_total     该工程处数 / 全国总处数（上限 19）

# ---------------- 开 / 关 / 切视图 ----------------
YAN_Mir_Open = {
	if = {
		limit = { has_country_flag = yan_mir_open }
		YAN_Mir_Close = yes
	}
	else = {
		set_country_flag = yan_mir_open
		if = {
			limit = { NOT = { check_variable = { YAN_mir_proj > 0 } } }
			set_variable = { YAN_mir_proj = 1 }
		}
		if = {
			limit = { NOT = { any_controlled_state = { has_state_flag = yan_mir_cursor } } }
			YAN_Mir_PickFirst = yes
		}
		YAN_Mir_Recalc = yes
		YAN_Mir_SyncSite = yes
	}
}

YAN_Mir_Close = {
	clr_country_flag = yan_mir_open
}

YAN_Mir_ShowRoster = {
	set_variable = { YAN_mir_view = 1 }
}

		add_to_variable = { ROOT.YAN_mir_site_count = 1 }
		# 还没见到游标 → 这一处排在游标前面，排位 +1

			limit = { check_variable = { ROOT.YAN_mir_seen = 0 } }

			limit = { has_state_flag = yan_mir_cursor }

# 两趟法：第一趟给合格州编序号并记下游标的序号，第二趟按目标序号落旗。
# 全程只比整数序号，不碰州 id，所以不依赖任何没把握的语法。
YAN_Mir_Cycle = {
	every_state = {
		limit = { has_state_flag = yan_mir_cursor }
		clr_state_flag = yan_mir_cursor
		set_state_flag = yan_mir_old_cursor
	}
	set_variable = { YAN_mir_ord = 0 }
	set_variable = { YAN_mir_cur_ord = 0 }
	every_controlled_state = {
		limit = { YAN_Mir_State_Eligible = yes }
		add_to_variable = { ROOT.YAN_mir_ord = 1 }
		if = {
			limit = { has_state_flag = yan_mir_old_cursor }
			set_variable = { ROOT.YAN_mir_cur_ord = ROOT.YAN_mir_ord }
		}
	}
	set_variable = { YAN_mir_target_ord = 0 }
	if = {

				limit = { check_variable = { YAN_mir_dir < 0 } }

					limit = { NOT = { check_variable = { YAN_mir_target_ord > 0 } } }

			limit = {
				check_variable = { ROOT.YAN_mir_placed = 0 }

			}

			set_variable = { ROOT.YAN_mir_placed = 1 }
		}
	}
	every_state = {
		limit = { has_state_flag = yan_mir_old_cursor }
		clr_state_flag = yan_mir_old_cursor
	}

YAN_Mir_Finish_guangsha = {
	clr_country_flag = yan_mir_act_guangsha

		add_building_construction = { type = industrial_complex level = 2 instant_build = yes }
		add_building_construction = { type = infrastructure level = 1 instant_build = yes }
		add_dynamic_modifier = { modifier = yan_mir_dm_guangsha }
		clr_state_flag = yan_mir_building_guangsha
		set_state_flag = yan_mir_site_guangsha
		set_variable = { ROOT.YAN_mir_built = 1 }
		set_variable = { ROOT.YAN_mir_slot_taken = 0 }
		if = {
			limit = {
				check_variable = { ROOT.YAN_mir_slot_taken = 0 }
				NOT = { check_variable = { ROOT.YAN_mir_s1_guangsha > 0 } }
			}
			set_variable = { ROOT.YAN_mir_s1_guangsha = THIS.id }
			set_variable = { ROOT.YAN_mir_slot_taken = 1 }
		}
		if = {
			limit = {
				check_variable = { ROOT.YAN_mir_slot_taken = 0 }
				NOT = { check_variable = { ROOT.YAN_mir_s2_guangsha > 0 } }
			}
			set_variable = { ROOT.YAN_mir_s2_guangsha = THIS.id }
			set_variable = { ROOT.YAN_mir_slot_taken = 1 }
		}

			limit = {

			set_variable = { ROOT.YAN_mir_slot_taken = 1 }
		}
	}

		add_dynamic_modifier = { modifier = yan_miracle_spirit }
	}
	YAN_Mir_Recalc = yes
	YAN_Mir_SyncSite = yes
}
