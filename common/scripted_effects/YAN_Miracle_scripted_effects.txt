# ---------------- 月度脉冲 ----------------
YAN_Mir_Monthly = {
	# 周期 / 季节
	add_to_variable = { YAN_mir_cycle = 1 }
	add_to_variable = { YAN_mir_season = 1 }
	if = {
		limit = { check_variable = { YAN_mir_season > 3 } }
		set_variable = { YAN_mir_season = 0 }
	}

	# 天气（春夏秋冬四套权重）
	if = {
		limit = { check_variable = { YAN_mir_season = 0 } }
		random_list = {
			45 = { set_variable = { YAN_mir_weather = 0 } }
			25 = { set_variable = { YAN_mir_weather = 1 } }
			20 = { set_variable = { YAN_mir_weather = 2 } }
			10 = { set_variable = { YAN_mir_weather = 4 } }
		}
	}
	if = {
		limit = { check_variable = { YAN_mir_season = 1 } }
		random_list = {
			50 = { set_variable = { YAN_mir_weather = 0 } }
			30 = { set_variable = { YAN_mir_weather = 1 } }
			13 = { set_variable = { YAN_mir_weather = 2 } }
			7 = { set_variable = { YAN_mir_weather = 3 } }
		}
	}
	if = {
		limit = { check_variable = { YAN_mir_season = 2 } }
		random_list = {
			55 = { set_variable = { YAN_mir_weather = 0 } }
			30 = { set_variable = { YAN_mir_weather = 1 } }
			10 = { set_variable = { YAN_mir_weather = 2 } }
			5 = { set_variable = { YAN_mir_weather = 3 } }
		}
	}
	if = {
		limit = { check_variable = { YAN_mir_season = 3 } }
		random_list = {
			25 = { set_variable = { YAN_mir_weather = 0 } }
			20 = { set_variable = { YAN_mir_weather = 1 } }
			30 = { set_variable = { YAN_mir_weather = 2 } }
			25 = { set_variable = { YAN_mir_weather = 3 } }
		}
	}

	# 速度：基础 + 稳定度 + 季节 + 天气 + 诏令 + 临时加成
	set_variable = { YAN_mir_speed = 1 }
	set_variable = { var = YAN_mir_tmp value = stability }
	multiply_variable = { YAN_mir_tmp = 0.3 }
	subtract_from_variable = { YAN_mir_tmp = 0.15 }
	add_to_variable = { YAN_mir_speed = YAN_mir_tmp }

	if = { limit = { check_variable = { YAN_mir_season = 0 } } add_to_variable = { YAN_mir_speed = 0.05 } }
	if = { limit = { check_variable = { YAN_mir_season = 2 } } add_to_variable = { YAN_mir_speed = 0.12 } }
	if = { limit = { check_variable = { YAN_mir_season = 3 } } subtract_from_variable = { YAN_mir_speed = 0.08 } }

	if = { limit = { check_variable = { YAN_mir_weather = 0 } } add_to_variable = { YAN_mir_speed = 0.10 } }
	if = {
		limit = { check_variable = { YAN_mir_weather = 2 } }
		if = { limit = { has_country_flag = yan_mir_edict_jt } subtract_from_variable = { YAN_mir_speed = 0.075 } }
		else = { subtract_from_variable = { YAN_mir_speed = 0.15 } }
	}
	if = {
		limit = { check_variable = { YAN_mir_weather = 3 } }
		if = { limit = { has_country_flag = yan_mir_edict_jt } subtract_from_variable = { YAN_mir_speed = 0.20 } }
		else = { subtract_from_variable = { YAN_mir_speed = 0.40 } }
	}

	if = { limit = { has_country_flag = yan_mir_edict_dx } add_to_variable = { YAN_mir_speed = 0.25 } }
	if = { limit = { has_country_flag = yan_mir_edict_qt } add_to_variable = { YAN_mir_speed = 0.12 } }
	if = { limit = { has_country_flag = yan_mir_edict_qj } add_to_variable = { YAN_mir_speed = 0.30 } }
	add_to_variable = { YAN_mir_speed = YAN_mir_bonus_speed }

	if = { limit = { check_variable = { YAN_mir_speed < 0.3 } } set_variable = { YAN_mir_speed = 0.3 } }
	if = { limit = { check_variable = { YAN_mir_speed > 2.2 } } set_variable = { YAN_mir_speed = 2.2 } }

	# 临时加成倒计时
	if = {
		limit = { check_variable = { YAN_mir_bonus_timer > 0 } }
		subtract_from_variable = { YAN_mir_bonus_timer = 1 }
	}
	else = {
		set_variable = { YAN_mir_bonus_speed = 0 }
	}
	if = {
		limit = { check_variable = { YAN_mir_rush_cd > 0 } }
		subtract_from_variable = { YAN_mir_rush_cd = 1 }
	}

	# 作坊产出

	set_variable = { YAN_mir_tmp = YAN_mir_blvl_tax }
	multiply_variable = { YAN_mir_tmp = 8 }
	add_political_power = YAN_mir_tmp
	set_variable = { YAN_mir_tmp = YAN_mir_blvl_lumber }
	multiply_variable = { YAN_mir_tmp = 40 }
	add_equipment_to_stockpile = { type = support_equipment amount = YAN_mir_tmp }
	set_variable = { YAN_mir_tmp = YAN_mir_blvl_quarry }
	multiply_variable = { YAN_mir_tmp = 80 }
	add_equipment_to_stockpile = { type = infantry_equipment amount = YAN_mir_tmp }
	set_variable = { YAN_mir_tmp = YAN_mir_blvl_foundry }
	multiply_variable = { YAN_mir_tmp = 30 }
	add_equipment_to_stockpile = { type = artillery_equipment amount = YAN_mir_tmp }
	set_variable = { YAN_mir_tmp = YAN_mir_blvl_mine }
	multiply_variable = { YAN_mir_tmp = 15 }
	add_fuel = YAN_mir_tmp
	set_variable = { YAN_mir_tmp = YAN_mir_blvl_camp }
	multiply_variable = { YAN_mir_tmp = 2000 }
	add_manpower = YAN_mir_tmp

	# 诏令月度维持
	if = { limit = { has_country_flag = yan_mir_edict_dx } add_stability = -0.005 }
	if = { limit = { has_country_flag = yan_mir_edict_qt } add_command_power = -5 }
	if = { limit = { has_country_flag = yan_mir_edict_wb } add_political_power = -10 }
	if = { limit = { has_country_flag = yan_mir_edict_qj } add_political_power = -15 }
	if = { limit = { has_country_flag = yan_mir_edict_jt } add_command_power = -3 }

	# 工程月度推进

	if = {
		limit = {
			check_variable = { YAN_mir_stat_wall = 1 }
			check_variable = { YAN_mir_work_wall > 0 }
			has_political_power > 24
			has_equipment = { infantry_equipment > 299 }
			has_equipment = { support_equipment > 149 }
		}
		add_political_power = -25
		add_fuel = -40
		add_equipment_to_stockpile = { type = infantry_equipment amount = -300 }
		add_equipment_to_stockpile = { type = support_equipment amount = -150 }
		clr_country_flag = yan_mir_short_wall
		set_variable = { YAN_mir_tmp = YAN_mir_work_wall }
		divide_variable = { YAN_mir_tmp = 15 }
		multiply_variable = { YAN_mir_tmp = 0.7 }
		add_to_variable = { YAN_mir_tmp = 0.3 }
		multiply_variable = { YAN_mir_tmp = 6 }
		multiply_variable = { YAN_mir_tmp = YAN_mir_speed }
		multiply_variable = { YAN_mir_tmp = 1.12 }
		if = {
			limit = {
				OR = {
					check_variable = { YAN_mir_season = 3 }
					check_variable = { YAN_mir_weather = 3 }
				}
			}
			multiply_variable = { YAN_mir_tmp = 1.15 }
		}
		add_to_variable = { YAN_mir_prog_wall = YAN_mir_tmp }
	}
	else_if = {
		limit = {
			check_variable = { YAN_mir_stat_wall = 1 }
			check_variable = { YAN_mir_work_wall > 0 }
		}
		set_country_flag = yan_mir_short_wall
	}
	if = { limit = { check_variable = { YAN_mir_prog_wall > 24 } } set_variable = { YAN_mir_stage_wall = 1 } }
	if = { limit = { check_variable = { YAN_mir_prog_wall > 49 } } set_variable = { YAN_mir_stage_wall = 2 } }
	if = { limit = { check_variable = { YAN_mir_prog_wall > 74 } } set_variable = { YAN_mir_stage_wall = 3 } }

	if = {
		limit = {
			check_variable = { YAN_mir_stat_sisui = 1 }
			check_variable = { YAN_mir_work_sisui > 0 }
			has_political_power > 19
			has_equipment = { artillery_equipment > 99 }
		}
		add_political_power = -20
		add_fuel = -60
		add_equipment_to_stockpile = { type = artillery_equipment amount = -100 }
