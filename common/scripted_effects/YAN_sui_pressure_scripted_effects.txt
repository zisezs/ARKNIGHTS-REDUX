#  · 岁醒树删除后不再由 YAN_init_mechanics / YAN_monthly_tick 调用；
#    山海众叛乱仍会写入 yan_sui_pressure，界园 / 岁陵司决议仍读取该变量。
#  · YAN_sui_pressure_init   开局初值 + 挂精神（重制时接回 YAN_init_mechanics）
#  · YAN_sui_pressure_recalc 钳制两变量，并把岁压摊到 7 个修正键上
#  · YAN_sui_pressure_monthly 月度反扑：未立约前岁压自己涨回来（由 YAN_monthly_tick 调用）
#  · YAN_sui_pressure_agree   终格「岁陵之约」：清零、撤精神、收束叛军（重制时再接回新树）
#  · 生成器 E:/tmp/sui12/gen_suixing.js

YAN_sui_pressure_init = {
	if = {
		limit = { NOT = { has_variable = yan_sui_pressure } }
		set_variable = { yan_sui_pressure = 100 }
	}
	YAN_sui_pressure_recalc = yes
	add_dynamic_modifier = { modifier = YAN_sui_pressure }
}

YAN_sui_pressure_recalc = {
	clamp_variable = { var = yan_sui_pressure min = 0 max = 100 }
	clamp_variable = { var = yan_sui_awakening min = 0 max = 100 }
	# industrial_capacity_factory = -100% × 岁压
	set_variable = { YAN_sui_p_fac_var = { value = yan_sui_pressure multiply = -0.01 } }
	# industrial_capacity_dockyard = -100% × 岁压
	set_variable = { YAN_sui_p_dock_var = { value = yan_sui_pressure multiply = -0.01 } }
	# production_speed_buildings_factor = -60% × 岁压
	set_variable = { YAN_sui_p_build_var = { value = yan_sui_pressure multiply = -0.006 } }
	# research_speed_factor = -40% × 岁压
	set_variable = { YAN_sui_p_res_var = { value = yan_sui_pressure multiply = -0.004 } }
	# army_org_factor = -30% × 岁压
	set_variable = { YAN_sui_p_org_var = { value = yan_sui_pressure multiply = -0.003 } }
	# supply_consumption_factor = 50% × 岁压
	set_variable = { YAN_sui_p_sup_var = { value = yan_sui_pressure multiply = 0.005 } }
	# stability_factor = -20% × 岁压
	set_variable = { YAN_sui_p_stab_var = { value = yan_sui_pressure multiply = -0.002 } }
}

YAN_sui_pressure_monthly = {
	if = {
		limit = {
			NOT = { has_country_flag = yan_sui_agreement_done }
			check_variable = { yan_sui_pressure > 0 }
		}
		if = {
			limit = { check_variable = { yan_sui_awakening > 79 } }
			add_to_variable = { yan_sui_pressure = 3 }
		}
		else = {
			limit = { check_variable = { yan_sui_awakening > 49 } }
			add_to_variable = { yan_sui_pressure = 1.5 }
		}
		YAN_sui_pressure_recalc = yes
	}
}

YAN_sui_pressure_agree = {
	set_variable = { yan_sui_pressure = 0 }
	YAN_sui_pressure_recalc = yes
	remove_dynamic_modifier = { modifier = YAN_sui_pressure }
	# 祂退回陵里，但没有睡死：苏醒度冻结在警戒下沿，不再自己涨
	set_variable = { yan_sui_awakening = 45 }
	add_stability = 0.10
	add_war_support = 0.10
	add_political_power = 150
	# 2026-10-25：开局精神已合并为【旧时代的大炎帝国】，不再单独移除；
	# 月度苏醒增长改由 yan_sui_agreement_done 旗标冻结（见 YAN_scripted_effects.txt）。
	if = {
		limit = { has_idea = YAN_Sui_disaster }
		remove_ideas = YAN_Sui_disaster
	}
	if = {
		limit = { has_idea = YAN_Behemoths_awaken }
		remove_ideas = YAN_Behemoths_awaken
	}
	# 叛军收束：山海众交还占领、遣散乱兵、停战
	# 见 common/scripted_effects/YAN_shanhai_scripted_effects.txt（生成器 gen_shanhai.js）
	YAN_shanhai_disband = yes
	set_country_flag = yan_sui_agreement_done
	# 主树 YAN_national_focus 已删除；重制时在此 load_focus_tree 切回新树。
}
