YAN_auto_ensure_armor = {
	if = { limit = { NOT = { has_variable = YAN_auto_armor_atk } } set_variable = { YAN_auto_armor_atk = 0 } }
	if = { limit = { NOT = { has_variable = YAN_auto_armor_def } } set_variable = { YAN_auto_armor_def = 0 } }
	if = { limit = { NOT = { has_variable = YAN_auto_armor_hp } } set_variable = { YAN_auto_armor_hp = 0 } }
	if = { limit = { NOT = { has_variable = YAN_auto_armor_thick } } set_variable = { YAN_auto_armor_thick = 0 } }
	if = { limit = { NOT = { has_variable = YAN_auto_armor_cost } } set_variable = { YAN_auto_armor_cost = 0 } }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = YAN_auto_spirit_armor } } }
		add_dynamic_modifier = { modifier = YAN_auto_spirit_armor }
	}
}

# 绿色大炎 · 每 30 天随机地块 buff
# ============================================================

YAN_auto_green_monthly = {
	if = {
		limit = {
			has_country_flag = YAN_auto_green_land
			has_dynamic_modifier = { modifier = YAN_auto_spirit_green }
		}
		random_owned_state = {
			limit = { NOT = { has_dynamic_modifier = { modifier = YAN_auto_green_state_buff } } }
			add_dynamic_modifier = { modifier = YAN_auto_green_state_buff }
		}
	}
}
