# ============================================================================
# 莱塔尼亚 ·「金律乐章 · 调律台」—— 和谐方程与结算
#
# 玩法文档：docs/莱塔尼亚设定集/09_小游戏设计_金律调律台.md
#
# ⚠️ 铁律：HOI4 不支持 $ID$ 参数化 scripted_effect（那是 EU4 的语法）。
#    带编号的效果一律展开成独立 effect：LTN_law_calc_v1..9、tax/levy/arts/tone_1..9、
#    flip_pleb..band。新增同类效果请照此办理，否则点击会静默失效。
#
# 变量表（全部是国家变量；GUI / 决议作用域里读写一律带 ROOT. 前缀）：
#   LTN_law_version      当前乐章版本号（第 1 版 = 建国）
#   LTN_law_cooldown     余波剩余天数（on_monthly_LTN 每月 -30）
#   LTN_law_view         面板页：0 调律台 / 1 条款卡 / 2 乐章史
#   LTN_law_voice_N_tax/_levy/_arts   第 N 声部三音孔（0..4）
#   LTN_law_voice_N_tone              调性（0 宽容 / 1 严苛）
#   LTN_law_voice_N_mood              民心（0..5，由音孔与调性推导）
#   LTN_law_voice_N_lamp              声部灯帧（0 金 / 1 暗金 / 2 红）
#   LTN_law_source/_dev/_taxhigh      民心合计 / 宽容声部数 / 高赋税声部数
#   LTN_law_tier                      档位（-2..+2）
#   LTN_law_harmony/_risk/_res_count/_conf_count
#   LTN_law_hist_last_harmony/_source/_dev   上一版颁布时的读数（乐章史页回看）
#   LTN_law_preview_*                 预演快照（供 [?var] 显示差值）
#   LTN_law_c_<id>                    条款卡启用旗标（切换只改旗标；
#                                     实质收益在「奏响」时由 LTN_law_clause_settle 兑现，
#                                     每张卡带 _done 旗标，一辈子只结算一次）
#
# 方程（"规则可查，算式不可见" —— 面板只给三档预测，不给公式）：
#   压力  p = tax + levy + arts                       （0..12）
#   民心  m = 档(p) + (宽容 ? +1 : -1)，钳 0..5       （档见 LTN_law_calc_v1..v9）
#   得分  s = Σm - 宽容声部数
#   档位  s>35→+2 / >29→+1 / >23→0 / >17→-1 / else→-2
#   和谐  = 档位 + 共鸣数 - 冲突数，钳 ±2
#   风险  = (版本号-1) + (和谐<0 ? +2) + (和谐<-1 ? +2)，钳 0..10
#
# 共鸣：平民入塔学堂 + 抑制剂官费 / 金律法卫扩编 + 军乐制式
# 冲突：女皇之声扩权 且 宽容声部>3（诸侯抵制）/ 金律法卫扩编 且 高赋税声部>3（地方抗税）
# ============================================================================

LTN_law_calc_v1 = {
	# 压力
	set_variable = { LTN_law_tmp_p = LTN_law_voice_1_tax }
	add_to_variable = { LTN_law_tmp_p = LTN_law_voice_1_levy }
	add_to_variable = { LTN_law_tmp_p = LTN_law_voice_1_arts }

	# 压力越小，民心越高
	if = { limit = { check_variable = { LTN_law_tmp_p < 3 } } set_variable = { LTN_law_tmp_m = 5 } }
	else_if = { limit = { check_variable = { LTN_law_tmp_p < 5 } } set_variable = { LTN_law_tmp_m = 4 } }
	else_if = { limit = { check_variable = { LTN_law_tmp_p < 7 } } set_variable = { LTN_law_tmp_m = 3 } }
	else_if = { limit = { check_variable = { LTN_law_tmp_p < 9 } } set_variable = { LTN_law_tmp_m = 2 } }
	else_if = { limit = { check_variable = { LTN_law_tmp_p < 11 } } set_variable = { LTN_law_tmp_m = 1 } }
	else = { set_variable = { LTN_law_tmp_m = 0 } }

	# 调性：宽容 +1（但乐章有偏差），严苛 -1
	if = { limit = { check_variable = { LTN_law_voice_1_tone = 0 } } add_to_variable = { LTN_law_tmp_m = 1 } }
	else = { add_to_variable = { LTN_law_tmp_m = -1 } }

	if = { limit = { check_variable = { LTN_law_tmp_m < 0 } } set_variable = { LTN_law_tmp_m = 0 } }
	if = { limit = { check_variable = { LTN_law_tmp_m > 5 } } set_variable = { LTN_law_tmp_m = 5 } }
	set_variable = { LTN_law_voice_1_mood = LTN_law_tmp_m }

	# 汇总
	add_to_variable = { LTN_law_source = LTN_law_voice_1_mood }
	if = { limit = { check_variable = { LTN_law_voice_1_tone = 0 } } add_to_variable = { LTN_law_dev = 1 } }
	if = { limit = { check_variable = { LTN_law_voice_1_tax > 2 } } add_to_variable = { LTN_law_taxhigh = 1 } }

	# 征收音孔（4 档拉满 / ≤1 档压低）：直接落到「征收音孔」精神
	if = { limit = { check_variable = { LTN_law_voice_1_tax > 3 } } add_to_variable = { LTN_dial_build = 0.009 } }
	else_if = { limit = { check_variable = { LTN_law_voice_1_tax < 2 } } subtract_from_variable = { LTN_dial_build = 0.011 } }
	if = { limit = { check_variable = { LTN_law_voice_1_levy > 3 } } add_to_variable = { LTN_dial_pop = 0.009 } }
	else_if = { limit = { check_variable = { LTN_law_voice_1_levy < 2 } } subtract_from_variable = { LTN_dial_pop = 0.011 } }
	if = { limit = { check_variable = { LTN_law_voice_1_arts > 3 } } add_to_variable = { LTN_dial_res = 0.009 } }
	else_if = { limit = { check_variable = { LTN_law_voice_1_arts < 2 } } subtract_from_variable = { LTN_dial_res = 0.011 } }
	# 声部灯：民心 >3 金 / >1 暗金 / 其余红
	if = { limit = { check_variable = { LTN_law_voice_1_mood > 3 } } set_variable = { LTN_law_voice_1_lamp = 0 } }
	else_if = { limit = { check_variable = { LTN_law_voice_1_mood > 1 } } set_variable = { LTN_law_voice_1_lamp = 1 } }
	else = { set_variable = { LTN_law_voice_1_lamp = 2 } }
}

		set_variable = { LTN_har_pp = 0 }
		set_variable = { LTN_har_res = 0 }

	}

		set_variable = { LTN_har_res = -0.10 }

	}

LTN_law_effect_recalc = {
	# 变量缺失兜底：老存档 / 变量被清时，保证三页互斥不失控
	if = { limit = { NOT = { has_variable = LTN_law_view } } set_variable = { LTN_law_view = 0 } }
	set_variable = { LTN_law_source = 0 }
	set_variable = { LTN_law_dev = 0 }
	set_variable = { LTN_law_taxhigh = 0 }
	# 征收音孔聚合：先清零，再由九个单声部块按档位累加（幂等）
	set_variable = { LTN_dial_pop = 0 }
	set_variable = { LTN_dial_build = 0 }
	set_variable = { LTN_dial_res = 0 }

	LTN_law_calc_v1 = yes
	LTN_law_calc_v2 = yes
	LTN_law_calc_v3 = yes
	LTN_law_calc_v4 = yes
	LTN_law_calc_v5 = yes
	LTN_law_calc_v6 = yes
	LTN_law_calc_v7 = yes
	LTN_law_calc_v8 = yes
	LTN_law_calc_v9 = yes

	# 档位
	set_variable = { LTN_law_score = LTN_law_source }
	subtract_from_variable = { LTN_law_score = LTN_law_dev }
	if = { limit = { check_variable = { LTN_law_score > 35 } } set_variable = { LTN_law_tier = 2 } }
	else_if = { limit = { check_variable = { LTN_law_score > 29 } } set_variable = { LTN_law_tier = 1 } }
	else_if = { limit = { check_variable = { LTN_law_score > 23 } } set_variable = { LTN_law_tier = 0 } }
	else_if = { limit = { check_variable = { LTN_law_score > 17 } } set_variable = { LTN_law_tier = -1 } }
	else = { set_variable = { LTN_law_tier = -2 } }

	# 共鸣
	set_variable = { LTN_law_res_count = 0 }
	if = {
		limit = {
			has_country_flag = LTN_law_c_pleb
			has_country_flag = LTN_law_c_inhib
		}
		add_to_variable = { LTN_law_res_count = 1 }
	}
	if = {
		limit = {
			has_country_flag = LTN_law_c_guard
			has_country_flag = LTN_law_c_band
		}
		add_to_variable = { LTN_law_res_count = 1 }
	}

	# 冲突
	set_variable = { LTN_law_conf_count = 0 }
	if = {
		limit = {
			has_country_flag = LTN_law_c_herald
			check_variable = { LTN_law_dev > 3 }
		}
		add_to_variable = { LTN_law_conf_count = 1 }
	}
	if = {
		limit = {
			has_country_flag = LTN_law_c_guard
			check_variable = { LTN_law_taxhigh > 3 }
		}
		add_to_variable = { LTN_law_conf_count = 1 }
	}

	# 和谐度
	set_variable = { LTN_law_harmony = LTN_law_tier }
	add_to_variable = { LTN_law_harmony = LTN_law_res_count }
	subtract_from_variable = { LTN_law_harmony = LTN_law_conf_count }
	if = { limit = { check_variable = { LTN_law_harmony < -2 } } set_variable = { LTN_law_harmony = -2 } }
	if = { limit = { check_variable = { LTN_law_harmony > 2 } } set_variable = { LTN_law_harmony = 2 } }

	# 失范风险
	set_variable = { LTN_law_risk = LTN_law_version }
	subtract_from_variable = { LTN_law_risk = 1 }
	if = { limit = { check_variable = { LTN_law_harmony < 0 } } add_to_variable = { LTN_law_risk = 2 } }
	if = { limit = { check_variable = { LTN_law_harmony < -1 } } add_to_variable = { LTN_law_risk = 2 } }
	if = { limit = { check_variable = { LTN_law_risk < 0 } } set_variable = { LTN_law_risk = 0 } }
	if = { limit = { check_variable = { LTN_law_risk > 10 } } set_variable = { LTN_law_risk = 10 } }

	LTN_law_apply_spirit = yes

	# 面板显示用：征收合计换算成整数百分数（×100 后取整，供 loc 的 [?var|0]）
	set_variable = { LTN_dial_pop_i = LTN_dial_pop }
	multiply_variable = { LTN_dial_pop_i = 100 }
	round_variable = LTN_dial_pop_i
	set_variable = { LTN_dial_build_i = LTN_dial_build }
	multiply_variable = { LTN_dial_build_i = 100 }
	round_variable = LTN_dial_build_i
	set_variable = { LTN_dial_res_i = LTN_dial_res }
	multiply_variable = { LTN_dial_res_i = 100 }
	round_variable = LTN_dial_res_i
}

# ---------------------------------------------------------------------------
LTN_law_effect_promulgate = {
	LTN_law_effect_recalc = yes

	# 乐章史：把「上一版」的读数留档，供面板第 3 页回看
	set_variable = { LTN_law_hist_last_harmony = LTN_law_harmony }
	set_variable = { LTN_law_hist_last_source = LTN_law_source }
	set_variable = { LTN_law_hist_last_dev = LTN_law_dev }

	LTN_law_clause_settle = yes

	add_to_variable = { LTN_law_version = 1 }
	set_variable = { LTN_law_cooldown = 90 }
	add_political_power = -50

	# 颁布当日的礼炮与警报：大同加赏，失范重罚
	if = { limit = { check_variable = { LTN_law_harmony > 1 } }
		add_stability = 0.03
		add_political_power = 25
	}
	else_if = { limit = { check_variable = { LTN_law_harmony > 0 } } add_stability = 0.02 }
	else_if = { limit = { check_variable = { LTN_law_harmony < -1 } } add_stability = -0.05 }
	else_if = { limit = { check_variable = { LTN_law_harmony < 0 } } add_stability = -0.03 }

	clr_country_flag = LTN_law_open
	country_event = { id = LTN_law.1 days = 1 }
}


# ---------------------------------------------------------------------------
# 每月脉冲：余波倒计时 + 重算 + 失范压顶
# ---------------------------------------------------------------------------
LTN_law_effect_tick = {
	if = { limit = { has_variable = LTN_law_cooldown }
		if = { limit = { check_variable = { LTN_law_cooldown > 0 } } subtract_from_variable = { LTN_law_cooldown = 30 } }
		if = { limit = { check_variable = { LTN_law_cooldown < 0 } } set_variable = { LTN_law_cooldown = 0 } }
	}

	LTN_law_effect_recalc = yes

	# 失范压顶：风险 >=8 时，每年有概率直接触发分声部事件链
	if = {
		limit = {
			check_variable = { LTN_law_risk > 7 }
			check_variable = { LTN_law_harmony < 0 }
			random = { chance = 25 }
		}
		country_event = { id = LTN_law.2 days = 1 }
	}
}
