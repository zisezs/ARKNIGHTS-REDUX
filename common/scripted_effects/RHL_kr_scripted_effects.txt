# 【由 docs/gen_rhl_kristen.cjs 生成，勿手改】民族精神变量建表 + 出任总辖的共用效果
# 动态精神的数值全部写成变量；没建表就做 add_to_variable 会被引擎静默失效，
# 所以起始格（RHL_kr_plan_dead）第一件事就是调本效果，之后每一格只做加法。
RHL_kr_init = {
	set_variable = { RHL_kr_research = 0 }
	set_variable = { RHL_kr_pp = 0 }
	set_variable = { RHL_kr_stab = 0 }
	set_variable = { RHL_kr_ws = 0 }
	set_variable = { RHL_kr_arms = 0 }
	set_variable = { RHL_kr_ind = 0 }
	set_variable = { RHL_kr_air = 0 }
	set_variable = { RHL_kr_fuel = 0 }
}

# ---- RHL_lone_star：出任总辖的完整流程（国策【推举她为总辖】与决议共用一份）----
# 顺序不能改：先转籍 → 招募 → 补领袖角色（promote_leader 才真的上位）→ 挂特质 → 建表挂累计精神。
# · **不动执政党**：RHL 的 ruling_party = despotism，现任总辖塞雷娅挂的就是
#   civilian_dictatorship_subtype。这里是给她补一个同子型的领袖角色再 promote，
#   等于「换党魁」而不是「换党」。写过 set_politics = { ruling_party = liberalism } 的那版
#   会把执政党整个换掉，台上就变成自由主义那个大众脸 NPC（2026-09-21 踩过，玩家原话：
#   「不是换执政党！」）。
# · 转籍必须写在 **CLB 的作用域里**：直接写 CLB_Kristen_Wright = { ... } 是拿当前国家
#   （RHL）去查这个角色，引擎记
#   tried to use character CLB_Kristen_Wright as scope, but could not find this character in country RHL
#   然后 recruit_character 照旧被拒（already assigned to CLB）→ 上位失败。
# · 不用 retire_country_leader —— 本 mod 实证那一步会让领袖位空掉。
# · 累计那半边用 has_dynamic_modifier 兜重入：决议不是 fire_only_once，
#   失败过一次（领袖没换成）可以攒够政治点再点一次，且不会把已经攒下的层数清零。
RHL_kr_make_chancellor = {
	if = {
		limit = {
			country_exists = CLB
			CLB = { has_character = CLB_Kristen_Wright }
		}
		CLB = {
			set_nationality = { target_country = ROOT character = CLB_Kristen_Wright }
		}
	}
	recruit_character = CLB_Kristen_Wright
	add_country_leader_role = {
		character = CLB_Kristen_Wright
		promote_leader = yes
		country_leader = { ideology = civilian_dictatorship_subtype }
	}
	add_country_leader_trait = RHL_lone_star
	set_country_flag = RHL_kr_flag_chancellor
	if = {
		limit = { NOT = { has_dynamic_modifier = RHL_kr_dyn_lone_star } }
		set_variable = { RHL_kr_launch_research = 0 }
		add_dynamic_modifier = { modifier = RHL_kr_dyn_lone_star }
	}
}

# ---- 【孤星】的累计半边：每发射一次火箭 +1% 科研速度 ----
# 领袖特质读不到变量（本 mod 的 SRC_lv_judger 同样绕法），所以数值走动态精神 RHL_kr_dyn_lone_star。
# 由火箭模块的发射结算 RHL_r_launch_resolve 每次调用一遍（见 common/scripted_effects/RHL_rocket_scripted_effects.txt）。
# 只有她在位才记账：换了人就不再往上加，动态精神本身也随决议/国策留在国家身上。
RHL_kr_lone_star_count = {
	if = {
		limit = { has_country_leader_with_trait = RHL_lone_star }
		add_to_variable = { RHL_kr_launch_research = 0.01 }
	}
}
