# ==============================================================================
#  大炎 · SideStory 事件链调度器
#  · 五条链，每条 10 个事件，严格按剧情时间顺序，间隔 60 天（约 2 个月）触发一个。
#      yan_honglu  洪炉示岁  events/YAN_Honglu.txt
#      yan_jjj     将进酒    events/YAN_Jiangjinjiu.txt
#      yan_hzr     画中人    events/YAN_Huazhongren.txt
#      yan_dly     登临意    events/YAN_Denglinyi.txt
#      yan_xjh     相见欢    events/YAN_Xiangjianhuan.txt
#  · 起链：本文件的 YAN_sidestory_start_check，挂在 YAN_monthly_tick 上，
#    每条链一个硬日期门 + 一个 started 旗标，只点一次火。
#  · 推进：不靠玩家点选项，而是写在每个事件的 immediate 里——
#      immediate = { country_event = { id = yan_xxx.N+1 days = 60 } }
#    这样即便上一封事件还挂在屏幕上没批，下一封也照样按期到，节奏不会被拖住。
#    第 10 个事件是收尾，不再往后排。
#  · 错峰 + 季节锚点：三个年关故事（洪炉示岁 / 画中人 / 相见欢）都钉在一月起链，
#    另两条插在中间。稳态下同时最多四条在推进，但 60 天的相位各不相同，
#    落到日历上是每周一封上下，不会同一天糊上来五封。
#    排期（事件 1 → 事件 10，每 60 天一封）：
#      yan_honglu  1095.1.1 → 1096.7
#      yan_jjj     1095.5.1 → 1096.11
#      yan_hzr     1096.1.1 → 1097.7
#      yan_dly     1096.5.1 → 1097.11
#      yan_xjh     1097.1.1 → 1098.7
#    收尾正好压在「西北都护府异变」（1098.1.1 起链）那场大火前后。
# ==============================================================================

YAN_sidestory_start_check = {
	# ---- 洪炉示岁（龙门 · 年关）----
	if = {
		limit = {
			date > 1095.1.1
			NOT = { has_country_flag = yan_honglu_started_flag }
		}
		set_country_flag = yan_honglu_started_flag
		country_event = { id = yan_honglu.1 days = 1 }
	}

	# ---- 将进酒（尚蜀）----
	if = {
		limit = {
			date > 1095.5.1
			NOT = { has_country_flag = yan_jjj_started_flag }
		}
		set_country_flag = yan_jjj_started_flag
		country_event = { id = yan_jjj.1 days = 1 }
	}

	# ---- 画中人（勾吴 · 除夕）----
	if = {
		limit = {
			date > 1096.1.1
			NOT = { has_country_flag = yan_hzr_started_flag }
		}
		set_country_flag = yan_hzr_started_flag
		country_event = { id = yan_hzr.1 days = 1 }
	}

	# ---- 登临意（玉门 / 蜀地）----
	if = {
		limit = {
			date > 1096.5.1
			NOT = { has_country_flag = yan_dly_started_flag }
		}
		set_country_flag = yan_dly_started_flag
		country_event = { id = yan_dly.1 days = 1 }
	}

	# ---- 相见欢（春节）----
	if = {
		limit = {
			date > 1097.1.1
			NOT = { has_country_flag = yan_xjh_started_flag }
		}
		set_country_flag = yan_xjh_started_flag
		country_event = { id = yan_xjh.1 days = 1 }
	}
}
