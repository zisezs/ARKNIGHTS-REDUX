			limit = {
				has_country_flag = YAN_suisui_unlocked
				NOT = { has_country_flag = YAN_suisui_locked }
			}
			# 老存档/AI 变量缺失时补初值
			if = {
				limit = { NOT = { has_variable = YAN_suisui_ai_days } }
				set_variable = { YAN_suisui_ai_days = 0 }
				set_variable = { YAN_suisui_ai_act = 0 }
			}

			# 每年主动使用一次：轮流选司、轮流用四个行动
			add_to_variable = { YAN_suisui_ai_days = 1 }
			if = {
				limit = {
					check_variable = { YAN_suisui_ai_days > 364 }
					check_variable = { YAN_suisui_loyalty > 1 }
				}
				subtract_from_variable = { YAN_suisui_ai_days = 365 }
				add_to_variable = { YAN_suisui_sel = 1 }
				if = {
					limit = { check_variable = { YAN_suisui_sel > 20 } }
					set_variable = { YAN_suisui_sel = 1 }
				}
				add_to_variable = { YAN_suisui_ai_act = 1 }
				if = {
					limit = { check_variable = { YAN_suisui_ai_act > 4 } }
					set_variable = { YAN_suisui_ai_act = 1 }
				}
				if = {
					limit = { check_variable = { YAN_suisui_ai_act = 1 } }
					YAN_suisui_act1 = yes
				}
				else_if = {
					limit = { check_variable = { YAN_suisui_ai_act = 2 } }
					YAN_suisui_act2 = yes
				}
				else_if = {
					limit = { check_variable = { YAN_suisui_ai_act = 3 } }
					YAN_suisui_act3 = yes
				}
				else = {
					YAN_suisui_act4 = yes
				}
			}

			# 1098 年后忠诚强制归零 → 岁怒（YNW 终幕 / yanse.1）
			if = {
				limit = {
					date > 1098.1.1
					check_variable = { YAN_suisui_loyalty > 0 }
				}
				set_variable = { YAN_suisui_loyalty = 0 }
				YAN_suisui_check_wrath = yes
			}
		}
	}
}

			}
			YNW = {
				YNW_upheaval_become_sui_state = yes

	}
