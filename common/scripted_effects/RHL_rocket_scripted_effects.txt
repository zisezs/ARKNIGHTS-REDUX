# ---- RHL_r_launch_resolve：掷点 → 5 档判定 → 结算/回写 → 成本/冷却 → 软着陆 ----
RHL_r_launch_resolve = {
	RHL_r_recompute = yes
	if = {
		limit = { check_variable = { RHL_r_valid = 1 } }
		set_temp_variable = { guaranteed = 0 }
		if = { limit = { check_variable = { RHL_r_guarantee = 1 } } set_temp_variable = { guaranteed = 1 } set_variable = { RHL_r_guarantee = 0 } }
		set_variable = { RHL_r_outcome = 4 }
		if = { limit = { check_variable = { guaranteed = 0 } }
			set_variable_to_random = { var = RHL_r_roll  min = 0  max = 100  integer = yes }
			set_variable = { RHL_r_threshold = { value = RHL_r_reliability  multiply = 100 } }
			if = { limit = { check_variable = { RHL_r_twr < 1 } } set_variable = { RHL_r_outcome = 1 } }
			if = { limit = { AND = { check_variable = { RHL_r_invcone = 1 } check_variable = { RHL_r_n1 > 1 } } } set_variable = { RHL_r_outcome = 1 } }
			if = { limit = { AND = { check_variable = { RHL_r_outcome = 4 } NOT = { check_variable = { RHL_r_threshold > RHL_r_roll } } } }
				set_variable = { RHL_r_fail_stage = 0 }
				set_variable_to_random = { var = RHL_r_fail_stage  min = 1  max = RHL_r_n_stages  integer = yes }
				set_variable = { RHL_r_outcome = 3 }
				if = { limit = { check_variable = { RHL_r_fail_stage = 1 } } set_variable = { RHL_r_outcome = 2 } }
			}
			if = { limit = { check_variable = { RHL_r_outcome = 4 } }
				set_temp_variable = { dv_margin = { value = RHL_r_dv_total  subtract = { value = RHL_r_dv_req  multiply = 1.15 } } }
				set_temp_variable = { perf_roll_gap = { value = { value = RHL_r_reliability  multiply = 50 }  subtract = RHL_r_roll } }
				if = { limit = { AND = { check_variable = { dv_margin > -1 } check_variable = { perf_roll_gap > -1 } } } set_variable = { RHL_r_outcome = 5 } }
			}
		}
		set_variable = { RHL_r_last_outcome = RHL_r_outcome }
		if = { limit = { check_variable = { RHL_r_outcome = 1 } } country_event = { id = rhl_rocket.51 days = 1 } }
		if = { limit = { check_variable = { RHL_r_outcome = 2 } } country_event = { id = rhl_rocket.52 days = 1 } }
		if = { limit = { check_variable = { RHL_r_outcome = 3 } } country_event = { id = rhl_rocket.53 days = 1 } }
		if = { limit = { check_variable = { RHL_r_outcome = 4 } } country_event = { id = rhl_rocket.54 days = 1 } }
		if = { limit = { check_variable = { RHL_r_outcome = 5 } } country_event = { id = rhl_rocket.55 days = 1 } }
		# F6：第一次打上天空 / 第一次摔下来，各有一次大事件（旗标立刻置，防 days=1 期间重复触发）
		if = { limit = { AND = { check_variable = { RHL_r_outcome > 3 } NOT = { has_country_flag = RHL_rocket_first_light } } }
			set_country_flag = RHL_rocket_first_light
			news_event = { id = rhl_rocket.60 days = 1 }
			# 超事件【“孤星”探测火箭发射成功】(rhlse.2)：全球广播，每个国家各收一份。
			# 定义端不写 major = yes（两者叠加会重复弹）；用 hours 不用 days，
			# 免得跨天投递遇上暂停/退出就把整条广播丢了。面板见 docs/wire_rhl_lone_star_se.cjs。
			every_country = { news_event = { id = rhlse.2 hours = 1 } }
		}
		if = { limit = { AND = { check_variable = { RHL_r_outcome < 3 } NOT = { has_country_flag = RHL_rocket_first_fall } } }
			set_country_flag = RHL_rocket_first_fall
			news_event = { id = rhl_rocket.61 days = 1 }
		}
		# F10：【孤星】每发射一次火箭 +1% 科研速度（只有她在位才记账，见 RHL_kr_lone_star_count）
		RHL_kr_lone_star_count = yes
		# 通用发射成本（对 5 档一致）≠ 档位惩罚；若需变更请由主理人裁定。
		if = { limit = { check_variable = { guaranteed = 0 } } add_political_power = -100 }
		add_timed_idea = { idea = RHL_rocket_production_drain  days = 30 }
		# F3：≥ 阈值 + 仍停在上一档，两档各判一次（+2 的跳跃不会再把升档整个跳过）
		if = { limit = { AND = { check_variable = { RHL_rocket_successes > 2 } has_idea = RHL_spirit_rocket_program_1 } } country_event = { id = rhl_rocket.56 days = 1 } }
		if = { limit = { AND = { check_variable = { RHL_rocket_successes > 5 } has_idea = RHL_spirit_rocket_program_2 } } country_event = { id = rhl_rocket.56 days = 1 } }
		# F4：≥ 阈值 + 本轮还没领过保底；先判 5 次那一档，否则 3 次那档会把 5 次挡掉
		if = { limit = { AND = { check_variable = { RHL_rocket_failures > 4 } check_variable = { RHL_r_guarantee = 0 } } } country_event = { id = rhl_rocket.58 days = 1 } }
		else_if = { limit = { AND = { check_variable = { RHL_rocket_failures > 2 } check_variable = { RHL_r_guarantee_reli = 0 } } } country_event = { id = rhl_rocket.57 days = 1 } }
		set_variable = { RHL_r_guarantee_reli = 0 }
		set_variable = { RHL_r_fail_stage = 0 }
		RHL_r_recompute = yes
		add_to_variable = { RHL_r_dirty = 1 }
	}
}
