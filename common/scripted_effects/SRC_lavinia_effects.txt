# ---- 继任：拉维妮娅·法尔科内就任第二任总统，莱昂图索转荣誉顾问 ----
# 顺序要紧：先 set_politics，再带着 promote_leader 补角色（否则升的是旧党的领袖）。
# 换领袖只用 add_country_leader_role + promote_leader = yes，不要再写
# retire_country_leader + promote_character 兜底段（见 [[hoi4-leader-change]]）。
#
# 【审判者】特质 = political_power_gain +100%（写在 common/country_leader/SRC_traits.txt），
# 设计稿还要「再减去当前稳定度」。领袖特质读不到变量，所以扣减那半边挂在国家动态修正
# SRC_lv_dyn_judger_scale 上（political_power_gain = SRC_lv_judge_tax，0 → -1），
# 由 SRC_lv_judger_scale_effect 按 has_stability 每 10% 一档刷新（每月一次，见 on_actions）。
# 反扑期稳定度锁在 0%，所以整条树上她都是净 +100%；终局解锁稳定度后自动往下掉。
SRC_lv_succession_takeover_effect = {
	if = {
		limit = {
			NOT = {
				has_country_flag = SRC_lv_flag_lavinia_role
			}
		}
		set_politics = {
			ruling_party = despotism
			elections_allowed = no
			last_election = "1101.9.1"
			election_frequency = 48
		}
		add_country_leader_role = {
			character = SRC_Lavinia_Falcone
			promote_leader = yes
			country_leader = {
				ideology = civilian_dictatorship_subtype
				traits = {
					SRC_lv_judger
				}
				expire = "1965.1.1.1"
			}
		}
		set_variable = { SRC_lv_judge_tax = 0 }
		add_dynamic_modifier = { modifier = SRC_lv_dyn_judger_scale }
		set_country_flag = SRC_lv_flag_lavinia_role
	}
	SRC_lv_advisor_purge_effect = yes
	activate_advisor = SRC_Leontuzzo_Bellone
	SRC_lv_advisor_purge_effect = yes
}

# ---- 【审判者】的扣减半边：按当前稳定度分档刷新 SRC_lv_judge_tax ----
# 引擎没有"把稳定度读进变量"的可靠先例（原版与本项目都没这么写过），
# 所以用 has_stability 触发符做 10% 一档的阶梯，误差 ≤10%，每月对齐一次。
SRC_lv_judger_scale_effect = {
	if = {
		limit = {
			has_stability < 0.1
		}
		set_variable = { SRC_lv_judge_tax = 0 }
	}
	else_if = {
		limit = {
			has_stability < 0.2
		}
		set_variable = { SRC_lv_judge_tax = -0.1 }
	}
	else_if = {
		limit = {
			has_stability < 0.3
		}
		set_variable = { SRC_lv_judge_tax = -0.2 }
	}
	else_if = {
		limit = {
			has_stability < 0.4
		}
		set_variable = { SRC_lv_judge_tax = -0.3 }
	}
	else_if = {
		limit = {
			has_stability < 0.5
		}
		set_variable = { SRC_lv_judge_tax = -0.4 }
	}
	else_if = {
		limit = {
			has_stability < 0.6
		}
		set_variable = { SRC_lv_judge_tax = -0.5 }
	}
	else_if = {
		limit = {
			has_stability < 0.7
		}
		set_variable = { SRC_lv_judge_tax = -0.6 }
	}
	else_if = {
		limit = {
			has_stability < 0.8
		}
		set_variable = { SRC_lv_judge_tax = -0.7 }
	}
	else_if = {
		limit = {
			has_stability < 0.9
		}
		set_variable = { SRC_lv_judge_tax = -0.8 }
	}
	else_if = {
		limit = {
			always = yes
		}
		set_variable = { SRC_lv_judge_tax = -0.9 }
	}
}

# ---- 清洗带【家族血统】标签的顾问（引擎原生按领袖特质批量摘，不点名）----
# SRC_Family_Lineage 定义在 common/country_leader/SRC_traits.txt，
# 挂在 common/characters/SRC.txt 里各家族顾问的 traits 上（含 army_chief 槽的瓦拉赫）。
# remove_ideas_with_trait 会把带着个标签的在任顾问一律去职；不在任的没有副作用。
# 白名单靠"根本不带这个标签"实现：莱昂图索/拉维妮娅/伊雷妮等是 SRC_Outsider_Lineage，
# 德米特里改挂 SRC_Federal_Service（联邦任职，写死在角色文件里），详见效果一览文档的顾问联动一节。
SRC_lv_advisor_purge_effect = {
	if = {
		limit = {
			has_country_flag = SRC_lv_flag_era
		}
		remove_ideas_with_trait = SRC_Family_Lineage
	}
}

# ---- 反扑期月度结算：稳定度锁 0 之外，每约 90 天流失一座非首都核心州 ----
SRC_lv_counterattack_tick_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = SRC_lv_fury_months
			}
		}
		set_variable = { SRC_lv_fury_months = 0 }
	}
	add_to_variable = { SRC_lv_fury_months = 1 }
	if = {
		limit = {
			SRC_lv_fury_months > 2
		}
		set_variable = { SRC_lv_fury_months = 0 }
		random_owned_state = {
			limit = {
				is_core_of = SRC
				is_capital = no
			}
			remove_core_of = SRC
			add_dynamic_modifier = { modifier = SRC_dyn_family_entrenchment }
			ROOT = {
				set_country_flag = SRC_lv_flag_core_lost
			}
		}
		if = {
			limit = {
				has_country_flag = SRC_lv_flag_core_lost
			}
			clr_country_flag = SRC_lv_flag_core_lost
			country_event = { id = src_lv.2 days = 1 }
		}
	}
	SRC_lv_advisor_purge_effect = yes
}
