# ==============================================================================
#  大炎 · 兵部·职方司 —— 援他国的共用脚本效果
#  调用方：common/decisions/YAN_Zhifang_Decisions.txt（八条「遣义从」决议）
#
#  写法照抄本工程已验证的先例（见 docs 里的 SRC_exp_effects.txt / REU.txt 志愿军格）：
#    · division_template 只能在【受援国作用域】现建，不能预置进 common/units
#    · 先建模板、后 create_unit，顺序反了生不出兵
#    · create_unit 的 owner = PREV ＝ 上一层的国家作用域（这里是受援国，不是大炎）
#    · 落兵点用 random_owned_controlled_state：首都被人占了就换个本州落，
#      两样都没有则这路义从无路可入（引擎什么都不生成，也不报错）
#
#  变量：yan_zf_sent（我方已遣义从路数，国家变量，读数见
#        common/scripted_localisation/YAN_zf_read_scripted_localisation.txt）
# ==============================================================================

# ---- 在受援国作用域执行：落 2 支满编「炎字义从」师 ----
YAN_zf_corps_effect = {
	if = {
		limit = {
			NOT = { has_template = "炎字义从" }
		}
		division_template = {
			name = "炎字义从"
			is_locked = yes
			regiments = {
				infantry = { x = 0 y = 0 }
				infantry = { x = 0 y = 1 }
				infantry = { x = 0 y = 2 }
				infantry = { x = 1 y = 0 }
				infantry = { x = 1 y = 1 }
				infantry = { x = 1 y = 2 }
				infantry = { x = 2 y = 0 }
				infantry = { x = 2 y = 1 }
				infantry = { x = 2 y = 2 }
				artillery_brigade = { x = 3 y = 0 }
				artillery_brigade = { x = 3 y = 1 }
			}
			support = {
				engineer = { x = 0 y = 0 }
				artillery = { x = 0 y = 1 }
				logistics_company = { x = 0 y = 2 }
			}
		}
	}
	random_owned_controlled_state = {
		limit = {
			OR = {
				is_capital = yes
				is_core_of = PREV
			}
		}
		create_unit = {
			division = "name = \"炎字义从\" division_template = \"炎字义从\" start_experience_factor = 0.35 start_equipment_factor = 1 start_manpower_factor = 1"
			owner = PREV
			count = 2
		}
	}
}

# ---- 在受援国作用域执行：收下大炎捐输的军械，我方库房同步扣减 ----
YAN_zf_gift_arms_effect = {
	add_equipment_to_stockpile = {
		type = infantry_equipment
		amount = 4000
		producer = YAN
	}
	add_equipment_to_stockpile = {
		type = support_equipment
		amount = 400
		producer = YAN
	}
	YAN = {
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = -4000
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = -400
		}
	}
}

# ---- 在大炎作用域执行：重算勘战册的两个计数 ----
# yan_zf_belligerents  ＝ 天下正在交战的国家数
# yan_zf_hot_fronts    ＝ 八条方向里此刻真有战事的条数
YAN_zf_survey_effect = {
	set_variable = { yan_zf_belligerents = 0 }
	set_variable = { yan_zf_hot_fronts = 0 }
	every_country = {
		limit = {
			exists = yes
			has_war = yes
		}
		YAN = {
			add_to_variable = { yan_zf_belligerents = 1 }
		}
	}
	every_country = {
		limit = {
			exists = yes
			has_war = yes
			OR = {
				tag = HGS
				tag = DKK
				tag = ANA
				tag = MIN
				tag = KAC
				tag = SRC
				tag = LTN
				tag = KZM
			}
		}
		YAN = {
			add_to_variable = { yan_zf_hot_fronts = 1 }
		}
	}
}
