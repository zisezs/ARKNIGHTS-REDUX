# ============================================================================
# 叙拉古 (SRC) · 开局事件链【叙拉古内战】的钩子（2026-09-19）
#
# 触发条件定稿：
#   1) on_daily_SRC 在该国存在的第一天跑一次，排 6 天延时 → 开局第 7 天弹事件 1。
#      用旗标保证整局只起一次链，不依赖具体日期（换书签也不用改）。
#   2) on_monthly 侦测 SR1 与 SR2 是否都已不存在；是则弹【改革派凯旋！】。
#
# 2026-09-19 修「改革派凯旋一次弹几十遍」：
#   原写法是 on_monthly 里套 every_country = { limit = { original_tag = SRC } … }。
#   on_monthly 并不保证只跑一次，套 every_country 等于把次数乘上场上现存国家的数量；
#   而且守卫用的全局旗 SRC_civil_war_done 要等玩家点了事件才置上，排队的 3 天窗口拦不住。
#   现在改用原版口径的 on_monthly_SRC —— 它就是"SRC 的国家作用域、每月一次"
#   （原版 on_monthly_GER 里直接写 add_offsite_building / has_dynamic_modifier，无任何循环），
#   再叠一个排队当场就置的 SRC_cw_flag_victory_queued 国家旗，保证整局只排一次。
#   注：common/on_actions/SRC_federal_on_actions.txt 的 on_monthly + every_country、
#   以及 YAN 的月度 tick 是同一个写法，可能有同样的乘算问题。
#
# 同名 on_action 跨文件是合并，所以新建本文件而不是改 SRC_federal_on_actions.txt
# （见 [[hoi4-focus-reward-modifiers]]）
# ============================================================================
on_actions = {

	on_daily_SRC = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_country_flag = SRC_cw_flag_chain_started
					}
				}
				set_country_flag = SRC_cw_flag_chain_started
				country_event = { id = src_cw.1 days = 6 }
			}
		}
	}

	on_monthly_SRC = {
		effect = {
			# ---- 人机兜底（2026-09-23）：叛军已被逐出全部领地、和会却还没吞并他们 ----
			# 不加这一段的话，"AI 必定打赢"就只能指望 AI 的议和诉求正好吞掉 SR1/SR2；
			# 判 SRC 首都被自己控制，是为了避免自己也已被打空时还去吞地。
			# 玩家一侧不受影响（照旧自己开和会），所以闸门是 is_ai = yes。
			if = {
				limit = {
					is_ai = yes
					has_country_flag = SRC_flag_civil_war
					OR = {
						has_war_with = SR1
						has_war_with = SR2
					}
					capital_scope = {
						is_controlled_by = ROOT
					}
				}
				SRC_crush_rebels_effect = yes
			}

			if = {
				limit = {
					has_country_flag = SRC_flag_civil_war
					NOT = {
						has_country_flag = SRC_cw_flag_victory_queued
					}
					SR1 = {
						exists = no
					}
					SR2 = {
						exists = no
					}
				}
				set_country_flag = SRC_cw_flag_victory_queued
				country_event = { id = src_cw.20 days = 3 }
			}
		}
	}
}
