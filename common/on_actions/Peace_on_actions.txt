				save_global_event_target_as = winning_country

			if = {
				limit = {
					event_target:losing_country = {
						any_allied_country = {
							has_war_together_with = event_target:losing_country
						}
					}
				}
				set_global_flag = losing_country_multiple
			}

			if = {
				limit = {
					event_target:winning_country = {
						any_allied_country = {
							has_war_together_with = event_target:winning_country
						}
					}
				}
				set_global_flag = winning_country_multiple
			}

			if = {

					# 	#ROOT是主要国家时，检测其他主要国家是否已投降
					# 	#等同于阵营内最后一个主要国家投降时，触发下列效果
					# 	#很遗憾，这个触发条件疑似永远无法满足，因为阵营内最后一个主要国家投降时不会触发on_capitulation
					# 	ROOT = {

					# 					is_major = yes
					# 				}
					# 				has_capitulated = yes
					# 			}
					# 		}
					# 	}

				if = {
					limit = {
						ROOT = {
							is_in_faction = yes
							any_allied_country = {
								is_major = yes
							}
						}
					}
					log = "war_goes_on"
					clr_global_flag = war_won
					set_global_flag = war_continuing
				}
				else = {
					log = "war_won"

				if = {
					limit = {

							original_tag = CLB

							original_tag = CFC
						}
					}
					every_owned_state = {
						limit = { is_core_of = CFC }
						add_core_of = CLB
						transfer_state_to = CFC
					}

						puppet = CFC
					}

				}

				if = {
					limit = {
						NOT = {
							has_global_flag = skip_default_capitulation
						}
					}
					log = "Default Capitulation"
					ROOT = {
						#所有敌国都分到占领的地
						#无阵营则各自分
						every_enemy_country = {
							limit = {
								is_in_faction = no
							}
							every_controlled_state = {
								limit = {
									is_owned_by = ROOT
								}
								PREV = {
									transfer_state = PREV
								}
							}
						}
						#有阵营的把占的地给阵营领袖
						every_enemy_country = {
							limit = {
								is_in_faction = yes
							}
							every_controlled_state = {
								limit = {
									is_owned_by = ROOT
								}
								PREV = {
									faction_leader = {
										transfer_state = PREV.PREV
									}
								}
							}
						}
