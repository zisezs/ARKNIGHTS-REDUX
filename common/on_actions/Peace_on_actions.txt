on_actions = {
	#### PEACE/WAR VICTORY ON_CAPITULATION BLOCK ####
	#### 割地机制
	#### PUT YOUR SHIT HERE
	#### 把你做的玩意放这
	# At the end of each global block, set the skip_default_capitulation GLOBAL flag
	# 要是想自定义个吞并效果，就在最后写上skip_default_capitulation这个global flag
	# Otherwise shit will just get annexed
	# 不然投降国家会被用这里面的默认方式割地
	# go away stinky vanilla peace
	# 不需要蠢驴做的狗屎割地系统

	# FROM = winner（战胜国）
	# ROOT = country being capitulated/loser（战败国）

	on_capitulation = {
		effect = {
			#### THIS IS INTENTIONAL DO NOT DELETE
			log = "[GetDateText]: FROM: [From.GetName] ROOT: [Root.GetName] on_capitulation"

			set_global_flag = show_peace_popup_alert
			sound_effect = "peace_summary_message"
			FROM = {
				save_global_event_target_as = winning_country
			} 
			ROOT = {
				save_global_event_target_as = losing_country
			}

			log = "[GetDateText]: winner/loser: FROM: [winning_country.GetName] ROOT: [losing_country.GetName] on_capitulation"

			if = {
				limit = {
					event_target:losing_country = {
						any_allied_country = {
							has_war_together_with = event_target:losing_country
						}
					}
				}
				set_global_flag = losing_country_multiple
			}

			if = {
				limit = {
					event_target:winning_country = {
						any_allied_country = {
							has_war_together_with = event_target:winning_country
						}
					}
				}
				set_global_flag = winning_country_multiple
			}

			#Annexation block
			#吞并代码区
			# this will only trigger ROOT (the loser) is in a faction and is NOT the only survivor of said faction
			# if ROOT is the only survivor then it will trigger the proper effects below
				
			if = {
				limit = {
					OR = {
						ROOT = { is_in_faction = no }
						ROOT = {
							all_allied_country = {
								if = {
									limit = {
										has_war_with = FROM
										is_major = yes
									}
									has_capitulated = yes
								}
							}
						}
					}
				}
				log = "war_won"
				clr_global_flag = war_continuing
				set_global_flag = war_won

				if = {
					limit = {
						FROM = {
							original_tag = CLB
						}
						ROOT = {
							original_tag = CFC
						}
					}
					every_owned_state = {
						limit = { is_core_of = CFC }
						add_core_of = CLB
						transfer_state_to = CFC
					}
					CLB = {
						puppet = CFC
					}
					CFC = {
						set_cosmetic_tag = Columbia_Front_Command_CFC
					}

					set_global_flag = skip_default_capitulation
				}

				else_if = {
					limit = {
						FROM = {
							original_tag = GUL
						}
						ROOT = {
							original_tag = VFA
						}
					}
					VFA = {
						white_peace = GUL
						every_owned_state = {
							limit = {
								NOT = {
									OR = {
										state = 501
										state = 859
									}
								}
							}
							transfer_state_to = GUL
						}
					}
					GUL = {
						set_country_flag = GUL_white_peace_with_VFA
					}

					set_global_flag = skip_default_capitulation
				}

				#高卢打赢莱塔尼亚：把它名下所有附庸（七城特别省、西方边境领、瓦隆公国这些殖民地，
				#以及后来册封的边境领/傀儡）一并吞进高卢，而不是让它们在本土被吞后原地独立。
				#顺序很关键：先吞附庸再吞本土 —— 主人一旦被并走，引擎当场把它的附庸放成独立国，
				#那时 is_subject_of / every_subject_country 都再也抓不到，属地就永远留在别人手里。
				else_if = {
					limit = {
						FROM = {
							original_tag = GUL
						}
						ROOT = {
							original_tag = LTN
						}
					}
					log = "GUL_exp: 莱塔尼亚向高卢投降，附庸随本土一并并入"
					LTN = {
						every_subject_country = {
							GUL = {
								annex_country = {
									target = PREV
									transfer_troops = no
								}
							}
						}
					}
					GUL = {
						white_peace = LTN
						annex_country = {
							target = LTN
							transfer_troops = no
						}
					}

					set_global_flag = skip_default_capitulation
				}

				else_if = {
					limit = {
						FROM = {
							OR = {
								original_tag = URS
								original_tag = UCM
								original_tag = ULB
								original_tag = UFS
								original_tag = KZC
								original_tag = BUS
								original_tag = BLT
								ursus_unifier = yes
							}
						}
						ROOT = {
							original_tag = KZM
						}
					}
					KZM = {
						every_owned_state = {
							limit = {
								OR = {
									is_Ursus_Kozacy_area = yes
									is_Ursus_Belaursnia_area = yes
									is_Ursus_Balti_area = yes
								}
							}
							transfer_state_to = FROM
						}
					}
					FROM = {
						white_peace = KZM
						puppet = KZM
					}

					set_global_flag = skip_default_capitulation
				}

				else_if = {
					limit = {
						FROM = {
							original_tag = KZM
						}
						ROOT = {
							OR = {
								original_tag = URS
								original_tag = UCM
								original_tag = ULB
								original_tag = UFS
								original_tag = KZC
								original_tag = BUS
								original_tag = BLT
								ursus_unifier = yes
							}
						}
					}
					ROOT = {
						every_owned_state = {
							limit = {
								OR = {
									is_Ursus_Kozacy_area = yes
									is_Ursus_Belaursnia_area = yes
									is_Ursus_Balti_area = yes
								}
							}
							transfer_state_to = KZM
						}
						white_peace = KZM
					}

					set_global_flag = skip_default_capitulation
				}

				else_if = {
					limit = {
						#怒号光明：战利品归卡兹戴尔红军（KAC）
						FROM = {
							OR = {
								original_tag = REU
								original_tag = KAC
							}
						}
						ROOT = {
							OR = {
								original_tag = KAZ
								original_tag = KAQ
							}
						}
						REU = {
							has_completed_focus = REU_Roar_for_Light
						}
						KAC = { exists = yes }
					}
					KAC = {
						annex_country = {
							target = ROOT
							transfer_troops = no
						}
					}

					set_global_flag = skip_default_capitulation
				}

				#叙拉古打赢莱塔尼亚：不许吞，改走【七丘的狼崽子们】的臣服线
				#主路是 on_weekly_SRC 的 80% 收网；这里兜住"崩盘太快、轮询漏掉窗口"的那一局，
				#否则下面的 Default Capitulation 会把莱塔尼亚直接并进叙拉古，傀儡就永远不出现了
				else_if = {
					limit = {
						FROM = {
							original_tag = SRC
						}
						ROOT = {
							original_tag = LTN
						}
						SRC = {
							NOT = { has_country_flag = SRC_exp_flag_ltan_vassal }
						}
					}
					log = "SRC_exp: 莱塔尼亚向叙拉古投降，改走臣服线（on_capitulation 兜底）"
					SRC = {
						SRC_exp_ltan_kneel_effect = yes
					}

					set_global_flag = skip_default_capitulation
				}

				#DEFAULT CAPITULATION
				#默认吞并
				if = {
					limit = { NOT = { has_global_flag = skip_default_capitulation } }
					log = "Default Capitulation"
					FROM = {
						every_enemy_country = {
							limit = {
								OR = {
									is_in_faction_with = ROOT
									is_subject_of = ROOT
									original_tag = ROOT
								}
							}
							every_enemy_country = {
								every_controlled_state = {
									limit = {
										is_owned_by = PREV.PREV
									}
									PREV = {
										transfer_state = PREV
									}
								}
							}
						}
						if = {
							limit = {
								ROOT = { exists = yes }
							}
							annex_country = {
								target = ROOT
								transfer_troops = no
							}
						}
					}
				}
			}
			else = {
				log = "war_goes_on"
				clr_global_flag = war_won
				set_global_flag = war_continuing
			}
			clr_global_flag = skip_default_capitulation
		}
	}
}
