on_actions = {
	# 西北都护府异变的月度兜底（引擎原生按国家钩子，ROOT = YNW）。
	# 做三件事，全部幂等：
	#   1) 终幕自愈：事件 .4 已送到但转换没做完 → 直接补做（选项点击被引擎丢掉时靠它）
	#   2) 链停滞兜底：起链后停滞计数过高（选项被丢掉、链卡住）→ 直接把终幕事件送来
	#   3) 宣战兜底：转换完成但仍未与大炎交战 → 每月补一次
	on_monthly_YNW = {
		effect = {
			# 停滞计数：起链时初始化，每步事件送达时由事件 immediate 归零
			if = {
				limit = { NOT = { has_variable = YNW_upheaval_stall } }
				set_variable = { YNW_upheaval_stall = 0 }
			}
			add_to_variable = { YNW_upheaval_stall = 1 }

			# 1) 终幕自愈
			if = {
				limit = {
					has_country_flag = YNW_upheaval_final_pending_flag
					NOT = { has_country_flag = YNW_upheaval_done_flag }
				}
				log = "YNW_upheaval: 月度自愈 —— 补做终幕转换"
				YNW_upheaval_become_sui_state = yes
			}

			# 2) 链停滞兜底：四个月没有任何一步送达，直接把终幕事件送来收尾
			if = {
				limit = {
					has_global_flag = YNW_upheaval_started
					NOT = { has_country_flag = YNW_upheaval_done_flag }
					NOT = { has_country_flag = YNW_upheaval_final_pending_flag }
					check_variable = { YNW_upheaval_stall > 4 }
				}
				log = "YNW_upheaval: 链停滞超过 4 个月，强制送终幕事件收尾"
				set_variable = { YNW_upheaval_stall = 0 }
				country_event = { id = yan_upheaval.4 days = 1 }
			}

			# 3) 宣战兜底：先退阵营（同阵营无法宣战），再对大炎宣战
			if = {
				limit = {
					has_country_flag = YNW_upheaval_done_flag
					is_in_faction = yes
				}
				leave_faction = yes
				log = "YNW_upheaval: 月度兜底 —— 脱离原阵营"
			}
			if = {
				limit = {
					has_country_flag = YNW_upheaval_done_flag
					is_subject = no
					YAN = { exists = yes }
					NOT = { has_war_with = YAN }
				}
				log = "YNW_upheaval: 月度兜底补宣战"
				declare_war_on = {
					target = YAN
					type = annex_everything
				}
			}
		}
	}
}
