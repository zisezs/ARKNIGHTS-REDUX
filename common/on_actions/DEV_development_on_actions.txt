######################
## 发展度共享国策树（DEV）on_actions
## 生成器 docs/gen_dev_development.cjs（勿手改本文件）
## on_startup / on_monthly_ZZZ 与其它文件的同名块合并执行；
## on_monthly_ZZZ 走 every_country 只跑一遍（同 00_development 的口径）。
######################

on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { DEV_has_dev_tree = yes }
				DEV_dev_tree_init = yes
			}
		}
	}
	on_monthly_ZZZ = {
		effect = {
			every_country = {
				limit = { DEV_has_dev_tree = yes }
				DEV_dev_tree_monthly = yes
			}
		}
	}
}
