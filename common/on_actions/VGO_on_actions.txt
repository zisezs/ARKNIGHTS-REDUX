on_actions = {
	on_monthly_VGO = {
		#革新军事理念
		effect = {
			if = {
				limit = {
					has_country_flag = VGO_innovating_military_concepts_flag
					check_variable = { VGO.VGO_military_army_attack_factor_dynamic_var < 0.05 }
				}
				add_to_variable = { VGO.VGO_military_army_attack_factor_dynamic_var = 0.05 }
			}
		}
		#西部地区工业化
		effect = {
			if = {
				limit = {
					has_country_flag = VGO_Industrialization_of_the_western_region_flag
					check_variable = { VGO.VGO_economy_production_factory_max_efficiency_factor_var < 0.2 }
				}
				add_to_variable = { VGO.VGO_economy_production_factory_max_efficiency_factor_var = 0.02 }
			}
		}
	}
}
