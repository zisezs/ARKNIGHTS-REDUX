on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					ROOT = {
						original_tag = REU
					}
				}
				FROM = {
					every_owned_state = {
						limit = {
							has_dynamic_modifier = {
								modifier = REU_infiltrated_state_3_modifier
								scope = REU
							}
						}
						force_enable_resistance = FROM
						start_resistance = FROM
						set_resistance = 65
						set_compliance = 0
					}
				}
			}
			else_if = {
				limit = {
					FROM = {
						original_tag = REU
					}
				}
				ROOT = {
					every_owned_state = {
						limit = {
							has_dynamic_modifier = {
								modifier = REU_infiltrated_state_3_modifier
								scope = REU
							}
						}
						force_enable_resistance = ROOT
						start_resistance = ROOT
						set_resistance = 55
						set_compliance = 0
					}
				}
			}
		}
	}
	on_startup = {
		effect = {
			# 我和柳德米拉的回忆录：决议系统开局即解锁
			# （Unlock 幂等：置 REU_ludmila_unlocked + 把正文拉到见面章节首节点 + 发一次解锁通知）
			REU = {
				REU_Ludmila_Unlock = yes
				# 将领特训：默认受训对象 = 索科洛夫，并把镜像副本/费用初始化（训练决议 cost = var:… 开局就得有值）
				REU_Sokolov_Training_Select_Sokolov = yes
			}
		}
	}
	on_war = {
		effect = {
			REU = {
				REU_Kazdel_Urgency_Check = yes
			}
		}
	}
	on_annex = {
		effect = {
			if = {
				limit = {
					FROM = { original_tag = URS }
				}
				every_country = {
					limit = {
						original_tag = REU
						NOT = { has_global_flag = REU_soviet_ursus_se_shown }
					}
					set_global_flag = REU_soviet_ursus_se_shown
					news_event = { id = reuse.1 }
				}
			}
		}
	}
	on_monthly = {
		effect = {
			every_country = {
				limit = {
					original_tag = REU
					NOT = { has_global_flag = REU_soviet_ursus_se_shown }
					NOT = { country_exists = URS }
				}
				set_global_flag = REU_soviet_ursus_se_shown
				news_event = { id = reuse.1 }
			}
		}
	}

	# 索科洛夫特训：每月兜一次费用变量（等级一变，核心效果已即时刷新；这里只作老档/异常保险）
	on_monthly_REU = {
		effect = {
			REU_Sokolov_Training_Refresh_Costs = yes
		}
	}
}
