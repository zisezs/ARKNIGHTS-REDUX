on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = { ROOT = { original_tag = REU } FROM = { original_tag = URS } }
						AND = { ROOT = { original_tag = URS } FROM = { original_tag = REU } }
					}
					NOT = { has_global_flag = REU_manifesto_of_the_war_shown }
					country_exists = REU
				}
				set_global_flag = REU_manifesto_of_the_war_shown
				REU = { country_event = { id = event_reu.126 } }
			}
			if = {
				limit = {
					OR = {
						AND = { ROOT = { original_tag = REU } FROM = { original_tag = URS } }
						AND = { ROOT = { original_tag = URS } FROM = { original_tag = REU } }
					}
					has_global_flag = REU_ursus_hell_mode
					NOT = { has_global_flag = REU_ursus_empowerment_cancelled }
				}
				every_country = {
					limit = {
						original_tag = URS
					}
					if = {
						limit = { has_idea = URS_Eastern_Dalia_Pacification }
						remove_ideas = URS_Eastern_Dalia_Pacification
					}
					add_timed_idea = {
						idea = URS_Eastern_Dalia_Pacification
						days = 365
					}
				}
				if = {
					limit = {
						country_exists = URS
						URS = { is_ai = yes }
					}
					URS = {
						REU_hell_ursus_reinforcements_effect = yes
					}
				}
			}
			if = {
				limit = {
					ROOT = {
						original_tag = REU
					}
				}
				FROM = {
					every_owned_state = {
						limit = {
							has_dynamic_modifier = {
								modifier = REU_infiltrated_state_3_modifier
								scope = REU
							}
						}
						force_enable_resistance = FROM
						start_resistance = FROM
						set_resistance = 65
						set_compliance = 0
					}
				}
			}
			else_if = {
				limit = {
					FROM = {
						original_tag = REU
					}
				}
				ROOT = {
					every_owned_state = {
						limit = {
							has_dynamic_modifier = {
								modifier = REU_infiltrated_state_3_modifier
								scope = REU
							}
						}
						force_enable_resistance = ROOT
						start_resistance = ROOT
						set_resistance = 55
						set_compliance = 0
					}
				}
			}
		}
	}

			REU = {
				REU_Ludmila_Unlock = yes
				# 将领特训：默认受训对象 = 索科洛夫，并把镜像副本/费用初始化（训练决议 cost = var:… 开局就得有值）
				REU_Sokolov_Training_Select_Sokolov = yes
				country_event = {
					id = event_reu.130
					days = 7
				}
			}
		}
	}
	on_war = {
		effect = {
			REU = {
				REU_Kazdel_Urgency_Check = yes
			}
		}
	}
	on_annex = {
		effect = {
			if = {
				limit = {
					FROM = { original_tag = URS }
				}
				every_country = {
					limit = {
						original_tag = REU
						NOT = { has_global_flag = REU_soviet_ursus_se_shown }
					}
					set_global_flag = REU_soviet_ursus_se_shown
					news_event = { id = reuse.1 }
				}
			}
		}
	}
	on_monthly = {
		effect = {
			every_country = {
				limit = {
					original_tag = REU
					NOT = { has_global_flag = REU_soviet_ursus_se_shown }
					NOT = { country_exists = URS }
				}
				set_global_flag = REU_soviet_ursus_se_shown
				news_event = { id = reuse.1 }
			}
		}
	}

	# 解放战争阶段：地狱模式下乌萨斯投降度过线即推进（133 另有苦难摇篮后 90 天兜底）
	on_daily_REU = {
		effect = {
			if = {
				limit = {
					has_global_flag = REU_ursus_hell_mode
					has_idea = REU_liberation_war_1
					any_country = {
						limit = { original_tag = URS }
						surrender_progress > 0.1
					}
				}
				country_event = { id = event_reu.133 }
			}
			if = {
				limit = {
					has_global_flag = REU_ursus_hell_mode
					has_idea = REU_liberation_war_2
					any_country = {
						limit = { original_tag = URS }
						surrender_progress > 0.3
					}
				}
				country_event = { id = event_reu.134 }
			}
			if = {
				limit = {
					NOT = { has_country_flag = REU_sarkaz_red_army_return_started }
					KAC = { exists = yes }
					has_character = REU_Conrad_Knecht
					any_neighbor_country = {
						original_tag = KAC
					}
				}
				activate_decision = REU_sarkaz_red_army_return
			}
			if = {
				limit = {
					is_ai = yes
					has_country_flag = East_Ursus_Upheaval_Enabled
					NOT = { KMG = { is_ai = no } }
					NOT = { UFE = { is_ai = no } }
					NOT = { has_idea = REU_ai_upheaval_might_idea }
				}
				add_ideas = REU_ai_upheaval_might_idea
			}
			if = {
				limit = {
					has_idea = REU_ai_upheaval_might_idea
					OR = {
						NOT = { is_ai = yes }
						NOT = { has_country_flag = East_Ursus_Upheaval_Enabled }
						KMG = { is_ai = no }
						UFE = { is_ai = no }
					}
				}
				remove_ideas = REU_ai_upheaval_might_idea
			}
		}
	}
