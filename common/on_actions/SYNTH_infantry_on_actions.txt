# ============================================================================
#  合成化步兵营 · 编队重组 —— 钩子
#  ⚠️ on_startup 默认作用域是 none（不是每国各一次）→ 必须自己 every_country。
#  ⚠️ 同名 on_action 跨文件是合并执行，这里只追加，不影响别处已有的 on_startup / on_monthly。
#  on_monthly 用「没有 synth_slot_1 才初始化」做兜底，覆盖内战建国等后成立的国家。
# ============================================================================

on_actions = {

	on_startup = {
		effect = {
			every_country = {
				limit = { exists = yes }
				SYNTH_init_country = yes
			}
		}
	}

	on_monthly = {
		effect = {
			SYNTH_init_country = yes
		}
	}
}
