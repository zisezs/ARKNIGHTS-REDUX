# ⚠️ on_monthly 是「每国各触发一次」→ 靠 ARF_Trade_Gate 的日闸门保证全局只结算一趟。

		effect = {
			every_country = {
				limit = { exists = yes }
				ARF_Trade_Init = yes
			}
			every_country = {
				limit = { exists = yes }
				ARF_Trade_Cold_Start = yes
			}
		}
	}

	on_monthly = {
		effect = {
			ARF_Trade_Gate = yes
		}
	}

}
