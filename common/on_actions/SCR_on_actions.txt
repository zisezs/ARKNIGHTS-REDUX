on_actions = {
	on_startup = {
		effect = {
			SCR = {
				SCR_Bacchanal_Init = yes
				country_event = {
					id = scr.1
					days = 2
				}
			}
		}
	}
	on_daily_SCR = {
		effect = {
			# 巡演冷却每日 -1
			if = {
				limit = { check_variable = { SCR_tour_cd > 0 } }
				add_to_variable = { SCR_tour_cd = -1 }
			}
			# 面板打开时每日重算一次静态地图标记与按钮可用状态
			if = {
				limit = { has_country_flag = scr_bac_open }
				add_to_variable = { SCR_bac_tick = 1 }
			}
		}
	}

	on_state_control_changed = {
		effect = {
			# 每征服一个地区，获得「灵感碎片」×1
			if = {
				limit = {
					ROOT = {
						original_tag = SCR
					}
					FROM.FROM = {
						NOT = { state = 867 }
						NOT = { is_owned_by = SCR }
					}
				}
				SCR = {
					add_to_variable = {
						var = SCR_inspiration
						value = 1
					}
				}
			}

			# 面板打开时，任何州控制权变化都立刻刷新面板内地图标记
			SCR = {
				if = {
					limit = { has_country_flag = scr_bac_open }
					add_to_variable = { SCR_bac_tick = 1 }
				}
			}
		}
	}
}
