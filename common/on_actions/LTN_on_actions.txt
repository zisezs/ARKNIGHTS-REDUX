# ============================================================================
# 莱塔尼亚 ·「金律乐章 · 调律台」—— on_actions
#
#  · on_startup     ：给 LTN 兜底初始化（老存档载入也安全，init 是幂等的）
#  · on_monthly_LTN ：月度脉冲 —— 余波倒计时 -30、全量重算、失范压顶
#
# ⚠️ 不要写 on_monthly + every_country{original_tag = LTN}：
#    本引擎里 on_monthly 按现存国家各触发一次（见 common/on_actions/YAN_on_actions.txt 的实证注释）。
#    tag 后缀的 on_monthly_LTN 才是「LTN 的国家作用域、每月一次」。
# ============================================================================
on_actions = {

	on_startup = {
		effect = {
			LTN = {
				LTN_law_effect_init = yes
				LTN_budget_effect_init = yes
			}
		}
	}

	on_monthly_LTN = {
		effect = {
			LTN_law_effect_tick = yes
			LTN_budget_effect_tick = yes
		}
	}

	# 「司辰的暴雨」：内战结束时司辰离场（ROOT = 胜方，FROM = 被吞并的败方）
	# 只认「对面挂着重塑之手旗标」这一条，双向都可能命中，所以两边各自判 flag 后再发事件。
	on_civil_war_end = {
		effect = {
			if = {
				limit = {
					original_tag = LTN
					has_country_flag = LTN_vt_vertin_present
					FROM = {
						has_country_flag = LTN_vt_remake_side
					}
				}
				country_event = { id = LTN_vertin.7 days = 3 }
			}
		}
	}
}
