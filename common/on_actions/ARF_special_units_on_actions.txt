on_actions = {

	# 特殊兵种招募令 + 高速战舰：开局给所有国家补上相关科技，
	# 让“招募令”“各型高速战舰”这些装备是已知可用的（否则 need 里挂着它们的支援连
	# 在编制器里会被静默隐藏）。
	# 用全局旗标守卫，避免 on_startup 若按国家各跑一次时重复遍历。
	on_startup = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_flag = arf_special_unit_token_tech_granted }
				}
				set_global_flag = arf_special_unit_token_tech_granted
				every_country = {
					set_technology = {
						arf_special_unit_token_tech = 1
						arf_dreadnought_warship_tech = 1
						arf_assault_warship_tech = 1
						arf_gunboat_warship_tech = 1
						arf_auxiliary_warship_tech = 1
					}
					# 高速战舰设计台：四个舰种的造价变量先补成裸装基础值，
					# 否则决议 available 里的 num_of_military_factories > arf_ws_<舰种>_cost
					# 会因为变量为空（=0）而对任何国家放行。
					set_variable = { arf_ws_gb_cost = 2 }
					set_variable = { arf_ws_as_cost = 5 }
					set_variable = { arf_ws_dr_cost = 10 }
					set_variable = { arf_ws_ax_cost = 2 }
				}
			}
		}
	}
}
