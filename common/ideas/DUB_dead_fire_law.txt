######################
## 深池·人力法案【死火复生】
##
## 设定：爱布拉娜是死火之主。她把自己的紫火分给战死者，死去的人便重新站起来列队——
##       代价照原作：「老去的迟迟不死，新生的频频夭折」，全国人口不再增长，工厂在血税下熄火。
## 槽位：mobilization_laws，与 common/ideas/_manpower.txt 同一条阶梯；
##       level = 0 排在 scraping_the_barrel（level 1）之下，是本模组最极端的一档。
## 门槛：无（2026-09-26 用户改判，花费 100 政治点，深池开局即可立）；
##       allowed = { original_tag = DUB } 只保证这条法在别的国家列表里不出现。
##       决议 TAR_rd_dead_fire 与国策 TAR_rd_war_bloodledger 仍置 DUB_dead_fire_unlocked，
##       但那面旗现在只是剧情记号，不再是本法的前提。
## loc：localisation/simp_chinese/ARF_DUB_l_simp_chinese.yml
######################
ideas = {
	mobilization_laws = {
		law = yes
		use_list_view = yes

		DUB_dead_fire_risen = {
			picture = DUB_dead_fire_risen

			cost = 100
			removal_cost = -1
			level = 0

			allowed = {
				original_tag = DUB
			}

			# 2026-09-26 用户改判：不设任何前提，深池开局即可在政治页直接立这条法。
			# （原先卡 DUB_dead_fire_unlocked ＋ 开战 ＋ 控制省份的 available 已整块删除；
			#   决议「饮火」与国策「血的账本」仍会置那个旗标，但它不再是这条法的门槛。）

			modifier = {
				conscription = 1.00			# 名册上不再有"不够格的人"
				monthly_population = -1.00		# 增长归零：老去的迟迟不死，新生的频频夭折
				industrial_capacity_factory = -0.40
				industrial_capacity_dockyard = -0.40
				production_speed_buildings_factor = -0.25
				consumer_goods_factor = 0.10		# 血税
				training_time_factor = -0.35		# 不问出身，即刻成军
				mobilization_speed = 10.00			# 动员速度 +1000%：番号一喊，队列自己长出来
				political_power_cost = -1.00		# 政治点数消耗 -100% ⇒ 0：签字不再花人心
				supply_consumption_factor = -0.30		# 补给消耗 -30%：尸体不吃饭，也不喝水
				army_org_factor = -0.15			# 行尸没有队列
				army_morale_factor = 0.10		# 也不知痛
				resistance_target = 0.10
				stability_factor = -0.15
				war_support_factor = 0.20
			}

			cancel_if_invalid = no

			# 法案一过，死者就有了番号：解锁深池专属营级兵种「死火行尸」
			# （common/units/DUB_dead_fire_units.txt ＋ common/technologies/DUB_dead_fire_tech.txt）
			on_add = {
				set_technology = { dub_dead_fire_walker_tech = 1 }
				# 立了这条法，深池就只认死火行尸师这一档编制
				country_lock_all_division_template = yes
				set_division_template_lock = {
					division_template = "死火行尸师"
					is_locked = no
				}
			}

			on_remove = {
				# 废法就把编制权还回去（否则锁会永久留在国家上）
				country_lock_all_division_template = no
			}

			ai_will_do = {
				# AI 必须肯立这条法：行尸科技挂在它的 on_add 上，不立法就没有行尸师。
				# 深池专属（allowed 已锁 original_tag = DUB），所以这里给高分不会影响别国。
				base = 50

				modifier = {
					add = 100
					has_war = yes
				}
				modifier = {
					add = 50
					has_offensive_war = yes
				}
				modifier = {
					factor = 0
					not = { has_manpower = 150000 }
				}
			}
		}
	}
}
