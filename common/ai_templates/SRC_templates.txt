# ==============================================================================
# 叙拉古 (SRC) 专属 AI 编制模板（2026-09-23）
#
# 为什么要有这份文件（这才是"人机叙拉古练不出强兵"的真因）：
#   common/ai_templates/generic.txt 的步兵阶梯是按大国产能标定档的 ——
#   第 3 档要 num_of_military_factories > 30、第 4 档要 > 45。
#   叙拉古开局总共 20 座军工，内战分裂后手里只剩 11 座（守旧派拿走 9 座，
#   见 common/scripted_effects/SRC_civilwar_effects.txt 的 585/578/561/510/532/498），
#   也就是说 AI 叙拉古**永远卡在 infantry_002（9 步 + 1 炮团）这一档**，
#   再多陆军经验也换不出强编制。本文件把门槛整体下移到本国可达区间（7/12/18 座）。
#
# 口径与约束：
#   · 一个 role 一个国家只能命中一份 role 级条目，所以 generic.txt 里
#     infantry/militia/armor 三块的 blocked_for 已经加上 SRC（互斥靠那一行）。
#   · 团与支援连全部在 common/units/ 里核过名，且闸门按「哪一条科技真的 enable_subunits
#     它」来写，不是按编号大小猜（common/technologies/ 实测）：
#       artillery_0     -> artillery_brigade / artillery
#       antitank_0      -> anti_tank_brigade / anti_tank
#       antiair_0       -> anti_air_brigade  / anti_air
#       light_tank_0 / medium_tank_0 -> light_armor / medium_armor
#       tech_recon      -> recon / mot_recon（叙拉古开局已研）
#       tech_engineers / tech_signal_company / tech_field_hospital /
#       tech_logistics_company / tech_military_police -> 各支援连，开局**没有**，
#       所以任何一档都不能把开局师建成依赖它们的样子（民兵第 1 档因此用 recon）。
#   · 叙拉古的兵是「警察 + 民兵」底子（城邦执法队、沃尔西尼防暴部队都是民兵团），
#     所以第 1 档留得很低，让它开局/内战期就有目标，而不是空转配额。
#   · 装甲占比交给 common/ai_strategy/SRC.txt 的 role_ratio 往下压
#     （默认 20 对 11 座军工的国家是虚高需求），这里只提供能落地的小规模装甲档。
# ==============================================================================

# ---- 步兵：四档阶梯，门槛按叙拉古本国产能下移 ----
infantry_SRC = {

	available_for = {
		SRC
	}

	role = infantry

	# 叙拉古 army 的主体就是步兵，换代优先级给满
	upgrade_prio = {
		base = 5

		modifier = {
			factor = 2
			num_of_military_factories > 10
		}
		modifier = {
			factor = 2
			has_war = yes
		}
	}

	# ---- 第 1 档：国民卫队雏形（纯步兵，缺炮也能先拉开架子）----
	SRC_infantry_001 = {

		reinforce_prio = 0
		custom_icon = 7

		upgrade_prio = {
			base = 1

			modifier = {
				factor = 6
				num_of_military_factories < 6
			}
			modifier = {
				factor = 0
				num_of_military_factories > 8
			}
		}

		target_template = {

			support = {
				engineer = 1
				recon = 1
			}

			regiments = {
				infantry = 8
			}
		}

		replace_at_match = 1.2
		replace_with = SRC_infantry_002
		target_min_match = 0.6
	}

	# ---- 第 2 档：内战期的制式师（8 步 + 1 炮团，支援连补通信）----
	SRC_infantry_002 = {

		reinforce_prio = 0
		custom_icon = 7

		upgrade_prio = {
			base = 1

			modifier = {
				factor = 5
				num_of_military_factories > 7
			}
			modifier = {
				factor = 0
				NOT = { has_tech = artillery_0 }
			}
		}

		can_upgrade_in_field = {
			has_equipment = { artillery_equipment > 400 }
		}

		target_template = {

			support = {
				engineer = 1
				recon = 1
				artillery = 1
				signal_company = 1
			}

			regiments = {
				infantry = 8
				artillery_brigade = 1
			}
		}

		replace_at_match = 1.5
		replace_with = SRC_infantry_003
		target_min_match = 0.6
	}

	# ---- 第 3 档：联邦正规师（9 步 + 2 炮 + 1 反坦克，团属支援连起步）----
	SRC_infantry_003 = {

		upgrade_prio = {
			base = 1

			modifier = {
				factor = 6
				num_of_military_factories > 12
			}
			modifier = {
				factor = 4
				date > 1100.1.1
			}
			modifier = {
				factor = 0
				NOT = { has_tech = antitank_0 }
			}
		}

		target_template = {

			support = {
				engineer = 1
				recon = 1
				artillery = 1
				anti_air = 1
				signal_company = 1
				field_hospital = 1
			}

			regimental_support = {
				artillery_support = 1
				anti_tank_support = 1
			}

			regiments = {
				infantry = 9
				artillery_brigade = 2
				anti_tank_brigade = 1
			}
		}

		replace_at_match = 1.5
		replace_with = SRC_infantry_004
		target_min_match = 0.5
	}

	# ---- 第 4 档：莱昂图索近卫师（10 步 + 2 炮 + 反坦克/防空，火力型满配）----
	SRC_infantry_004 = {

		upgrade_prio = {
			base = 1

			modifier = {
				factor = 8
				num_of_military_factories > 18
			}
			modifier = {
				factor = 5
				date > 1101.1.1
			}
			modifier = {
				factor = 2
				has_tech = antitank_2
			}
			modifier = {
				factor = 0
				NOT = { has_tech = antitank_0 }
			}
			modifier = {
				factor = 0
				NOT = { has_tech = antiair_0 }
			}
		}

		target_template = {

			support = {
				engineer = 1
				recon = 1
				artillery = 1
				anti_air = 1
				anti_tank = 1
				signal_company = 1
				field_hospital = 1
				logistics_company = 1
			}

			regimental_support = {
				artillery_support = 2
				anti_tank_support = 1
				anti_air_support = 1
			}

			regiments = {
				infantry = 10
				artillery_brigade = 2
				anti_tank_brigade = 1
				anti_air_brigade = 1
			}
		}
	}
}
