#  · 作用域：国家。等级取国家侧镜像变量 REU_sok_<skill>_lv。

	OR = {
		AND = { check_variable = { REU_sok_scout_lv = 0 } has_political_power > 9 }
		AND = { check_variable = { REU_sok_scout_lv = 1 } has_political_power > 14 }
		AND = { check_variable = { REU_sok_scout_lv = 2 } has_political_power > 19 }
		AND = { check_variable = { REU_sok_scout_lv = 3 } has_political_power > 24 }
		AND = { check_variable = { REU_sok_scout_lv = 4 } has_political_power > 29 }
		AND = { check_variable = { REU_sok_scout_lv = 5 } has_political_power > 34 }
		AND = { check_variable = { REU_sok_scout_lv = 6 } has_political_power > 39 }
		AND = { check_variable = { REU_sok_scout_lv = 7 } has_political_power > 44 }
		AND = { check_variable = { REU_sok_scout_lv = 8 } has_political_power > 49 }
		AND = { check_variable = { REU_sok_scout_lv = 9 } has_political_power > 54 }
	}
}
