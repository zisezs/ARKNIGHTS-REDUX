# 费奥多尔在位。
#
# 设计稿对这条线的每一格都写了「费奥多尔是现任国家领袖，否则国策取消」，
# 所以 available 全挂这个。
#
# 为什么不是光写 has_country_leader：乌萨斯国内**有两个人**带
# conservative_subtype 的 country_leader 角色（URS_Feodor_Vladimirovich_Ryurik
# 与 URS_Islam_Vitte，见 common/characters/URS.txt 第 11/25 行），而执政党是
# conservatism——引擎在多领袖同意识形态时选谁当领袖，我没法在机外验证。
# 一旦选错，整条改革线会全部点不动，代价太大，所以加一条兜底：
# mod 自己认「元首是沙皇」（history 里的 firstleader_tsar）且费奥多尔还在国内，
# 就放行。语义比设计稿略宽（费奥多尔被架空但没被赶走时仍算在位），但不会死锁。
#
# 注意：has_country_leader 的 ruling_only 默认就是 yes，不写也一样。
URS_dr_fyodor_in_power = {
	OR = {
		has_country_leader = {
			character = URS_Feodor_Vladimirovich_Ryurik
			ruling_only = yes
		}
		AND = {
			has_country_flag = firstleader_tsar
			has_character = URS_Feodor_Vladimirovich_Ryurik
		}
	}
}

# 维特还在政府里（设计稿：「维特是现任国家议员」，否则问题之源取消）。
# 开局 history 就挂着他 second_in_command 的二把手 advisor。
URS_dr_vitte_in_office = {
	OR = {
		has_idea = URS_Islam_Vitte_second_in_command
		has_idea = URS_Islam_Vitte_advisor
	}
}
