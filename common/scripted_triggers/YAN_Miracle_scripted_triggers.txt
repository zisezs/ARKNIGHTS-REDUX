# 大炎 · 奇迹工程 —— 脚本触发器（生成物）
# 选择态由 YAN_mir_sel 决定，详情页共用一套控件

YAN_Mir_CanAdd = {
	NOT = { check_variable = { YAN_mir_stat_sel = 3 } }
	check_variable = { YAN_mir_work_total < 60 }
}

YAN_Mir_CanSub = {
	check_variable = { YAN_mir_work_sel > 0 }
}

YAN_Mir_CanStart = {
	NOT = { check_variable = { YAN_mir_stat_sel = 3 } }
	OR = {
		check_variable = { YAN_mir_stat_sel = 1 }
		check_variable = { YAN_mir_work_sel > 0 }
	}
}

YAN_Mir_CanRush = {
	check_variable = { YAN_mir_rush_cd < 1 }
	check_variable = { YAN_mir_stat_sel = 1 }
	has_political_power > 49
}
