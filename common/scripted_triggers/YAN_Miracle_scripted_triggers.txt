# ---- 州 scope：当前工程（YAN_mir_proj）的选址要求，逐条「与」 ----
YAN_Mir_Proj_Req = {
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 1 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			is_capital = no
			infrastructure < 5
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 2 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			OR = {
				industrial_complex > 2
				arms_factory > 2
			}
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 3 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			is_coastal = no
			infrastructure > 1
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 4 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			OR = {
				is_coastal = yes
				naval_base > 0
			}
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 5 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			arms_factory > 1
			infrastructure > 2
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 6 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			is_coastal = yes
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 7 compare = equals }
		}
		AND = {
			OR = {
				is_capital = yes
				AND = {
					is_core_of = ROOT
					industrial_complex > 3
				}
			}
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 8 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			is_in_home_area = no
		}
	}
	if = {
		limit = {
			check_variable = { ROOT.YAN_mir_proj = 9 compare = equals }
		}
		AND = {
			is_core_of = ROOT
			infrastructure > 4
			OR = {
				industrial_complex > 3
				arms_factory > 3
			}
		}
	}
}
