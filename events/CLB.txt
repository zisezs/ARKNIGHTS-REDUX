		name = elec_prim_clb.7.b
		trigger = {
			OR = {
				has_government = progressivism

					has_government = liberalism
					NOT = {
						OR = {
							has_country_flag = CLB_liberalism_in_A_flag
							has_country_flag = CLB_liberalism_in_B_flag
						}
					}
				}
			}
		}

		set_country_flag = CLB_Graham_nominated
		set_country_flag = CLB_Chaney_nominated_vice
	}

	option = {
		name = elec_prim_clb.7.c

		trigger = {
			has_government = socialism
			NOT = {
				has_country_flag = CLB_Charles_and_conservatist #切尼已成为副总统候选人
			}
		}

		set_country_flag = CLB_Chaney_nominated
		set_country_flag = CLB_Alessandro_nominated_vice
	}

	option = {
		name = elec_prim_clb.7.e

		trigger = {
			OR = {
				has_government = socialism
				AND = {
					has_government = liberalism
					has_country_flag = CLB_liberalism_in_A_flag
				}
			}
		}

		set_country_flag = CLB_Charles_nominated

		if = {
			limit = {
				has_country_flag = CLB_Charles_and_populist
			}
			set_country_flag = CLB_JD_nominated_vice
		}
		else_if = {
			limit = {
				has_country_flag = CLB_Charles_and_conservatist
			}
			set_country_flag = CLB_Chaney_nominated_vice
		}
		else_if = {
			limit = {
				has_country_flag = CLB_Charles_and_progressivist
			}
			set_country_flag = CLB_Eichenfeller_nominated_vice
		}
	}

	option = {
		name = elec_prim_clb.7.f

		trigger = {
			has_country_leader = {
				character = CLB_Dwayne_Charles
				ruling_only = yes
			}
		}

		set_country_flag = CLB_Charles_nominated
		set_country_flag = CLB_Chaney_nominated_vice
	}

	option = {
		name = elec_prim_clb.7.g

		trigger = {
			has_country_leader = {

				ruling_only = yes
			}
		}

	}

#94年选举日
country_event = {
	id = election_clb.1
	title = election_clb.1.t
	desc = election_clb.1.d

	immediate = {
		hidden_effect = {
			CLB_election_start = yes
			every_owned_state = {
				CLB_state_vote = yes
			}
			CLB_election_results = yes

			force_update_map_mode = {
				mapmode = election_map_mode
			}

			# 莱茵生命已启动夺权：选票照记，结果由十科合议厅内部推选定案
			if = {
				limit = {
					has_country_flag = CLB_rhine_coup_started
				}
				set_variable = { win_election_parties = 0 }
				set_country_flag = CLB_flag_internal_selection
				country_event = {
					id = inauguration_clb.29
					days = 5
				}
			}
			else = {
				CLB_change_leader = yes
			}
		}
	}

	is_triggered_only = yes

	option = {
		name = election_clb.1.rhine
		trigger = {
			has_country_flag = CLB_flag_internal_selection
		}

		CLB_clear_nomination = yes #清除所有提名flag
	}

	option = {
		name = election_clb.1.a
		trigger = {
			check_variable = { win_election_parties = 1 }
			has_country_flag = CLB_A_win_election_flag
		}

		CLB_clear_nomination = yes #清除所有提名flag
	}

	option = {
		name = election_clb.1.b
		trigger = {
			check_variable = { win_election_parties = 1 }
			has_country_flag = CLB_B_win_election_flag
		}

		CLB_clear_nomination = yes #清除所有提名flag
	}

	option = {
		name = election_clb.1.c
		trigger = {
			check_variable = { win_election_parties = 1 }
			NOT = {
				has_country_flag = CLB_communism_in_A_flag
			}
			has_country_flag = CLB_communism_win_election_flag
		}

		CLB_clear_nomination = yes #清除所有提名flag
	}

	option = {
		name = election_clb.1.e
		trigger = {
			check_variable = { win_election_parties = 1 }
			NOT = {
				has_country_flag = CLB_socialism_in_A_flag
			}
			has_country_flag = CLB_socialism_win_election_flag
		}

		CLB_clear_nomination = yes #清除所有提名flag
	}

	option = {
		name = election_clb.1.f
		trigger = {
			check_variable = { win_election_parties = 1 }
			NOT = {
				OR = {
					has_country_flag = CLB_liberalism_in_A_flag
					has_country_flag = CLB_liberalism_in_B_flag
				}
			}
			has_country_flag = CLB_liberalism_win_election_flag
		}

		CLB_clear_nomination = yes #清除所有提名flag
	}

	option = {
		name = election_clb.1.g
		trigger = {
			check_variable = { win_election_parties = 1 }
			NOT = {
				has_country_flag = CLB_fascism_in_B_flag
			}
			has_country_flag = CLB_fascism_win_election_flag
		}

		CLB_clear_nomination = yes #清除所有提名flag
	}

	option = {
		name = election_clb.1.h

		trigger = {
			check_variable = { win_election_parties > 1 }
		}
	}
}
