add_namespace = TAR_rp

######################
## 塔拉复国 · 战争带终幕事件（定都与定位两步落地）
## 调用方：common/scripted_effects/DUB_redeem_diplo_effects.txt
##   TAR_dip_capital_event_effect → TAR_rp.20（定都）
##   TAR_dip_crown_settle_effect  → TAR_rp.21（谁坐在哪）
## 作用域：ROOT ＝ DUB（深池）。两个事件都是 is_triggered_only，不发 major，不越作用域。
## 容器变量是裸名（tar_dip_*），不带容器前缀——与国策 completion_reward 同一口径。
######################

# 把首都写在哪个名字上
country_event = {
	id = TAR_rp.20
	title = TAR_rp.20.t
	desc = TAR_rp.20.d
	picture = GFX_report_event_Generic_Palace_Hall

	is_triggered_only = yes

	# 伦蒂尼姆：敌人的心脏
	option = {
		name = TAR_rp.20.a
		if = {
			limit = { 434 = { is_owned_and_controlled_by = ROOT } }
			set_capital = { state = 434 remember_old_capital = no }
		}
		else = {
			custom_effect_tooltip = TAR_rp.20_notyet_tt
		}
		add_to_variable = { var = tar_dip_fear value = 0.10 }
		add_to_variable = { var = tar_dip_legit value = 0.10 }
		add_stability = { war_party = no factor = -0.05 }
		set_country_flag = TAR_rd_dip_cap_lundenric
		custom_effect_tooltip = TAR_rp.20_a_tt
	}

	# 纳斯尔沙：红龙的巢，也是我们的坟
	option = {
		name = TAR_rp.20.b
		if = {
			limit = { 593 = { is_owned_and_controlled_by = ROOT } }
			set_capital = { state = 593 remember_old_capital = no }
		}
		else = {
			custom_effect_tooltip = TAR_rp.20_notyet_tt
		}
		add_to_variable = { var = tar_dip_legit value = 0.15 }
		add_political_power = 60
		set_country_flag = TAR_rd_dip_cap_nasirsha
	}

	# 小丘郡：哪儿也不去，就从烧过的地方开始
	option = {
		name = TAR_rp.20.c
		add_to_variable = { var = tar_dip_trade value = 0.15 }
		add_to_variable = { var = tar_dip_unrest value = -0.10 }
		add_stability = { war_party = yes factor = 0.05 }
		set_country_flag = TAR_rd_dip_cap_hills
		custom_effect_tooltip = TAR_rp.20_c_tt
	}
}

# 谁坐在哪：冠冕、账本、桌子
country_event = {
	id = TAR_rp.21
	title = TAR_rp.21.t
	desc = TAR_rp.21.d
	picture = GFX_report_event_Generic_Courthouse

	is_triggered_only = yes

	# 交给台前那条红龙
	option = {
		name = TAR_rp.21.a
		add_country_leader_role = {
			character = TAR_Eblana_Dublinn
			promote_leader = yes
			country_leader = {
				ideology = dictatorship_subtype
				expire = "2000.1.1"
				id = -1
			}
		}
		add_to_variable = { var = tar_dip_fear value = 0.15 }
		add_war_support = 0.10
		set_country_flag = TAR_rd_dip_crown_eglana
		custom_effect_tooltip = TAR_rp.21_a_tt
	}

	# 交给另一位：军队姓塔拉，但不姓谁
	option = {
		name = TAR_rp.21.b
		set_politics = {
			ruling_party = authoritarianism
			elections_allowed = no
		}
		add_country_leader_role = {
			character = TAR_Loughshinny_Dublinn
			promote_leader = yes
			country_leader = {
				ideology = stratocracy_subtype
				expire = "2000.1.1"
				id = -1
			}
		}
		add_to_variable = { var = tar_dip_crown value = 0.15 }
		add_political_power = -40
		set_country_flag = TAR_rd_dip_crown_lough
	}

	# 三样东西分给三个人：谁也别想独坐
	option = {
		name = TAR_rp.21.c
		add_to_variable = { var = tar_dip_table value = 0.20 }
		add_to_variable = { var = tar_dip_legit value = 0.10 }
		add_stability = { war_party = yes factor = 0.08 }
		set_country_flag = TAR_rd_dip_crown_council
		custom_effect_tooltip = TAR_rp.21_c_tt
	}
}
