# 编号约定：
#   .1         决议弹出的整肃选择窗

#   .3         引蛇出洞：定日子（文书在此发出，起事排期也在此落下；最后一项反悔）
#   .10–.15    六路叛军起义告急文书（.10 UCM / .11 ULB / .12 UFS / .13 KZC / .14 BUS / .15 BLT）
#   .20        起义落地执行（择定的那一天；结算全写在 immediate，不依赖玩家点击）
#   .30        乌萨斯对卡西米尔宣战的突袭预警（发给卡西米尔，2 天前）
#   .31        正式宣战执行（并把备战度结算成战时精神）
#   .40        自动获胜结算
#   .42        平叛报捷（每平定一路发一次，给备战度与缴获）
#   .50        战争爆发新闻。旧 worldnews_kzm.9 的正文写的是「卡西米尔抢先发动攻势」，
#              与本链「帝国先手」相反，故另起一份；.9 随旧入口一起停用。
#
# 引擎坑：title / desc 都必须写成 key = 值。原第 23、69 行把 desc 键裸写了一行，
#   结果是这两个选择窗连正文都没有（error.log: Unexpected token: ursus_kzmwar.1.d）。
# 选项一律显式写 name，别漏 —— 第 4 个选项不写 name 就会 auto 成 ursus_kzmwar.X.d，
#   与 desc 抢同一个 loc 键。
######################################################

# ── 起义落地：择定的那一天执行 ──
# 结算全部放在 immediate，事件是否被点掉都不影响战争发生
country_event = {
	id = ursus_kzmwar.20
	title = ursus_kzmwar.20.t
	desc = ursus_kzmwar.20.d
	picture = GFX_report_event_Generic_explosion

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			clr_country_flag = URS_kzmwar_erupt_scheduled
			if = {
				limit = { has_country_flag = URS_kzmwar_brewing_ucm }
				clr_country_flag = URS_kzmwar_brewing_ucm
				URS_kzmwar_erupt_ucm = yes
			}
			if = {
				limit = { has_country_flag = URS_kzmwar_brewing_ulb }
				clr_country_flag = URS_kzmwar_brewing_ulb
				URS_kzmwar_erupt_ulb = yes
			}
			if = {
				limit = { has_country_flag = URS_kzmwar_brewing_ufs }
				clr_country_flag = URS_kzmwar_brewing_ufs
				URS_kzmwar_erupt_ufs = yes
			}
			if = {
				limit = { has_country_flag = URS_kzmwar_brewing_kzc }
				clr_country_flag = URS_kzmwar_brewing_kzc
				URS_kzmwar_erupt_kzc = yes
			}
			if = {
				limit = { has_country_flag = URS_kzmwar_brewing_bus }
				clr_country_flag = URS_kzmwar_brewing_bus
				URS_kzmwar_erupt_bus = yes
			}
			if = {
				limit = { has_country_flag = URS_kzmwar_brewing_blt }
				clr_country_flag = URS_kzmwar_brewing_blt
				URS_kzmwar_erupt_blt = yes
			}
		}
	}

	option = {
		name = ursus_kzmwar.20.a
	}
}
