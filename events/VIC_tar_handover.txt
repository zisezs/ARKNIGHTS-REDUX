######################
## 塔拉复国 · 移交与会晤（承载国：DUB 深池）
##
## 分工：TAR ＝ 维多利亚威灵顿公爵领（走它自己的铁公爵⇄红龙线）；
##       DUB ＝ 深池，塔拉复国主体（复国树 DUB_restoration、决议组、精神、事件都在这里）。
##
## 已核事实：
##   · 塔拉诸省 18 个全部带 Tara_state_flag；621 是 DUB 的核心省（开局 owner 原本挂在 VIC，已改）
##   · TAR/DUB 开局都可能是零领土 tag → exists = no，必须先给地再谈合并
##   · 换玩家控制国＝目标国 scope 内 change_tag_from = ROOT（原版 BBA_Italy.txt:5685 同法）
##   · 内战闸门＝全局旗标 VIC_civil_war_start；本组事件只在内战前/会晤窗口内生效
##   · 不用 major = yes（会作用域泄漏给所有大国）
##   · annex_country 不给核心：先手动 transfer_state_to + add_core_of
######################
add_namespace = vic_tar

# ───────────────── vic_tar.1 维多利亚要不要把塔拉交出去 ─────────────────
country_event = {
	id = vic_tar.1
	title = vic_tar.1.t
	desc = vic_tar.1.d
	picture = GFX_report_event_Generic_Manor_Hall

	fire_only_once = yes
	mean_time_to_happen = { days = 90 random_days = 60 }

	trigger = {
		NOT = { has_global_flag = VIC_civil_war_start }
		NOT = { has_country_flag = TAR_handled_flag }
		NOT = { original_tag = DUB }
		custom_trigger_tooltip = {
			tooltip = vic_tar.1.req
			any_owned_state = { has_state_flag = Tara_state_flag }
		}
	}

	# a. 交给深池，并且由玩家执掌
	option = {
		name = vic_tar.1.a
		custom_effect_tooltip = vic_tar.1.a.tt

		hidden_effect = {
			set_country_flag = TAR_handled_flag
			every_owned_state = {
				limit = { has_state_flag = Tara_state_flag }
				transfer_state_to = DUB
				add_core_of = DUB
			}
			DUB = {
				set_capital = { state = 621 remember_old_capital = no }
				# 不先给科技就 load_oob，DUB_Start 的营级单位全是 active = no → 静默空壳
				inherit_technology = ROOT
				set_research_slots = 4
				load_oob = DUB_Start
				set_country_flag = TAR_independence_flag
				set_stability = 0.55
				set_war_support = 0.70
				add_political_power = 150
				set_cosmetic_tag = Kingdom_of_Tara
				change_tag_from = ROOT
			}
		}
		ai_chance = { factor = 0 }
	}

	# b. 只放它独立，玩家留在原位
	option = {
		name = vic_tar.1.b
		custom_effect_tooltip = vic_tar.1.b.tt
		add_political_power = -40
		add_war_support = -0.05

		hidden_effect = {
			set_country_flag = TAR_handled_flag
			every_owned_state = {
				limit = { has_state_flag = Tara_state_flag }
				transfer_state_to = DUB
				add_core_of = DUB
			}
			DUB = {
				set_capital = { state = 621 remember_old_capital = no }
				inherit_technology = ROOT
				set_research_slots = 4
				load_oob = DUB_Start
				set_country_flag = TAR_independence_flag
				set_stability = 0.5
				set_war_support = 0.6
				add_political_power = 80
				set_cosmetic_tag = Kingdom_of_Tara
			}
		}
		ai_chance = { factor = 35 }
	}

	# c. 冠冕不许有缺口
	option = {
		name = vic_tar.1.c
		add_political_power = 60
		add_stability = 0.05
		set_country_flag = TAR_handled_flag
		custom_effect_tooltip = vic_tar.1.c.tt

		every_owned_state = {
			limit = { has_state_flag = Tara_state_flag }
			add_dynamic_modifier = { modifier = TAR_garrison_burden_dm }
		}
		ai_chance = { factor = 65 }
	}
}

# ───────────────── vic_tar.3 与威灵顿会晤：吞并威灵顿公爵领 ─────────────────
# 触发：深池侧决议 TAR_rd_gastriel_pact（common/decisions/DUB_redeem_Decisions.txt）
# 口径变更：不再是「两支塔拉对等合并」——公爵领作为一个政权结束，它的顾问与将领全体转到我名下。
# 沿用三条实机验证的硬规矩：先转地给核心再吞；annex_country 不给核心；零领土 tag 也能进作用域。
country_event = {
	id = vic_tar.3
	title = vic_tar.3.t
	desc = vic_tar.3.d
	picture = GFX_report_event_Generic_Cold_Forest

	fire_only_once = yes
	is_triggered_only = yes

	# a. 赴约：把公爵领连同它的人才一起接进塔拉
	option = {
		name = vic_tar.3.a
		custom_effect_tooltip = vic_tar.3.a.tt

		hidden_effect = {
			if = {
				limit = { TAR = { exists = yes } }
				if = {
					limit = { has_war_with = TAR }
					white_peace = TAR
				}
				TAR = {
					every_owned_state = {
						limit = { has_state_flag = Tara_state_flag }
						transfer_state_to = ROOT
						add_core_of = ROOT
					}
					every_character = {
						set_nationality = ROOT
					}
				}
				annex_country = { target = TAR transfer_troops = yes }
			}
			else = {
				# 公爵领还只是个零领土名分：地无从可取，人照单全收
				TAR = {
					every_character = {
						set_nationality = ROOT
					}
				}
			}
			# 把「我们脚下、原属威灵顿名下」的省补成我们的核心（annex_country 只给地不给核心）
			TAR_dip_core_held_effect = yes
			set_country_flag = TAR_pact_done
			add_political_power = 80
			add_stability = 0.10
			add_war_support = 0.05
		}
		ai_chance = { factor = 100 }
	}
}
