			every_state = {
				limit = { has_state_flag = gul_rebel_in_this_area_flag }
				save_event_target_as = gul_rebel_in_this_area
			}
		}
	}
	
	is_triggered_only = yes
	
	option = {
		name = event_gul.1.a
		GUL = {
			set_temp_variable = { vic_ocup_fac_num_tool = 0 }
			subtract_from_temp_variable = { vic_ocup_fac_num_tool = event_target:gul_rebel_in_this_area.vic_ocup_civs_num_var }
			subtract_from_temp_variable = { vic_ocup_fac_num_tool = event_target:gul_rebel_in_this_area.vic_ocup_mils_num_var }
			custom_effect_tooltip = GUL_broken_economy_idea_tt
			add_to_variable = {
				GUL.vic_ocup_fac_num_var = vic_ocup_fac_num_tool
				tooltip = vic_ocup_fac_num_tt
			}
		}
		custom_effect_tooltip = generic_skip_one_line_tt
		event_target:gul_rebel_in_this_area = {
			custom_effect_tooltip = things_had_happened_oh_shit_tt
			effect_tooltip = {
				custom_effect_tooltip = gul_rebel_in_this_area_tt
			}
			clr_state_flag = gul_rebel_in_this_area_flag
			if = {
				limit = { check_variable = { vic_ocup_civs_num_var > 0 } }
				VIC = {
					set_temp_variable = { industrial_complex_sub = event_target:gul_rebel_in_this_area.vic_ocup_civs_num_var }
					set_temp_variable = { industrial_complex_var = 0 }
					subtract_from_temp_variable = { industrial_complex_var = industrial_complex_sub }
					add_offsite_building = { type = industrial_complex level = industrial_complex_var }
				}
			}
			if = {
				limit = { check_variable = { vic_ocup_mils_num_var > 0 } }
				VIC = {
					set_temp_variable = { arms_factory_sub = event_target:gul_rebel_in_this_area.vic_ocup_mils_num_var }
					set_temp_variable = { arms_factory_var = 0 }
					subtract_from_temp_variable = { arms_factory_var = arms_factory_sub }
					add_offsite_building = { type = arms_factory level = arms_factory_var }
				}
			}
			
		}
		ai_chance = {
			base = 10
		}
	}
}
