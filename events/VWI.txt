#温德米尔是否死亡
country_event = {
	id = event_vwi.1
	title = event_vwi.1.t
	desc = event_vwi.1.d

	is_triggered_only = yes

	option = {
		name = event_vwi.1.a
		
		country_event = {
			id = event_vwi.2
		}

		ai_chance = {
			factor = 7
		}
	}

	option = {
		name = event_vwi.1.b

		if = {
			limit = {
				power_balance_value = {
					id = VWI_power_balance
					value > 0
				}
			}
			complete_national_focus = VWI_Terra_bless_the_Queen

			activate_advisor = VWI_Anfilis_Windermere_second_in_command
		}
		else = {
			complete_national_focus = VWI_keep_calm_and_carry_on
		}

		ai_chance = {
			factor = 3
		}
	}
}

#温德米尔的继任者
country_event = {
	id = event_vwi.2
	title = event_vwi.2.t
	desc = event_vwi.2.d

	is_triggered_only = yes

	option = {
		name = event_vwi.2.a

		trigger = {
			NOT = {
				AND = {
					power_balance_value = {
						id = VWI_power_balance
						value > 0
					}

					has_character = VIC_Alexandrina_Vina_Victoria
				}
			}
		}

		retire_character = VWI_Anfilis_Windermere

		promote_character = VWI_Delphine_Windermere

		if = {
			limit = {
				NOT = {
					power_balance_value = {
						id = VWI_power_balance
						value > 0
					}
				}
				has_character = VIC_Alexandrina_Vina_Victoria
			}
			set_country_leader_description = {
				desc = VWI_Delphine_Windermere_royal_desc
			}

			complete_national_focus = VWI_keep_calm_and_carry_on
		}
		else = {
			set_country_leader_description = {
				desc = VWI_Delphine_Windermere_republic_desc
			}

			complete_national_focus = VWI_we_will_meet_again
		}
	}

	option = {
		name = event_vwi.2.b

		trigger = {
			power_balance_value = {
				id = VWI_power_balance
				value > 0
			}

			has_character = VIC_Alexandrina_Vina_Victoria
		}

		retire_character = VWI_Anfilis_Windermere

		promote_character = VIC_Alexandrina_Vina_Victoria

		set_country_leader_description = {
			desc = VIC_Alexandrina_Vina_Victoria_side_Windermere_desc
		}

		complete_national_focus = VWI_Terra_bless_the_Queen

		activate_advisor = VWI_Delphine_Windermere_second_in_command
	}
}

country_event = {
	id = event_vwi.10008
	title = event_vwi.10008.t
	desc = event_vwi.10008.d
	picture = GFX_report_event_VIC_queen_Vina_friends

	is_triggered_only = yes

	option = {
		name = event_vwi.10008.a

		set_country_flag = VWI_Terra_bless_the_Queen

		if = {
			limit = {
				power_balance_value = {
					id = VWI_power_balance
					value > 0
				}
			}
			set_politics = {
				ruling_party = despotism
				elections_allowed = no
			}			
		}
		else = {
			set_politics = {
				ruling_party = authoritarianism
				elections_allowed = no
				long_name = VWI_party_name_kings_party_long
				name = VWI_party_name_kings_party
			}
			promote_character = {
				character = VIC_Alexandrina_Vina_Victoria
				ideology = dual_monarchy_subtype
			}
		}

		set_country_leader_description = {
			desc = VIC_Alexandrina_Vina_Victoria_side_Windermere_desc
		}

		set_cosmetic_tag = VWI_model_army

		clr_country_flag = firstleader_commander_in_chief
		set_country_flag = firstleader_queen

		clr_country_flag = secondleader_leading_body
		set_country_flag = secondleader_prime_minister

		activate_advisor = VWI_Delphine_Windermere_second_in_command
	}
}
