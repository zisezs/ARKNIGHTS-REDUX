# 宣政司改革：国策点开之后，中枢第一次把手伸进十九司的账册
country_event = {
	id = yan_civ.1
	title = yan_civ.1.t
	desc = yan_civ.1.d
	picture = GFX_report_event_Generic_Courthouse

	is_triggered_only = yes

	option = {
		name = yan_civ.1.a
		# 先清账：改革进度往前走两步
		add_to_variable = { YAN_civ_rf_progress = 2 }
		custom_effect_tooltip = YAN_civ_reform_progress_p2_tt
	}

	option = {
		name = yan_civ.1.b
		# 先定章程：稳住朝局，换一笔政治点数
		add_political_power = 50
	}
}
