######################
## 塔拉王国 ·《挽歌燃烧殆尽》事件组（namespace: TAR_ea，20 个）
##
## 依据文本：act41side《挽歌燃烧殆尽》EA-ST-1…EA-ST-3（原文核对见
##   docs/剧情拆解_维多利亚与塔拉/13_塔拉_挽歌燃烧殆尽.md），
##   以及 docs/TAR_塔拉复国方略_红龙线细化.md 第十节的 R1—R7 风险清单。
##
## 触发：pulse 事件链（fire_only_once + mean_time_to_happen），不用 on_actions。
##   根闸门 = TAR_independence_flag（events/TAR.txt:11 与 events/VIC_tar_handover.txt 都写）
##   后续每一格都读前一格在 immediate 里置下的 TAR_ea_ev<N>，分支条件用 OR。
##
## 口径：
##   · 只用国内效果（PP／稳定／战争支持度／人力／定时精神／州级修正／旗标），
##     不写结盟、宣战、关系修正、市场准入、科技解锁（工程约定）。
##   · 选项 loc 一律 .oa/.ob/.oc，把 .d 留给 desc（避免 X.d 撞键）。
##   · 与复国树（TAR_rd.*）互不覆盖：本组只读它的旗标与变量，
##     不替玩家打开 TAR_rd_relic_office / TAR_rd_bell_read 等国策闸门。
##   · 引用的精神全部定义在 common/ideas/TAR_elegy_ideas.txt，
##     州级「黑雾」在 common/dynamic_modifiers/TAR_elegy_dynamic_modifiers.txt。
##
## 脊柱：1 归乡 → 2 征料 → 3 见紫火 → 4 祛火 → 5 报产量 → 6 验间谍 → 7 无人救的孩子
##   → 8 珍宝择主 → 9 送仆臣上路 → 10 粮车归来 → 11 拖像 → 12 英雄之死
##   → 13 只开一条缝 → 14 还火 → 15 祭台 → 16 六百年的账 → 17 拖棺
##   → 18 城门举火 → 19 夜尽之时 → 20 封底纪事
######################

# ══════ 1 涉过寒夜（EA-ST-1：石高原野的流民抵达纳斯尔纱城门外）══════
country_event = {
	id = TAR_ea.1
	title = TAR_ea.1.t
	desc = TAR_ea.1.d
	picture = GFX_report_event_Generic_Cold_Forest

	fire_only_once = yes
	mean_time_to_happen = { days = 18 random_days = 14 }

	trigger = {
		tag = TAR
		has_country_flag = TAR_independence_flag
	}

	immediate = { set_country_flag = TAR_ea_ev1 }

	# a. 先安顿在聚落外围，城里还没准备好
	option = {
		name = TAR_ea.1.oa
		add_political_power = 25
		add_stability = 0.03
		if = {
			limit = { has_idea = TAR_ea_granary_short }
			remove_ideas = TAR_ea_granary_short
		}
		add_timed_idea = { idea = TAR_ea_granary_short days = 180 }
		set_country_flag = TAR_ea_refugees_outside
		custom_effect_tooltip = TAR_ea.1.oa.tt
		ai_chance = { factor = 40 }
	}
	# b. 开门进城，塔拉人不等塔拉人
	option = {
		name = TAR_ea.1.ob
		add_political_power = -15
		add_war_support = 0.04
		add_manpower = 45000
		set_country_flag = TAR_ea_refugees_in
		custom_effect_tooltip = TAR_ea.1.ob.tt
		random_owned_controlled_state = {
			limit = { is_capital = yes }
			add_dynamic_modifier = { modifier = TAR_ea_fog_dm }
		}
		ai_chance = { factor = 35 }
	}
	# c. 收下那枚哨笛，把这条家规记下来

		name = TAR_ea.1.oc
