	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = LTN_budget.1.oa
		ai_chance = { factor = 40 }
		add_political_power = -50
		add_to_variable = { LTN_budget_sup = 15 }
		custom_effect_tooltip = LTN_budget_sup_tt
	}
	option = {
		name = LTN_budget.1.ob
		ai_chance = { factor = 35 }
		add_political_power = -50
		add_to_variable = { LTN_budget_sup = 10 }

		if = { limit = { NOT = { check_variable = { LTN_conspiracy > 20 } } } set_variable = { LTN_conspiracy = 20 } }
		custom_effect_tooltip = LTN_conspiracy_tt
		custom_effect_tooltip = LTN_budget_sup_tt

		name = LTN_budget.1.oc

		name = LTN_budget.11.ob

			NOT = {

					has_war_with = GUL
