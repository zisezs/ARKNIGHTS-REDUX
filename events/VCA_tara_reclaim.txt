	title = vca_tara.1.t
	desc = vca_tara.1.d
	picture = GFX_report_event_Generic_Palace_Hall

	is_triggered_only = yes
	fire_only_once = yes

		}
	}

	option = {
		name = vca_tara.1.a
		add_war_support = 0.05
		add_stability = -0.03
	}
}
