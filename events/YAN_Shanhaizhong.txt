# 【动员乡勇】每有新势力竖旗就发一次：首都编 5 支四步乡勇
country_event = {
	id = yan_shz.22
	title = yan_shz.22.t
	desc = yan_shz.22.d
	picture = GFX_report_event_Generic_Barracks

	is_triggered_only = yes

	option = {
		name = yan_shz.22.a
		# 模板只建一次；create_unit 落首都（capital_scope），owner = ROOT = 大炎
		if = {
			limit = { NOT = { has_country_flag = YAN_shz_militia_template_flag } }
			set_country_flag = YAN_shz_militia_template_flag
			division_template = {
				name = "乡勇"
				is_locked = yes
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 1 y = 0 }
					infantry = { x = 1 y = 1 }
				}
			}
		}
		capital_scope = {
			create_unit = {
				division = "name = \"乡勇\" division_template = \"乡勇\" start_experience_factor = 0 start_equipment_factor = 0.5"
				owner = ROOT
				count = 5
			}
		}
		custom_effect_tooltip = YAN_shz_militia_tt
	}
}
