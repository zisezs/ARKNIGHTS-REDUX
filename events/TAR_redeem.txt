##   .6 ← 决议「召回圣物」remove_effect（被清洗派系的怨恨）

## 口径：只用国内效果（PP／稳定／修正符变量／旗标／师团与建筑），

# ── .3 空棺的三种官方说法 ──

	title = TAR_rd.3.t
	desc = TAR_rd.3.d

	option = { # 神迹：红龙复生
		name = TAR_rd.3.a

		name = TAR_rd.3.b

		add_war_support = 0.05

		custom_effect_tooltip = TAR_rd.3.c.tt

	}

# ── .5 收回借出的火：染火反弹 ──
country_event = {
	id = TAR_rd.5
	title = TAR_rd.5.t
	desc = TAR_rd.5.d
	picture = GFX_report_event_Generic_explosion
	is_triggered_only = yes

	trigger = {
		has_variable = TAR_rd_fire_stock
		TAR_rd_fire_stock > 3
	}

	option = { # 办一场公共还火
		name = TAR_rd.5.a
		add_political_power = -35
		add_stability = 0.05
		custom_effect_tooltip = TAR_rd.5.a.tt
		if = {
			limit = { has_variable = TAR_rd_fire_stock }
			add_to_variable = { TAR_rd_fire_stock = -2 }
		}
		ai_chance = { factor = 60 }
	}
	option = { # 让火继续烧
		name = TAR_rd.5.b
		add_war_support = 0.10
		add_stability = -0.10
		if = {
			limit = { has_variable = TAR_rd_grudge }
			add_to_variable = { TAR_rd_grudge = 1 }
		}
		custom_effect_tooltip = TAR_rd.5.b.tt
		ai_chance = { factor = 40 }
	}
}

# ── .6 召回圣物：被清洗者的账 ──

	}
}
